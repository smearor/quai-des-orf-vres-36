9/// Errors a [`crate::MoveRequest`] can fail validation with.
10#[derive(Debug, Clone, Copy, PartialEq, Eq, Error)]
11pub enum MoveError {
12    /// The game is not in a phase that accepts moves.
13    #[error("moves are not accepted in phase {actual} (expected {expected})")]
14    WrongPhase {
15        /// The phase that accepts moves.
16        expected: GamePhase,
17        /// The phase the game is actually in.
18        actual: GamePhase,
19    },
20    /// A different player's turn is active.
21    #[error("not the turn of {actual}; expected {expected}")]
22    NotPlayersTurn {
23        /// The role that may move now.
24        expected: Role,
25        /// The role that tried to move.
26        actual: Role,
27    },
28    /// No map edge connects the two stations via this transport.
29    #[error("no {transport} route between {from} and {to}")]
30    NoRoute {
31        /// Start station of the attempted hop.
32        from: StationId,
33        /// Attempted destination.
34        to: StationId,
35        /// Transport mode the route was attempted with.
36        transport: Transport,
37    },
38    /// The player has no ticket of the requested kind left.
39    #[error("{role} has no {ticket} tickets left")]
40    TicketExhausted {
41        /// The role that ran out of tickets.
42        role: Role,
43        /// The exhausted ticket kind.
44        ticket: TicketKind,
45    },
46    /// The destination already holds another detective.
47    #[error("station {0} is already occupied by pawn {1}")]
48    DestinationOccupied(StationId, PawnId),
49    /// The station does not exist on the map.
50    #[error("unknown station {0}")]
51    UnknownStation(StationId),
52    /// This ticket kind is reserved for Fugitive.
53    #[error("{0} tickets are Fugitive-only")]
54    FugitiveOnlyTicket(TicketKind),
55    /// A batobus route needs Fugitive's black ticket.
56    #[error("batobus routes require the black ticket")]
57    BatobusRequiresBlack,
58}
