6/// Shape of a move within one turn.
7#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
8#[serde(rename_all = "lowercase")]
9pub enum MoveKind {
10    /// A single hop paid with one transport ticket.
11    Single(Hop),
12    /// Fugitive's double ticket: two consecutive hops in one turn.
13    Double {
14        /// First hop of the double move.
15        first: Hop,
16        /// Second hop, starting where `first` ended.
17        second: Hop,
18    },
19}
20
21impl MoveKind {
22    /// The hop-count shape of this move.
23    pub fn shape(&self) -> MoveShape {
24        match self {
25            Self::Single(_) => MoveShape::Single,
26            Self::Double { .. } => MoveShape::Double,
27        }
28    }
29
30    /// The hops of this move in order (one for `Single`, two for
31    /// `Double`).
32    pub fn hops(&self) -> Vec<Hop> {
33        match self {
34            Self::Single(hop) => vec![*hop],
35            Self::Double { first, second } => vec![*first, *second],
36        }
37    }
38
39    /// The station the move ends on.
40    pub fn destination(&self) -> qdo36_map::StationId {
41        match self {
42            Self::Single(hop) => hop.to,
43            Self::Double { second, .. } => second.to,
44        }
45    }
46}
