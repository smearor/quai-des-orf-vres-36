1use serde::Deserialize;
2use serde::Serialize;
3use std::fmt;
4use thiserror::Error;
5
6/// The number of detectives hunting Fugitive.
7///
8/// Scotland Yard plays with 2-5 detectives (the sixth seat is Fugitive
9/// himself).
10#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
11#[serde(try_from = "u8", into = "u8")]
12pub struct DetectiveCount(u8);
13
14impl DetectiveCount {
15    /// Fewest detectives a game supports.
16    pub const MIN: u8 = 2;
17    /// Most detectives a game supports.
18    pub const MAX: u8 = 5;
19
20    /// The numeric count.
21    pub const fn count(self) -> u8 {
22        self.0
23    }
24}
25
26/// Error for an out-of-range detective count.
27#[derive(Debug, Clone, Copy, PartialEq, Eq, Error)]
28#[error("detective count {0} is out of range {min}..={max}", min = DetectiveCount::MIN, max = DetectiveCount::MAX)]
29pub struct DetectiveCountError(pub u8);
30
31impl TryFrom<u8> for DetectiveCount {
32    type Error = DetectiveCountError;
33
34    fn try_from(count: u8) -> Result<Self, Self::Error> {
35        if (Self::MIN..=Self::MAX).contains(&count) {
36            Ok(Self(count))
37        } else {
38            Err(DetectiveCountError(count))
39        }
40    }
41}
42
43impl From<DetectiveCount> for u8 {
44    fn from(count: DetectiveCount) -> Self {
45        count.0
46    }
47}
48
49impl fmt::Display for DetectiveCount {
50    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
51        write!(f, "{}", self.0)
52    }
53}

55#[cfg(test)]
56mod tests {
57    use super::*;
58
59    #[test]
60    fn accepts_range_and_rejects_outside() {
61        for count in DetectiveCount::MIN..=DetectiveCount::MAX {
62            assert!(DetectiveCount::try_from(count).is_ok());
63        }
64        for count in [0, 1, 6, 255] {
65            assert!(matches!(DetectiveCount::try_from(count), Err(DetectiveCountError(c)) if c == count));
66        }
67    }
