1use crate::TicketKind;
2use serde::Deserialize;
3use serde::Serialize;

5/// One player's ticket inventory.
6#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize)]
7pub struct TicketCounts {
8    /// Remaining taxi tickets.
9    pub taxi: u8,
10    /// Remaining bus tickets.
11    pub bus: u8,
12    /// Remaining metro tickets.
13    pub metro: u8,
14    /// Remaining black tickets (Fugitive only).
15    pub black: u8,
16    /// Remaining double tickets (Fugitive only).
17    pub double: u8,
18}
19
20impl TicketCounts {
21    /// `u8::MAX` stands for an effectively unlimited stock
22    /// (Fugitive's travel tickets).
23    pub const UNLIMITED: u8 = u8::MAX;
24    /// The starting inventory of a detective (classic Scotland Yard:
25    /// 10 taxi, 8 bus, 4 metro).
26    pub const DETECTIVE_START: Self = Self {
27        taxi: 10,
28        bus: 8,
29        metro: 4,
30        black: 0,
31        double: 0,
32    };
33    /// Fugitive's starting pool: unlimited travel plus 5 black and
34    /// 2 double tickets.
35    pub const FUGITIVE_START: Self = Self {
36        taxi: Self::UNLIMITED,
37        bus: Self::UNLIMITED,
38        metro: Self::UNLIMITED,
39        black: 5,
40        double: 2,
41    };
42
43    /// The default detective stock - for serde `default` attributes.
44    pub const fn detective_start() -> Self {
45        Self::DETECTIVE_START
46    }
47
48    /// The default fugitive stock - for serde `default` attributes.
49    pub const fn fugitive_start() -> Self {
50        Self::FUGITIVE_START
51    }
52
53    /// The remaining count of a ticket kind.
54    pub fn get(&self, kind: TicketKind) -> u8 {
55        match kind {
56            TicketKind::Taxi => self.taxi,
57            TicketKind::Bus => self.bus,
58            TicketKind::Metro => self.metro,
59            TicketKind::Black => self.black,
60            TicketKind::Double => self.double,
61        }
62    }
63
64    /// Adds one ticket of `kind` - spent detective tickets feed
65    /// Fugitive's pool (saturates at [`Self::UNLIMITED`]).
66    pub fn gain(&mut self, kind: TicketKind) {
67        let slot = match kind {
68            TicketKind::Taxi => &mut self.taxi,
69            TicketKind::Bus => &mut self.bus,
70            TicketKind::Metro => &mut self.metro,
71            TicketKind::Black => &mut self.black,
72            TicketKind::Double => &mut self.double,
73        };
74        *slot = slot.saturating_add(1);
75    }
76
77    /// Decrements a ticket kind by one; returns `false` when none left.
78    pub fn spend(&mut self, kind: TicketKind) -> bool {
79        let slot = match kind {
80            TicketKind::Taxi => &mut self.taxi,
81            TicketKind::Bus => &mut self.bus,
82            TicketKind::Metro => &mut self.metro,
83            TicketKind::Black => &mut self.black,
84            TicketKind::Double => &mut self.double,
85        };
86        if *slot == 0 {
87            false
88        } else {
89            *slot -= 1;
90            true
91        }
92    }
93}

100    fn spend_decrements_and_blocks_at_zero() {
101        let mut counts = TicketCounts { taxi: 1, ..Default::default() };
102        assert!(counts.spend(TicketKind::Taxi));
103        assert_eq!(counts.get(TicketKind::Taxi), 0);
104        assert!(!counts.spend(TicketKind::Taxi));
