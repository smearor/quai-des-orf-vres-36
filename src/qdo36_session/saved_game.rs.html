1//! `SavedGame` - the persistent snapshot for autosave/resume.
2
3use serde::Deserialize;
4use serde::Serialize;
5
6use crate::GameConfig;
7use crate::GameSession;
8use crate::GameState;
9use crate::SessionError;
10
11/// A complete session snapshot: config, current state and the
12/// undo/redo stacks, so a restored game continues with full history.
13///
14/// Note: `state` carries Fugitive's hidden position - a save file is
15/// private storage by definition and must be handled like the shield
16/// zone, never as a public projection.
17#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
18pub struct SavedGame {
19    /// The configuration the game was started with.
20    pub config: GameConfig,
21    /// The live state at save time.
22    pub state: GameState,
23    /// Undo stack at save time.
24    pub past: Vec<GameState>,
25    /// Redo stack at save time.
26    pub future: Vec<GameState>,
27}
28
29impl SavedGame {
30    /// Captures a session for persistence.
31    pub fn snapshot(session: &GameSession) -> Self {
32        Self {
33            config: session.config.clone(),
34            state: session.state.clone(),
35            past: session.past.clone(),
36            future: session.future.clone(),
37        }
38    }
39
40    /// Restores a session; the stored config is re-validated so a
41    /// hand-edited or corrupt save cannot seed an invalid game.
42    pub fn restore(self) -> Result<GameSession, SessionError> {
43        self.config.validate()?;
44        Ok(GameSession {
45            config: self.config,
46            state: self.state,
47            past: self.past,
48            future: self.future,
49        })
50    }
51
52    /// Serializes to TOML.
53    pub fn to_toml(&self) -> Result<String, SessionError> {
54        toml::to_string_pretty(self).map_err(|err| SessionError::Save(err.to_string()))
55    }
56
57    /// Parses from TOML.
58    pub fn from_toml(text: &str) -> Result<Self, SessionError> {
59        toml::from_str(text).map_err(|err| SessionError::Load(err.to_string()))
60    }
61}

63#[cfg(test)]
64mod tests {
65    use super::*;
66    use qdo36_map::GameMap;
67    use qdo36_map::Transport;
68    use qdo36_material::TicketKind;
69    use qdo36_moves::MoveKind;
70    use qdo36_moves::MoveRequest;
71    use rand::SeedableRng;
72    use rand::rngs::StdRng;
73
74    /// A session three moves into a game.
75    fn played() -> GameSession {
76        let map = GameMap::bundled().unwrap_or_else(|_| panic!("bundled map must load"));
77        let mut session = GameSession::deal(GameConfig::for_map(map.meta()), &map, &mut StdRng::seed_from_u64(36)).unwrap_or_else(|_| panic!("deal"));
78        session.begin().unwrap_or_else(|_| panic!("begin"));
79        for _ in 0..3 {
80            let hop = session
81                .legal_destinations(&map, session.state.active)
82                .first()
83                .copied()
84                .unwrap_or_else(|| panic!("a legal move"));
85            let ticket = match hop.transport {
86                Transport::Taxi => TicketKind::Taxi,
87                Transport::Bus => TicketKind::Bus,
88                Transport::Metro => TicketKind::Metro,
89                Transport::Batobus | Transport::Underground => TicketKind::Black,
90            };
91            session
92                .request_move(
93                    &map,
94                    &MoveRequest {
95                        actor: session.state.active,
96                        ticket,
97                        kind: MoveKind::Single(hop),
98                    },
99                )
100                .unwrap_or_else(|err| panic!("{err}"));
101        }
102        session
103    }
104
105    #[test]
106    fn toml_roundtrip_restores_the_session() {
107        let session = played();
108        let saved = SavedGame::snapshot(&session);
109        let toml = saved.to_toml().unwrap_or_else(|_| panic!("ser"));
110        let parsed = SavedGame::from_toml(&toml).unwrap_or_else(|_| panic!("de"));
111        let restored = parsed.restore().unwrap_or_else(|_| panic!("restore"));
112        assert_eq!(restored.state, session.state, "state survives the roundtrip");
113        assert_eq!(restored.past.len(), session.past.len(), "undo stack survives");
114    }
115
116    #[test]
117    fn restore_rejects_a_corrupt_config() {
118        let session = played();
119        let mut saved = SavedGame::snapshot(&session);
120        saved.config.max_rounds = qdo36_material::RoundNumber::new(0);
121        assert!(saved.restore().is_err(), "invalid config must fail restore");
122    }
123
124    #[test]
125    fn restore_rejects_corrupt_toml() {
126        assert!(SavedGame::from_toml("not = [toml").is_err());
127    }
