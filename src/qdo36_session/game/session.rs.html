6/// One running (or finished) game with its undo history.
7///
8/// Mirrors tangibles-chess's `GameSession`: `past`/`future` stacks
9/// implement full undo/redo - a move pushes the pre-move state onto
10/// `past` and clears `future`.
11#[derive(Debug, Clone, Serialize, Deserialize)]
12pub struct GameSession {
13    /// Configuration fixed at setup.
14    pub config: GameConfig,
15    /// Current state.
16    pub state: GameState,
17    /// Undo stack - one entry per committed move.
18    pub past: Vec<GameState>,
19    /// Redo stack - repopulated by undo.
20    pub future: Vec<GameState>,
21}
22
23impl GameSession {
24    /// Whether undo is possible right now (also gated by
25    /// `config.allow_undo` and `InProgress`/replay semantics).
26    pub fn can_undo(&self) -> bool {
27        self.config.allow_undo && !self.past.is_empty()
28    }
29
30    /// Whether redo is possible right now.
31    pub fn can_redo(&self) -> bool {
32        self.config.allow_undo && !self.future.is_empty()
33    }
34}
