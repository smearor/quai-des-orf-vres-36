1//! `GameSession` - the rules engine: deal, move validation, turn
2//! cycle, win conditions, undo/redo.
3//!
4//! A command-centric state machine: every input path (pawn, tap,
5//! keypad, MCP) produces a [`MoveRequest`] and passes the same
6//! validation pipeline - phase, turn, ticket, route, occupancy -
7//! before [`GameSession::apply`] mutates state. `past`/`future` give
8//! chess-style undo/redo with a physical [`crate::RestorationPlan`].
9//!
10//! The engine splits across a directory module (chess-style):
11//! [`validate`] owns the ordered checks, [`apply`] owns mutation and
12//! follow-up transitions; this file keeps the public surface.

43/// One running (or finished) game with its undo history.
44///
45/// The session owns every invariant: the hidden `fugitive_pos` only
46/// leaves through projections, `allow_undo` gates `undo`/`redo`, and
47/// turn order is enforced by `active` - never by callers.
48#[derive(Debug, Clone, Serialize, Deserialize)]
49pub struct GameSession {
50    /// Configuration fixed at setup.
51    pub config: GameConfig,
52    /// Current state.
53    pub state: GameState,
54    /// Undo stack - one entry per committed move.
55    pub past: Vec<GameState>,
56    /// Redo stack - repopulated by undo.
57    pub future: Vec<GameState>,
58}
59
60impl GameSession {
61    /// Deals a new game: validates the config, samples start stations
62    /// without replacement, seats detectives in pawn order and hides
63    /// Fugitive on the last dealt station. The session stays in
64    /// [`GamePhase::Deal`] until [`Self::begin`].
65    ///
66    /// `rng` is injected so deals are reproducible - tests and the
67    /// balancing simulator seed it deterministically.
68    pub fn deal(config: GameConfig, map: &GameMap, rng: &mut impl Rng) -> Result<Self, SessionError> {
69        config.validate()?;
70        let seats = SeatCount::for_game(config.detectives);
71        let starts: Vec<StationId> = map.start_stations().map(|station| station.id).collect();
72        let dealt: Vec<StationId> = starts.sample(rng, seats.as_usize()).copied().collect();
73        if dealt.len() < seats.as_usize() {
74            return Err(SessionError::TooFewStartStations {
75                needed: seats,
76                available: SeatCount::of_len(starts.len()),
77            });
78        }
79        // Deal layout: one station per detective, the last one hides
80        // Fugitive.
81        let positions: BTreeMap<PawnId, StationId> = config.detectives.pawns().zip(dealt.iter().copied()).collect();
82        let Some(&fugitive_pos) = dealt.last() else {
83            return Err(SessionError::TooFewStartStations {
84                needed: seats,
85                available: SeatCount::of_len(0),
86            });
87        };
88        let mut tickets = BTreeMap::new();
89        for pawn in config.detectives.pawns() {
90            tickets.insert(Role::Detective(pawn), config.detective_stock);
91        }
92        tickets.insert(Role::Fugitive, config.fugitive_stock);
93        Ok(Self {
94            config,
95            state: GameState {
96                positions,
97                fugitive_pos,
98                tickets,
99                round: RoundNumber::new(1),
100                active: Role::Fugitive,
101                phase: GamePhase::Deal,
102                log: GameLog::default(),
103                outcome: None,
104            },
105            past: Vec::new(),
106            future: Vec::new(),
107        })
108    }
109
110    /// Leaves the Deal phase; Fugitive moves first (round 1).
111    pub fn begin(&mut self) -> Result<(), MoveError> {
112        if self.state.phase != GamePhase::Deal {
113            return Err(MoveError::WrongPhase {
114                expected: GamePhase::Deal,
115                actual: self.state.phase,
116            });
117        }
118        self.state.phase = GamePhase::InProgress;
119        Ok(())
120    }
121
122    /// Validates and applies a move - the single entry every input
123    /// path funnels through. Validation runs in a fixed order so the
124    /// reported [`MoveError`] is deterministic:
125    /// phase → turn → ticket → route → occupancy → apply.
126    pub fn request_move(&mut self, map: &GameMap, request: &MoveRequest) -> Result<MoveOutcome, MoveError> {
127        self.validate_phase()?;
128        self.validate_turn(request.actor)?;
129        self.validate_ticket(request)?;
130        self.validate_hops(map, request)?;
131        Ok(self.apply(map, *request))
132    }
133
134    /// The stations `role` could legally reach this turn - for UI
135    /// move hints and the balancing simulator.
136    pub fn legal_destinations(&self, map: &GameMap, role: Role) -> Vec<Hop> {
137        let Some(from) = self.position_of(role) else { return Vec::new() };
138        let Some(stock) = self.state.tickets.get(&role) else { return Vec::new() };
139        map.neighbors(from)
140            .iter()
141            .filter(|neighbor| self.station_free(neighbor.station))
142            .filter(|neighbor| self.can_afford(stock, neighbor.transport, role))
143            .map(|neighbor| Hop {
144                from,
145                to: neighbor.station,
146                transport: neighbor.transport,
147            })
148            .collect()
149    }
150}
