119    /// PNG screenshot of the app window.
120    #[tool(description = "Take a screenshot of the app window; returns a PNG image.")]
121    async fn screenshot(&self) -> Result<CallToolResult, ErrorData> {
122        let result = self.command(CommandKind::Screenshot).await??;
123        let CommandResult::Screenshot(reply) = result else {
124            return Err(McpError::InvalidParams("unexpected reply kind for screenshot".to_string()).into());
125        };
126        Ok(CallToolResult::success(vec![
127            ContentBlock::image(reply.png_base64, "image/png".to_string()),
128            ContentBlock::text(format!("screenshot ({:.0}x{:.0})", reply.width.as_f64(), reply.height.as_f64())),
129        ]))
130    }

147    /// Pan/rotate/zoom the view for visual debugging (view-only,
148    /// not persisted).
149    #[tool(description = "Set the view: 'zoom' factor (0.5-5.0), 'rotate' in quarter turns (0-3), 'pan_mm' [dx,dy] in surface-mm.")]
150    async fn set_view(&self, Parameters(args): Parameters<SetViewArgs>) -> Result<CallToolResult, ErrorData> {
151        let zoom = args
152            .zoom
153            .map(MapScale::try_new)
154            .transpose()
155            .map_err(|e| ErrorData::invalid_params(e.to_string(), None))?;
156        let rotate = args
157            .rotate
158            .map(|turns| match turns {
159                0 => Ok(DisplayRotation::Deg0),
160                1 => Ok(DisplayRotation::Deg90),
161                2 => Ok(DisplayRotation::Deg180),
162                3 => Ok(DisplayRotation::Deg270),
163                other => Err(ErrorData::invalid_params(format!("rotate must be 0-3, got {other}"), None)),
164            })
165            .transpose()?;
166        self.forward(CommandKind::SetView {
167            zoom,
168            rotate,
169            pan: args.pan_mm.map(|[dx, dy]| Vec2::new(dx, dy)),
170            compass: args.compass_mm.map(|[x, y]| Vec2::new(x, y)),
171        })
172        .await
173    }

219impl QdoMcp {
220    /// Forwards a command to the app-side game host and serializes
221    /// the `CommandResult` - a domain failure becomes a tool-level
222    /// `isError` reply instead of an `{"Err": ...}` payload.
223    async fn forward(&self, kind: CommandKind) -> Result<CallToolResult, ErrorData> {
224        let result = self.command(kind).await??;
225        self.reply(&result)
226    }
227
228    /// Serializes a reply as pretty JSON text content.
229    fn reply(&self, value: &impl serde::Serialize) -> Result<CallToolResult, ErrorData> {
230        let text = serde_json::to_string_pretty(value).map_err(McpError::from)?;
231        Ok(CallToolResult::success(vec![ContentBlock::text(text)]))
232    }
