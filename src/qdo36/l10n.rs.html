8/// Compile-time embedded Fluent bundles.
9const RESOURCES: &[(&str, &str)] = &[
10    ("en", include_str!("../../../i18n/en/main.ftl")),
11    ("de", include_str!("../../../i18n/de/main.ftl")),
12];
13
14/// Localization wrapper: negotiates the UI language from
15/// `LANGUAGE`/`LANG` once, then serves message lookups with English
16/// fallback.
17pub struct L10n {
18    /// Active Fluent bundle (negotiated locale).
19    bundle: FluentBundle<FluentResource>,
20}
21
22impl L10n {
23    /// Negotiates from the process locale environment.
24    pub fn system() -> Self {
25        let requested: Vec<LanguageIdentifier> = std::env::var("LANGUAGE")
26            .ok()
27            .map(|l| l.split(':').filter_map(|l| l.parse().ok()).collect())
28            .unwrap_or_default();
29        let supported: Vec<LanguageIdentifier> = RESOURCES.iter().filter_map(|(code, _)| code.parse().ok()).collect();
30        let fallback: LanguageIdentifier = "en".parse().unwrap_or_default();
31        let negotiated = negotiate_languages(&requested, &supported, Some(&fallback), NegotiationStrategy::Filtering);
32        Self::for_locale(negotiated.first().map(|l| (*l).clone()).unwrap_or(fallback))
33    }
34
35    /// A bundle for a concrete locale.
36    pub fn for_locale(locale: LanguageIdentifier) -> Self {
37        let mut bundle = FluentBundle::new(vec![locale.clone()]);
38        let locale_code = locale.to_string();
39        if let Some((_, text)) = RESOURCES.iter().find(|(code, _)| *code == locale_code)
40            && let Ok(resource) = FluentResource::try_new(text.to_string())
41        {
42            let _ = bundle.add_resource(resource);
43        }
44        // English as compile-time fallback for missing keys.
45        if locale_code != "en"
46            && let Some((_, text)) = RESOURCES.iter().find(|(code, _)| *code == "en")
47            && let Ok(resource) = FluentResource::try_new(text.to_string())
48        {
49            bundle.add_resource_overriding(resource);
50        }
51        Self { bundle }
52    }
53
54    /// Looks up a message id; falls back to the id itself when absent.
55    pub fn msg(&self, id: &str) -> String {
56        let mut errors = Vec::new();
57        self.bundle
58            .get_message(id)
59            .and_then(|message| message.value())
60            .map(|pattern| self.bundle.format_pattern(pattern, None, &mut errors).to_string())
61            .unwrap_or_else(|| id.to_string())
62    }
63
64    /// Looks up a message id with arguments.
65    #[allow(dead_code)] // used by later phases (status lines, dialogs)
66    pub fn msg_args(&self, id: &str, args: &FluentArgs) -> String {
67        let mut errors = Vec::new();
68        self.bundle
69            .get_message(id)
70            .and_then(|message| message.value())
71            .map(|pattern| self.bundle.format_pattern(pattern, Some(args), &mut errors).to_string())
72            .unwrap_or_else(|| id.to_string())
73    }
74}

76#[cfg(test)]
77mod tests {
78    use std::collections::BTreeSet;
79    use std::path::Path;
80    use std::path::PathBuf;

82    use super::RESOURCES;

84    /// Every `l10n.msg`/`l10n.msg_args` literal id in the app sources
85    /// resolves to a Fluent id in each embedded catalog. `msg` falls
86    /// back to the id itself, so a typo otherwise surfaces only as an
87    /// untranslated line at runtime.
88    #[test]
89    fn message_ids_exist_in_every_locale() {
90        for key in literal_msg_ids() {
91            for (locale, ftl) in RESOURCES {
92                assert!(ftl_ids(ftl).contains(&key.as_str()), "`{key}` used via l10n.msg but missing in i18n/{locale}/main.ftl");
93            }
94        }
95    }
96
97    /// Fluent ids defined in a catalog (top-level `id =` lines).
98    fn ftl_ids(ftl: &str) -> BTreeSet<&str> {
99        ftl.lines()
100            .filter_map(|line| line.split_once('=').map(|(id, _)| id.trim()))
101            .filter(|id| !id.is_empty())
102            .collect()
103    }
104
105    /// Message ids passed literally to `l10n.msg`/`l10n.msg_args`
106    /// across the app sources.
107    fn literal_msg_ids() -> BTreeSet<String> {
108        let mut ids = BTreeSet::new();
109        for file in rust_files(&Path::new(env!("CARGO_MANIFEST_DIR")).join("src")) {
110            // Skip this module: its test names the call sites and would
111            // match the scan needles itself.
112            if file.file_name().is_some_and(|name| name == "l10n.rs") {
113                continue;
114            }
115            let Ok(source) = std::fs::read_to_string(&file) else {
116                continue;
117            };
118            for needle in ["l10n.msg(\"", "l10n.msg_args(\""] {
119                for (index, _) in source.match_indices(needle) {
120                    if let Some(end) = source[index + needle.len()..].find('"') {
121                        ids.insert(source[index + needle.len()..index + needle.len() + end].to_string());
122                    }
123                }
124            }
125        }
126        ids
127    }
128
129    /// All `.rs` files below `dir`, recursively.
130    fn rust_files(dir: &Path) -> Vec<PathBuf> {
131        let mut files = Vec::new();
132        let Ok(entries) = std::fs::read_dir(dir) else {
133            return files;
134        };
135        for entry in entries.flatten() {
136            let path = entry.path();
137            if path.is_dir() {
138                files.extend(rust_files(&path));
139            } else if path.extension().is_some_and(|ext| ext == "rs") {
140                files.push(path);
141            }
142        }
143        files
144    }
