1//! `LayersConfig` - the persisted layer-orientation snapshot in
2//! `config.toml`.
3//!
4//! Every flag is optional so a hand-edited or older config merges
5//! into the current defaults; `capture` freezes the live
6//! `SceneLayers` for the next save.
7
8use crate::ui::map_scene::LabelMode;
9use crate::ui::map_scene::SceneLayers;
10use crate::ui::style::HouseMode;
11use qdo36_map::LineName;
12use qdo36_map::Transport;
13use serde::Deserialize;
14use serde::Serialize;
15use std::collections::BTreeSet;

17/// The persistable layer + orientation state (`config.toml`).
18#[derive(Debug, Clone, Default, Serialize, Deserialize)]
19pub struct LayersConfig {
20    /// Water polylines.
21    #[serde(default)]
22    pub water: Option<bool>,
23    /// City-block layer.
24    #[serde(default)]
25    pub blocks: Option<bool>,
26    /// Backdrop landmark pins.
27    #[serde(default)]
28    pub landmarks: Option<bool>,
29    /// Taxi edges.
30    #[serde(default)]
31    pub taxi: Option<bool>,
32    /// Bus edges.
33    #[serde(default)]
34    pub bus: Option<bool>,
35    /// Metro edges.
36    #[serde(default)]
37    pub metro: Option<bool>,
38    /// Batobus edges.
39    #[serde(default)]
40    pub batobus: Option<bool>,
41    /// Underground edges.
42    #[serde(default)]
43    pub underground: Option<bool>,
44    /// Parallel-transports overlay.
45    #[serde(default)]
46    pub parallel_transports: Option<bool>,
47    /// Highlighted transports (own colors).
48    #[serde(default)]
49    pub highlight: Vec<Transport>,
50    /// Metro lines hidden by name.
51    #[serde(default)]
52    pub hide_lines: Vec<LineName>,
53    /// Station discs.
54    #[serde(default)]
55    pub stations: Option<bool>,
56    /// Station name labels.
57    #[serde(default)]
58    pub labels: Option<bool>,
59    /// Station number labels.
60    #[serde(default)]
61    pub numbers: Option<bool>,
62    /// Importance-score labels.
63    #[serde(default)]
64    pub scores: Option<bool>,
65    /// Station-label orientation.
66    #[serde(default)]
67    pub label_mode: Option<LabelMode>,
68    /// Landmark-pin orientation.
69    #[serde(default)]
70    pub landmark_mode: Option<LabelMode>,
71    /// District-caption orientation.
72    #[serde(default)]
73    pub district_mode: Option<LabelMode>,
74    /// Park-caption orientation.
75    #[serde(default)]
76    pub park_mode: Option<LabelMode>,
77    /// House rendering mode (`radial-iso`/`iso`/`top-down`).
78    #[serde(default)]
79    pub house_mode: Option<HouseMode>,
80    /// The compass rose.
81    #[serde(default)]
82    pub compass: Option<bool>,
83    /// Bridge-station markers.
84    #[serde(default)]
85    pub bridges: Option<bool>,
86}
87
88impl LayersConfig {
89    /// Freezes `layers` into the persisted shape.
90    pub fn capture(layers: &SceneLayers) -> Self {
91        Self {
92            water: Some(layers.water),
93            blocks: Some(layers.blocks),
94            landmarks: Some(layers.landmarks),
95            taxi: Some(layers.transports.taxi),
96            bus: Some(layers.transports.bus),
97            metro: Some(layers.transports.metro),
98            batobus: Some(layers.transports.batobus),
99            underground: Some(layers.transports.underground),
100            parallel_transports: Some(layers.parallel_transports),
101            highlight: layers.highlight.iter().copied().collect(),
102            hide_lines: layers.hidden_lines.iter().cloned().collect(),
103            stations: Some(layers.stations),
104            labels: Some(layers.labels),
105            numbers: Some(layers.numbers),
106            scores: Some(layers.scores),
107            label_mode: Some(layers.label_mode),
108            landmark_mode: Some(layers.landmark_mode),
109            district_mode: Some(layers.district_mode),
110            park_mode: Some(layers.park_mode),
111            house_mode: Some(layers.house_mode),
112            compass: Some(layers.compass),
113            bridges: Some(layers.bridges),
114        }
115    }
116
117    /// Merges the stored flags onto `layers` (`None` keeps the
118    /// current value).
119    pub fn apply_to(&self, layers: &mut SceneLayers) {
120        if let Some(v) = self.water {
121            layers.water = v;
122        }
123        if let Some(v) = self.blocks {
124            layers.blocks = v;
125        }
126        if let Some(v) = self.landmarks {
127            layers.landmarks = v;
128        }
129        if let Some(v) = self.taxi {
130            layers.transports.taxi = v;
131        }
132        if let Some(v) = self.bus {
133            layers.transports.bus = v;
134        }
135        if let Some(v) = self.metro {
136            layers.transports.metro = v;
137        }
138        if let Some(v) = self.batobus {
139            layers.transports.batobus = v;
140        }
141        if let Some(v) = self.underground {
142            layers.transports.underground = v;
143        }
144        if let Some(v) = self.parallel_transports {
145            layers.parallel_transports = v;
146        }
147        layers.highlight = self.highlight.iter().copied().collect::<BTreeSet<_>>();
148        layers.hidden_lines = self.hide_lines.iter().cloned().collect();
149        if let Some(v) = self.stations {
150            layers.stations = v;
151        }
152        if let Some(v) = self.labels {
153            layers.labels = v;
154        }
155        if let Some(v) = self.numbers {
156            layers.numbers = v;
157        }
158        if let Some(v) = self.scores {
159            layers.scores = v;
160        }
161        if let Some(v) = self.label_mode {
162            layers.label_mode = v;
163        }
164        if let Some(v) = self.landmark_mode {
165            layers.landmark_mode = v;
166        }
167        if let Some(v) = self.district_mode {
168            layers.district_mode = v;
169        }
170        if let Some(v) = self.park_mode {
171            layers.park_mode = v;
172        }
173        if let Some(v) = self.house_mode {
174            layers.house_mode = v;
175        }
176        if let Some(v) = self.compass {
177            layers.compass = v;
178        }
179        if let Some(v) = self.bridges {
180            layers.bridges = v;
181        }
182    }
183}

185#[cfg(test)]
186mod tests {
187    use super::*;
188
189    #[test]
190    fn a_capture_round_trips_through_toml() {
191        let mut layers = SceneLayers {
192            water: false,
193            district_mode: LabelMode::Screen,
194            park_mode: LabelMode::Screen,
195            ..SceneLayers::default()
196        };
197        layers.highlight.insert(Transport::Taxi);
198        layers.hidden_lines.insert(LineName::from("M4"));
199        let text = toml::to_string(&LayersConfig::capture(&layers)).unwrap();
200        let mut restored = SceneLayers::default();
201        let config: LayersConfig = toml::from_str(&text).unwrap();
202        config.apply_to(&mut restored);
203        assert!(!restored.water);
204        assert_eq!(restored.district_mode, LabelMode::Screen);
205        assert_eq!(restored.park_mode, LabelMode::Screen);
206        assert!(restored.highlight.contains(&Transport::Taxi));
207        assert!(restored.hidden_lines.contains(&LineName::from("M4")));
208    }
209
210    #[test]
211    fn missing_flags_keep_the_defaults() {
212        let config: LayersConfig = toml::from_str("water = false").unwrap();
213        let mut layers = SceneLayers::default();
214        config.apply_to(&mut layers);
215        assert!(!layers.water);
216        assert!(layers.stations);
217    }
