30/// Command-line options.
31#[derive(Debug, clap::Parser)]
32#[command(name = "qdo36", about = "Quai des Orfèvres 36 - Scotland Yard for the smart desk")]
33struct Args {
34    /// tangible-objects daemon GraphQL endpoint.
35    #[arg(long)]
36    daemon_url: Option<String>,
37    /// Locale override (e.g. `de`, `en`).
38    #[arg(long)]
39    locale: Option<String>,
40    /// City board (`paris` or `berlin`).
41    #[arg(long)]
42    city: Option<City>,
43    /// Expose the game-domain MCP endpoint (Streamable HTTP, e.g.
44    /// `127.0.0.1:8091` for LAN agents) - reserved for WP05.
45    #[arg(long)]
46    mcp_listen: Option<SocketAddr>,
47    /// Display rotation: quarter turns applied to surface-mm
48    /// positions before the mm-to-pixel scale - needed when the
49    /// desktop shows the panel rotated (e.g. GNOME portrait on a
50    /// landscape touchscreen).
51    #[arg(long, value_name = "0|90|180|270")]
52    display_rotation: Option<DisplayRotation>,
53    /// Calibration offset between the Eiffel tower's pin frame and
54    /// map north, in degrees.
55    #[arg(long, value_name = "DEG")]
56    heading_offset: Option<HeadingOffsetDeg>,
57}

59fn main() -> glib::ExitCode {
60    tracing_subscriber::fmt()
61        .with_env_filter(EnvFilter::try_from_default_env().unwrap_or_else(|_| EnvFilter::new("info")))
62        .init();
63
64    let args = <Args as clap::Parser>::parse();
65    let mut settings = AppSettings::load();
66    settings.apply_overrides(&args);
67    // CLI flags are sticky overrides: they land in the persisted
68    // settings and stay effective until changed again.
69    settings.save();
70
71    let l10n = settings
72        .locale
73        .as_deref()
74        .and_then(|code| code.parse::<LanguageIdentifier>().ok())
75        .map_or_else(L10n::system, L10n::for_locale);
76    let l10n = Rc::new(l10n);
77
78    let application = adw::Application::builder().application_id("io.github.smearor.qdo36").build();
79
80    // Bundled OFL fonts must be registered before the first text layout,
81    // so they land here instead of inside `connect_activate`.
82    fonts::install();
83
84    // The MCP endpoint shares state through a watch snapshot; without
85    // the flag the app runs unchanged.
86    let mcp_listen = args.mcp_listen;
87
88    application.connect_activate(move |app| {
89        if let Some(addr) = mcp_listen {
90            mcp_bridge::attach(addr);
91        }
92        if let Err(err) = Qdo36Icons::install() {
93            warn!("icon set install failed: {err}");
94        }
95        if let Some(display) = gdk4::Display::default()
96            && let Err(err) = Qdo36Icons::install_icon_theme(&display)
97        {
98            warn!("icon theme registration failed: {err}");
99        }
100        install_css();
101        install_shutdown_handlers(app);
102        let window = AppWindow::new(app, settings.clone(), l10n.clone());
103        window.window.present();
104        if std::env::var_os("QDO36_NO_FULLSCREEN").is_none() {
105            // Fullscreen must be requested once the window is
106            // presented; before that the content's natural size can
107            // overshoot the monitor and the compositor clamps it
108            // visually.
109            window.window.fullscreen();
110        }
111    });
112
113    // `run()` would feed argv to GTK's own option parser, which rejects
114    // our flags; clap already consumed them.
115    application.run_with_args(&["qdo36"])
116}

118/// Quits the app on SIGINT/SIGTERM so Ctrl+C on the desk exits
119/// cleanly. Daemon registration cleanup (`unregister_owner`) hooks in
120/// here once the controller exists (P2).
121fn install_shutdown_handlers(app: &adw::Application) {
122    let (tx, rx) = async_channel::unbounded::<()>();
123    std::thread::spawn(move || {
124        let mut signals = match Signals::new([signal_hook::consts::SIGINT, signal_hook::consts::SIGTERM]) {
125            Ok(signals) => signals,
126            Err(err) => {
127                warn!("signal handler install failed: {err}");
128                return;
129            }
130        };
131        if signals.forever().next().is_some() {
132            let _ = tx.send_blocking(());
133        }
134    });
135    let app = app.clone();
136    MainContext::default().spawn_local(async move {
137        if rx.recv().await.is_ok() {
138            controller::with_controller(|controller| {
139                controller.borrow_mut().release_registrations();
140            });
141            app.quit();
142        }
143    });
144}

146/// Registers the application stylesheet.
147fn install_css() {
148    let provider = gtk4::CssProvider::new();
149    provider.load_from_string(include_str!("../resources/style.css"));
150    if let Some(display) = gdk4::Display::default() {
151        gtk4::style_context_add_provider_for_display(&display, &provider, gtk4::STYLE_PROVIDER_PRIORITY_APPLICATION);
152    }
153}
