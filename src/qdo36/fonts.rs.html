1//! Bundled font registration for the GTK frontend.
2//!
3//! The OFL TTFs shipped in `resources/fonts` are registered with the
4//! process fontconfig at startup, so Pango text (Fluent labels, HUD,
5//! overlays) resolves the bundled families (Patrick Hand, Comic Neue,
6//! Cormorant Garamond, EB Garamond, Courier Prime, Orbitron, Share Tech
7//! Mono) instead of relying on whatever the host system happens to
8//! install. Generated artwork needs no fonts at all — its `<text>`
9//! elements are already outlined to paths by `qdo36-artwork`.
10
11use fontconfig_sys as fc;
12use std::ffi::CString;
13use std::path::PathBuf;
14use tracing::info;
15use tracing::warn;
16
17/// Registers the bundled font directory with fontconfig.
18///
19/// Runs before the application activates so the faces are available to
20/// the first text layout. A missing directory is non-fatal: the app
21/// keeps working with system-font fallbacks.
22pub(crate) fn install() {
23    let fonts_dir = bundled_fonts_dir();
24    if !fonts_dir.is_dir() {
25        warn!(dir = %fonts_dir.display(), "bundled font directory missing; falling back to system fonts");
26        return;
27    }
28    let Ok(path) = CString::new(fonts_dir.as_os_str().to_string_lossy().as_bytes()) else {
29        warn!("bundled font path is not valid UTF-8; falling back to system fonts");
30        return;
31    };
32    unsafe {
33        let config = fc::FcConfigGetCurrent();
34        if config.is_null() {
35            warn!("fontconfig unavailable; falling back to system fonts");
36            return;
37        }
38        if fc::FcConfigAppFontAddDir(config, path.as_ptr().cast()) == 0 {
39            warn!(dir = %fonts_dir.display(), "bundled font registration failed");
40        } else {
41            info!(dir = %fonts_dir.display(), "bundled fonts registered");
42        }
43    }
44}
45
46/// Locates `resources/fonts` — first next to the running executable
47/// (installed layout), then relative to the crate (development layout).
48fn bundled_fonts_dir() -> PathBuf {
49    if let Ok(exe) = std::env::current_exe()
50        && let Some(parent) = exe.parent()
51    {
52        let candidate = parent.join("resources").join("fonts");
53        if candidate.is_dir() {
54            return candidate;
55        }
56    }
57    PathBuf::from(concat!(env!("CARGO_MANIFEST_DIR"), "/resources/fonts"))
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63    use std::ffi::CStr;
64
65    /// Queries fontconfig for `family` after `install()` and returns the
66    /// matched face's filename, if any.
67    fn matched_file(family: &str) -> Option<String> {
68        unsafe {
69            let config = fc::FcConfigGetCurrent();
70            let pattern = fc::FcNameParse(CString::new(family).ok()?.as_ptr().cast());
71            let mut result = 0u32;
72            let matched = fc::FcFontMatch(config, pattern, &mut result);
73            if matched.is_null() || result != fc::FcResultMatch {
74                return None;
75            }
76            let mut file: *mut fc::FcChar8 = std::ptr::null_mut();
77            if fc::FcPatternGetString(matched, c"file".as_ptr(), 0, &mut file) != fc::FcResultMatch {
78                return None;
79            }
80            let name = CStr::from_ptr(file.cast()).to_string_lossy().into_owned();
81            Some(name)
82        }
83    }
84
85    #[test]
86    fn registers_bundled_fonts() {
87        install();
88        let matched = matched_file("Patrick Hand");
89        assert!(matched.is_some(), "Patrick Hand did not resolve");
90        assert!(matched.unwrap_or_default().contains("PatrickHand"), "resolved to an unexpected file");
91    }
