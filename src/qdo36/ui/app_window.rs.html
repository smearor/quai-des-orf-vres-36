28/// The application window: a fullscreen stack switching between the
29/// daemon setup screen and the map scene.
30pub struct AppWindow {
31    /// The GTK window.
32    pub window: adw::ApplicationWindow,
33    /// The screen stack (setup ↔ map).
34    stack: gtk4::Stack,
35    /// The setup status page.
36    setup: adw::StatusPage,
37    /// The map scene widget.
38    scene: MapScene,
39    /// Fluent lookups.
40    l10n: Rc<L10n>,
41    /// Persisted settings snapshot.
42    settings: AppSettings,
43}
44
45impl AppWindow {
46    /// Builds the window and kicks off the daemon handshake.
47    pub fn new(app: &adw::Application, settings: AppSettings, l10n: Rc<L10n>) -> Rc<Self> {
48        let window = adw::ApplicationWindow::builder().application(app).build();
49
50        let stack = gtk4::Stack::new();
51        let setup = adw::StatusPage::builder()
52            .title(l10n.msg("app-title"))
53            .description(l10n.msg("daemon-connecting"))
54            .icon_name("qdo-map-symbolic")
55            .build();
56        stack.add_named(&setup, Some("setup"));
57
58        let scene = MapScene::new();
59        stack.add_named(&scene, Some("map"));
60
61        window.set_content(Some(&stack));
62
63        let this = Rc::new(Self {
64            window,
65            stack,
66            setup,
67            scene,
68            l10n,
69            settings,
70        });
71        this.start_handshake();
72        this
73    }
74
75    /// Runs the blocking daemon handshake on a worker thread, then
76    /// hands results back to the glib main loop.
77    fn start_handshake(self: &Rc<Self>) {
78        let Ok(controller) = controller::init_controller() else {
79            self.setup.set_description(Some("internal: tokio runtime failed"));
80            return;
81        };
82
83        // The worker thread drives the blocking handshake via the
84        // runtime Handle (Send); the controller itself is thread-local
85        // and cannot be reached from there.
86        let handle = controller.borrow().host().handle();
87        let endpoint = self.settings.daemon_url.clone();
88        let city = self.settings.city;
89        let (tx, rx) = async_channel::bounded(1);
90        std::thread::spawn(move || {
91            let result = AnchorPattern::for_city(city).and_then(|pattern| TangibleHost::connect_and_subscribe(&handle, &endpoint, &pattern));
92            let _ = tx.send_blocking(result);
93        });
94
95        let map = GameMap::for_city(city).map(Rc::new).ok();
96        let scale = self.settings.map_scale;
97        let this = self.clone();
98        MainContext::default().spawn_local(async move {
99            match rx.recv().await {
100                Ok(Ok((source, _pattern_id, events))) => {
101                    this.on_connected(&controller, source, events, map, scale);
102                }
103                Ok(Err(err)) => {
104                    warn!("daemon handshake failed: {err}");
105                    let status = match &err {
106                        DaemonError::Connect { .. } => this.l10n.msg("daemon-unreachable"),
107                        _ => err.to_string(),
108                    };
109                    this.setup.set_description(Some(&status));
110                }
111                Err(_) => this.setup.set_description(Some("connect channel closed")),
112            }
113        });
114    }
115
116    /// Connect succeeded: store the source, report the surface size in
117    /// the status line and start the tangible event pump feeding the
118    /// scene.
119    fn on_connected(
120        self: &Rc<Self>,
121        controller: &Rc<RefCell<Controller>>,
122        source: GraphQlSource,
123        events: async_channel::Receiver<TangibleEvent>,
124        map: Option<Rc<GameMap>>,
125        scale: MapScale,
126    ) {
127        let (status, surface) = {
128            let mut controller = controller.borrow_mut();
129            let surface = controller.host().surface(&source).ok();
130            controller.set_source(source);
131            let status = match surface {
132                Some(surface) => format!("{} ({:.0} x {:.0} mm)", self.l10n.msg("daemon-ready"), surface.width.as_f64(), surface.height.as_f64()),
133                None => self.l10n.msg("daemon-ready"),
134            };
135            (status, surface)
136        };
137        let prompt = self.l10n.msg("place-tower");
138        self.setup.set_description(Some(&format!("{status}\n\n{prompt}")));
139        info!("daemon handshake complete");
140
141        let Some(map) = map else {
142            self.setup.set_description(Some("map dataset failed to load"));
143            return;
144        };
145        let state = Rc::new(RefCell::new(self.scene_state(&map, surface, scale)));
146
147        // Tap hit-testing: widget-px -> map-mm -> nearest station
148        // within the snap radius; the selection rings it.
149        let gesture = gtk4::GestureClick::new();
150        {
151            let state = state.clone();
152            let scene = self.scene.clone();
153            gesture.connect_released(move |_, _, px, py| {
154                let station = scene.station_at(&state.borrow(), Vec2::new(px, py));
155                if let Some(station) = station {
156                    state.borrow_mut().selected = Some(station);
157                    scene.update(Rc::new(state.borrow().clone()));
158                }
159            });
160            self.scene.add_controller(gesture);
161        }
162
163        // Event pump: tangible events arrive on the glib loop; every
164        // pose updates the scene; the first tracked anchor advances
165        // to the map screen.
166        let this = self.clone();
167        MainContext::default().spawn_local(async move {
168            while let Ok(event) = events.recv().await {
169                controller::with_controller(|controller| {
170                    let tracked = matches!(event, TangibleEvent::Added(_) | TangibleEvent::Moved(_));
171                    let loss_started = {
172                        let mut controller = controller.borrow_mut();
173                        let loss_started = controller.apply_event(&event);
174                        state.borrow_mut().anchor = controller.anchor();
175                        loss_started
176                    };
177                    this.scene.update(Rc::new(state.borrow().clone()));
178                    if tracked && this.stack.visible_child_name().is_some_and(|name| name == "setup") {
179                        this.stack.set_visible_child_name("map");
180                    }
181                    if loss_started {
182                        // The anchor was lost: re-check once the grace
183                        // period elapsed; a re-tracked tower meanwhile
184                        // cancels the downgrade.
185                        let state = state.clone();
186                        let scene = this.scene.clone();
187                        MainContext::default().spawn_local(async move {
188                            glib::timeout_future(anchor_tracker::ANCHOR_LOSS_GRACE).await;
189                            controller::with_controller(|controller| {
190                                let mut controller = controller.borrow_mut();
191                                if controller.settle_anchor_loss() {
192                                    state.borrow_mut().anchor = controller.anchor();
193                                    scene.update(Rc::new(state.borrow().clone()));
194                                }
195                            });
196                        });
197                    }
198                });
199            }
200        });
201    }

203    /// Builds the scene state: map, anchor config (from the city's
204    /// anchor spec), virtual anchor and surface.
205    fn scene_state(&self, map: &Rc<GameMap>, surface: Option<SurfaceMm>, scale: MapScale) -> SceneState {
206        let anchor_pos = map
207            .meta()
208            .anchor
209            .as_ref()
210            .and_then(|spec| map.station(spec.station))
211            .or_else(|| map.stations().next())
212            .map(|station| station.pos)
213            .unwrap_or_else(|| Vec2::new(0.0, 0.0));
214        let mut config = AnchorConfig::at_station(map.bounds(), anchor_pos);
215        config.heading_offset = self.settings.heading_offset_deg.to_rotation_rad();
216        SceneState {
217            map: map.clone(),
218            config,
219            // The event pump starts before any tangible event - the
220            // virtual fallback pose.
221            anchor: qdo36_tangibles::AnchorState::default_virtual(),
222            surface,
223            scale,
224            selected: None,
225            display_rotation: self.settings.display_rotation,
226            style: self.settings.map_style,
227            assets: crate::ui::art::ArtAssets::detect(),
228            anchor_hint: Some(self.l10n.msg("tower-missing")),
229        }
230    }
