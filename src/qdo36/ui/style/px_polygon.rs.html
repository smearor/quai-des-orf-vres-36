1//! `PxPolygon` - a px-space polygon view over a position slice.
2
3use crate::ui::style::renderer::PxPos;
4use smearor_commons_geometry::Vec2;
5use std::ops::Deref;
6
7/// A px-space closed ring over a borrowed `PxPos` slice - the
8/// render counterpart of `Polygon`'s ring queries (`segments`,
9/// `centroid`), without owning the points.
10#[derive(Debug, Clone, Copy)]
11pub struct PxPolygon<'a>(pub &'a [PxPos]);
12
13impl PxPolygon<'_> {
14    /// The ring's vertex centroid in widget px.
15    pub fn centroid(&self) -> PxPos {
16        let count = self.0.len().max(1) as f64;
17        Vec2::new(self.0.iter().map(|p| p.x).sum::<f64>() / count, self.0.iter().map(|p| p.y).sum::<f64>() / count)
18    }
19
20    /// The ring's consecutive edge segments (`(i, i + 1)`,
21    /// wrapping the end back to the start).
22    pub fn segments(&self) -> impl Iterator<Item = (PxPos, PxPos)> + '_ {
23        let n = self.0.len();
24        (0..n).map(move |i| (self.0[i], self.0[(i + 1) % n]))
25    }
26
27    /// The polygon shrunk toward its centroid by `fraction` -
28    /// a px-space `Polygon::offset` light (roof bodies, insets).
29    pub fn shrunk(&self, fraction: f64) -> Vec<PxPos> {
30        let c = self.centroid();
31        self.0
32            .iter()
33            .map(|p| Vec2::new(p.x + (c.x - p.x) * fraction, p.y + (c.y - p.y) * fraction))
34            .collect()
35    }
36}
37
38impl Deref for PxPolygon<'_> {
39    type Target = [PxPos];
40
41    fn deref(&self) -> &Self::Target {
42        self.0
43    }
44}
