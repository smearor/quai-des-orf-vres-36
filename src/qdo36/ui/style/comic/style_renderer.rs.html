40/// The countryside texture tile: a staggered halftone dot grid on
41/// paper - comic print grain, infinitely repeatable.
42fn outer_tile() -> Option<ImageSurface> {
43    let tile = ImageSurface::create(cairo::Format::ARgb32, 24, 24).ok()?;
44    let tile_ctx = Context::new(&tile).ok()?;
45    PALETTE.paper().apply(&tile_ctx);
46    tile_ctx.paint().ok()?;
47    // Offset halftone dots - coarse enough to read at map-fit zoom.
48    PALETTE.ink().apply_alpha(&tile_ctx, Alpha::new(0.22));
49    for (x, y) in [(6.0, 6.0), (18.0, 18.0)] {
50        tile_ctx.arc(x, y, 2.8, 0.0, TAU);
51        tile_ctx.fill().ok()?;
52    }
53    // Secondary dots in a cooler tone - distinct from the ink
54    // dots instead of the same color twice.
55    PALETTE.water().apply_alpha(&tile_ctx, Alpha::new(0.16));
56    for (x, y) in [(6.0, 18.0), (18.0, 6.0)] {
57        tile_ctx.arc(x, y, 1.6, 0.0, TAU);
58        tile_ctx.fill().ok()?;
59    }
60    Some(tile)
61}

68    fn draw_water(&self, ctx: &Context, points: &[Vec2<Millimeters>], px_per_mm: PixelsPerMm, to_px: ToPx<'_>) {
69        // Two bank lines around a darker river body - the Seine
70        // reads as a band, not a stroke. The 7 mm band width is
71        // map-space, so the river zooms with the map.
72        let banks = WaterBanks::new(points, WATER_HALF_WIDTH);
73        if banks.left.is_empty() {
74            return;
75        }
76        let left_bank: Vec<PxPos> = to_px.points(&banks.left);
77        let right_bank: Vec<PxPos> = to_px.points(&banks.right);
78        ctx.band_path(&left_bank, &right_bank);
79        PALETTE.water_fill().apply(ctx);
80        ctx.fill().ok();
81        ctx.set_line_cap(LineCap::Round);
82        ctx.set_line_width(px_per_mm.scale(Millimeters::new(1.2)).max(Pixels::new(0.5)).as_f64());
83        PALETTE.water().apply(ctx);
84        for bank in [&left_bank, &right_bank] {
85            ctx.px_polyline_path(bank);
86            ctx.stroke().ok();
87        }
88    }

90    fn draw_bridge(&self, ctx: &Context, from: PxPos, to: PxPos, px_per_mm: PixelsPerMm) {
91        ctx.set_line_cap(LineCap::Round);
92        ctx.move_to(from.x, from.y);
93        ctx.line_to(to.x, to.y);
94        ctx.set_line_width(px_per_mm.scale(Millimeters::new(5.0)).max(Pixels::new(0.5)).as_f64());
95        PALETTE.paper().apply(ctx);
96        ctx.stroke_preserve().ok();
97        ctx.set_line_width(px_per_mm.scale(Millimeters::new(2.2)).max(Pixels::new(0.4)).as_f64());
98        PALETTE.station().apply(ctx);
99        ctx.stroke().ok();
100    }

102    fn draw_outer(&self, ctx: &Context, spec: &OuterSpec<'_>, to_px: ToPx<'_>) {
103        if let Some(tile) = outer_tile() {
104            let pin = to_px.pos(spec.pin);
105            ctx.tile_fill(&tile, &pin, spec.width.as_f64(), spec.height.as_f64());
106        }
107        ctx.fill_ground(spec.ground, &PALETTE.paper(), to_px);
108    }

110    fn draw_block(&self, ctx: &Context, block: &BlockSpec<'_>, to_px: ToPx<'_>) {
111        ctx.set_line_join(cairo::LineJoin::Round);
112        // Parks first: green blobs under the houses (which avoid
113        // them anyway); the names draw later, over the streets.
114        for park in block.parks {
115            let blob: Vec<PxPos> = to_px.points(&park.polygon);
116            ctx.px_polyline_path(&blob);
117            ctx.close_path();
118            PALETTE.park().apply(ctx);
119            ctx.fill_preserve().ok();
120            PALETTE.block().apply_alpha(ctx, Alpha::new(0.4));
121            ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(0.4)).max(Pixels::new(0.2)).as_f64());
122            ctx.stroke().ok();
123        }
124        // Houses: raised fill over the paper, roof details in the
125        // `roof` tone (contrast against the fill), block-ink
126        // outline. Fill shade varies with width and variant.
127        ctx.draw_houses(block, &PALETTE, to_px);
128        // Decorative streets crossing the block interior - muted,
129        // under the captions.
130        ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(1.0)).max(Pixels::new(0.3)).as_f64());
131        for street in block.streets {
132            let path: Vec<PxPos> = to_px.points(street);
133            ctx.px_polyline_path(&path);
134            PALETTE.street().apply_alpha(ctx, Alpha::new(0.7));
135            ctx.stroke().ok();
136        }
137        // Park names over the streets, then the district caption.
138        for park in block.parks {
139            ctx.park_label(
140                park.name.as_str(),
141                park.polygon.centroid(),
142                to_px,
143                block.px_per_mm,
144                &CaptionStyle::park(PALETTE.paper(), PALETTE.label(), block.park_orientation),
145            );
146        }
147        if let Some(label) = &block.label {
148            ctx.district_label(
149                label,
150                to_px,
151                block.px_per_mm,
152                &CaptionStyle::district(PALETTE.paper(), PALETTE.label(), block.district_orientation),
153            );
154        }
155        // Block outline on top.
156        if block.outline.len() >= 3 {
157            let outline: Vec<PxPos> = to_px.points(block.outline);
158            ctx.px_polyline_path(&outline);
159            ctx.close_path();
160            ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(0.6)).max(Pixels::new(0.3)).as_f64());
161            PALETTE.block().apply(ctx);
162            ctx.stroke().ok();
163        }
164    }

166    fn draw_edge(&self, ctx: &Context, edge: &EdgeSpec<'_>) {
167        let px = edge.px_per_mm;
168        ctx.set_line_cap(LineCap::Round);
169        if edge.transport == Transport::Underground {
170            ctx.set_dash(&[px.scale(Millimeters::new(4.0)).as_f64(), px.scale(Millimeters::new(3.0)).as_f64()], 0.0);
171        }
172        let color = if edge.highlight {
173            PALETTE.transport_color(edge.transport, edge.line, edge.map)
174        } else {
175            PALETTE.edge()
176        };
177        color.apply(ctx);
178        ctx.set_line_width(px.scale(Millimeters::new(2.4)).max(Pixels::new(0.5)).as_f64());
179        ctx.edge_path(edge);
180        if edge.wavy {
181            // Boat wake: ink rim under the colored wave.
182            ctx.set_line_width(px.scale(Millimeters::new(3.2)).max(Pixels::new(0.7)).as_f64());
183            PALETTE.ink().apply(ctx);
184            ctx.stroke_preserve().ok();
185            ctx.set_line_width(px.scale(Millimeters::new(2.2)).max(Pixels::new(0.5)).as_f64());
186            color.apply(ctx);
187        }
188        if edge.dashed {
189            // Metro lane in the combined overlay: dark rim under
190            // the dashed color stroke keeps lookalike colors
191            // (M1 vs taxi, M6/M12 vs bus) apart.
192            ctx.set_dash(&[px.scale(Millimeters::new(4.0)).as_f64(), px.scale(Millimeters::new(2.5)).as_f64()], 0.0);
193            ctx.set_line_width(px.scale(Millimeters::new(3.4)).max(Pixels::new(0.7)).as_f64());
194            PALETTE.ink().apply(ctx);
195            ctx.stroke_preserve().ok();
196            ctx.set_line_width(px.scale(Millimeters::new(2.4)).max(Pixels::new(0.5)).as_f64());
197            color.with_min_luminance(Luminance::new(0.5)).apply(ctx);
198        }
199        ctx.stroke().ok();
200        ctx.set_dash(&[], 0.0);
201    }

203    fn draw_landmark(&self, ctx: &Context, pos: PxPos, scale: ScaleFactor, px_per_mm: PixelsPerMm) {
204        ctx.new_path();
205        PALETTE.dim().apply(ctx);
206        ctx.set_line_width(1.5);
207        ctx.arc(pos.x, pos.y, px_per_mm.scale(Millimeters::new(2.2 * scale.get())).max(Pixels::new(0.5)).as_f64(), 0.0, TAU);
208        ctx.stroke().ok();
209    }

211    fn draw_station(&self, ctx: &Context, spec: &StationSpec<'_>) {
212        // `show_text` (labels) leaves a current point; `arc` would
213        // otherwise insert a stray line_to into the first ring.
214        ctx.new_path();
215        let pos = spec.pos;
216        let px_per_mm = spec.px_per_mm;
217        let size = match spec.importance {
218            Some(ImportanceClass::Minor) => 0.75,
219            Some(ImportanceClass::Hub) => 1.35,
220            _ => 1.0,
221        };
222        let radius = px_per_mm
223            .scale(Millimeters::new(3.0 * size * spec.station_scale.get()))
224            .max(Pixels::new(0.4))
225            .as_f64();
226        // Fill: inside the metro overlay the disc splits into line
227        // color sectors (1 = full disc, 2 = halves, 3 = thirds...).
228        if spec.metro_overlay && !spec.lines.is_empty() {
229            let sectors = spec.lines.len() as f64;
230            for (i, line) in spec.lines.iter().enumerate() {
231                let start = -std::f64::consts::FRAC_PI_2 + i as f64 * TAU / sectors;
232                ctx.move_to(pos.x, pos.y);
233                ctx.arc(pos.x, pos.y, radius, start, start + TAU / sectors);
234                ctx.close_path();
235                PALETTE.transport_color(Transport::Metro, Some(line), spec.map).apply(ctx);
236                ctx.fill().ok();
237            }
238        } else {
239            ctx.arc(pos.x, pos.y, radius, 0.0, TAU);
240            PALETTE.station().apply(ctx);
241            ctx.fill_preserve().ok();
242            ctx.set_line_width(1.4);
243            PALETTE.ink().apply(ctx);
244            ctx.stroke().ok();
245        }
246        // One ring per served transport, outermost first.
247        for (i, transport) in spec.served.iter().enumerate() {
248            ctx.arc(
249                pos.x,
250                pos.y,
251                radius
252                    + px_per_mm
253                        .scale(Millimeters::new(1.1 * spec.ring_width.get() * (spec.served.len() - i) as f64))
254                        .max(Pixels::new(0.4))
255                        .as_f64(),
256                0.0,
257                TAU,
258            );
259            PALETTE.base_color(*transport).apply(ctx);
260            ctx.set_line_width(0.9 * spec.ring_width.get());
261            ctx.stroke().ok();
262        }
263        if spec.selected {
264            ctx.arc(pos.x, pos.y, radius + px_per_mm.scale(Millimeters::new(3.0)).max(Pixels::new(0.4)).as_f64(), 0.0, TAU);
265            PALETTE.select().apply(ctx);
266            ctx.set_line_width(2.5);
267            ctx.stroke().ok();
268        }
269    }

271    fn draw_label(&self, ctx: &Context, spec: &LabelSpec<'_>) {
272        let pos = spec.pos;
273        ctx.select_font_face("sans-serif", FontSlant::Normal, FontWeight::Bold);
274        ctx.set_font_size(
275            spec.px_per_mm
276                .scale(Millimeters::new(if spec.inside { 3.4 } else { 4.0 } * spec.label_scale.get()))
277                .max(Pixels::new(8.0))
278                .as_f64(),
279        );
280        let Ok(extents) = ctx.text_extents(spec.text) else { return };
281        let _saved = ctx.save();
282        let angle = spec.orientation.angle(pos);
283        if !angle.is_zero() {
284            ctx.translate(pos.x, pos.y);
285            ctx.rotate(angle.as_f64());
286            ctx.translate(-pos.x, -pos.y);
287        }
288        let y = if spec.inside {
289            pos.y + extents.height() / 2.0
290        } else {
291            pos.y + spec.radius.as_f64() + spec.px_per_mm.scale(Millimeters::new(5.5)).max(Pixels::new(10.0)).as_f64()
292        };
293        ctx.new_path();
294        ctx.move_to(pos.x - extents.width() / 2.0, y);
295        // In-disc numbers draw in ink - `label` is paper-colored.
296        if spec.inside {
297            PALETTE.ink().apply(ctx);
298        } else {
299            PALETTE.label().apply(ctx);
300        }
301        ctx.show_text(spec.text).ok();
302        let _restored = ctx.restore();
303    }

305    fn draw_anchor(&self, ctx: &Context, pos: PxPos, tracked: bool, px_per_mm: PixelsPerMm) {
306        ctx.new_path();
307        let radius = px_per_mm.scale(Millimeters::new(6.0)).max(Pixels::new(0.5)).as_f64();
308        ctx.arc(pos.x, pos.y, radius, 0.0, TAU);
309        if tracked {
310            PALETTE.accent().apply(ctx);
311            ctx.fill_preserve().ok();
312            Color::<f64>::new(1.0, 1.0, 1.0).apply(ctx);
313        } else {
314            PALETTE.dim().apply(ctx);
315        }
316        ctx.set_line_width(1.8);
317        ctx.stroke().ok();
318    }

320    fn draw_hint(&self, ctx: &Context, text: &str, width: Pixels, height: Pixels, bottom: Pixels) {
321        ctx.select_font_face("sans-serif", FontSlant::Normal, FontWeight::Bold);
322        ctx.set_font_size(18.0);
323        let Ok(extents) = ctx.text_extents(text) else { return };
324        let (pad_x, pad_y) = (14.0, 10.0);
325        let (box_w, box_h) = (extents.width() + 2.0 * pad_x, extents.height() + 2.0 * pad_y);
326        let (x, y) = ((width.as_f64() - box_w) / 2.0, height.as_f64() - box_h - 24.0 - bottom.as_f64());
327        PALETTE.panel().apply_alpha(ctx, Alpha::new(0.72));
328        ctx.rectangle(x, y, box_w, box_h);
329        ctx.fill().ok();
330        PALETTE.label().apply(ctx);
331        ctx.move_to(x + pad_x - extents.x_bearing(), y + pad_y - extents.y_bearing());
332        ctx.show_text(text).ok();
333    }
