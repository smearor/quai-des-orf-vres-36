1//! `StylePalette` - the semantic color tokens a map style carries.
2//!
3//! The field names mirror the SVG style-book tokens (`ARTWORK.md` §2:
4//! bg, ink, accent, water, taxi, bus, batobus, station, highlight,
5//! panel_bg) so the cairo renderers speak the same vocabulary as the
6//! generated artwork. Metro edges are deliberately absent - their
7//! colors come from `map.meta().lines`, not from the style.
8
9use qdo36_map::GameMap;
10use qdo36_map::LineName;
11use qdo36_map::Transport;
12
13use crate::ui::style::RgbColor;
14
15/// One style's semantic colors.
16#[derive(Debug, Clone, Copy)]
17pub struct StylePalette {
18    /// The field the scene paints on.
19    pub paper: RgbColor,
20    /// Primary foreground ink: station outlines, labels, hairlines.
21    pub ink: RgbColor,
22    /// Signature accent: gold, neon glow, comic ink-pop.
23    pub accent: RgbColor,
24    /// Dimmed secondary: virtual anchor, muted elements.
25    pub dim: RgbColor,
26    /// Water fill/wash.
27    pub water: RgbColor,
28    /// Selection highlight ring.
29    pub select: RgbColor,
30    /// Station node fill.
31    pub station: RgbColor,
32    /// Label and hint text.
33    pub label: RgbColor,
34    /// Hint/overlay box fill.
35    pub panel: RgbColor,
36    /// Taxi route color.
37    pub taxi: RgbColor,
38    /// Bus route color.
39    pub bus: RgbColor,
40    /// Batobus route color.
41    pub batobus: RgbColor,
42    /// Underground secret-route color.
43    pub underground: RgbColor,
44}
45
46impl StylePalette {
47    /// Edge color per transport: fixed routes come from the palette,
48    /// metro edges carry their declared line color from `map.lines`.
49    pub fn transport_color(&self, transport: Transport, line: Option<&LineName>, map: &GameMap) -> RgbColor {
50        match transport {
51            Transport::Taxi => self.taxi,
52            Transport::Bus => self.bus,
53            Transport::Batobus => self.batobus,
54            Transport::Underground => self.underground,
55            Transport::Metro => line
56                .and_then(|name| map.meta().lines.get(name))
57                .map(|color| {
58                    let [r, g, b] = color.rgb_u8();
59                    RgbColor::from_u8(r, g, b)
60                })
61                .unwrap_or(self.ink),
62        }
63    }
64}
