3use crate::ui::style::StyleColor;
4use crate::ui::style::StylePalette;

7use std::sync::LazyLock;
8
9/// Near-black field, cyan glow family, pink selection.
10pub(crate) static PALETTE: LazyLock<StylePalette> = LazyLock::new(|| {
11    palette! {
12        // #05080D - near-black field.
13        StyleColor::Paper => Color::<f64>::new(0.02, 0.03, 0.05),
14        // #1AF2FF - glow cyan (same as the accent; neon is monochrome-led).
15        StyleColor::Ink => Color::<f64>::new(0.10, 0.95, 1.0),
16        // #1AF2FF - signature glow: edge cores, anchor, hint text.
17        StyleColor::Accent => Color::<f64>::new(0.10, 0.95, 1.0),
18        // #4D7380 - dimmed cyan: virtual anchor.
19        StyleColor::Dim => Color::<f64>::new(0.30, 0.45, 0.50),
20        // #1AF2FF - glowing Seine bank lines.
21        StyleColor::Water => Color::<f64>::new(0.10, 0.95, 1.0),
22        // #0A2130 - dark river body, distinct from the near-black field.
23        StyleColor::WaterFill => Color::<f64>::new(0.04, 0.13, 0.19),
24        // #FF4DD9 - pink selection ring.
25        StyleColor::Select => Color::<f64>::new(1.0, 0.30, 0.85),
26        // #1AF2FF - station discs glow on the dark field.
27        StyleColor::Station => Color::<f64>::new(0.10, 0.95, 1.0),
28        // #66D9F2 - label text, slightly muted cyan.
29        StyleColor::Label => Color::<f64>::new(0.40, 0.85, 0.95),
30        // #00050A - hint badge fill (drawn at ~0.85 alpha).
31        StyleColor::Panel => Color::<f64>::new(0.0, 0.02, 0.04),
32        // #FFD91A - taxi yellow glow.
33        StyleColor::Taxi => Color::<f64>::new(1.0, 0.85, 0.10),
34        // #33F266 - bus green glow.
35        StyleColor::Bus => Color::<f64>::new(0.20, 0.95, 0.40),
36        // #00D9F2 - batobus cyan.
37        StyleColor::Batobus => Color::<f64>::new(0.0, 0.85, 0.95),
38        // #FF7326 - underground secret routes, orange glow.
39        StyleColor::Underground => Color::<f64>::new(1.0, 0.45, 0.15),
40        // #1A2B38 - dim neutral edge ink (standard view, no highlight).
41        StyleColor::Edge => Color::<f64>::new(0.10, 0.17, 0.22),
42        // #B44DFF - purple glow: generic metro ring mark and the
43        // metro lane in parallel-transports view.
44        StyleColor::Metro => Color::<f64>::new(0.71, 0.30, 1.0),
45        // #14505E - low-glow cyan: city-block outlines, dimmer than
46        // the route lanes.
47        StyleColor::Block => Color::<f64>::new(0.08, 0.31, 0.37),
48        // #0C2733 - deep teal fill for the procedural houses.
49        StyleColor::Building => Color::<f64>::new(0.05, 0.15, 0.20),
50        StyleColor::Park => Color::<f64>::new(0.10, 0.38, 0.26),
51        // #2E7A8C - lifted cyan roof lines on the dark building fill.
52        StyleColor::Roof => Color::<f64>::new(0.18, 0.48, 0.55),
53        // #3A4A52 - muted slate, distinct from block cyan and the
54        // orange underground routes.
55        StyleColor::Street => Color::<f64>::new(0.23, 0.29, 0.32),
56    }
57    .into()
58});
