40/// The countryside texture tile: a faint diagonal field grid with
41/// dim glow dots - quiet neon field lines over the dark paper.
42fn outer_tile() -> Option<ImageSurface> {
43    let tile = ImageSurface::create(cairo::Format::ARgb32, 26, 26).ok()?;
44    let tile_ctx = Context::new(&tile).ok()?;
45    PALETTE.paper().apply(&tile_ctx);
46    tile_ctx.paint().ok()?;
47    // Diagonal field grid with glow nodes - coarse enough to read
48    // at map-fit zoom.
49    PALETTE.accent().apply_alpha(&tile_ctx, Alpha::new(0.13));
50    tile_ctx.set_line_width(0.8);
51    for offset in [0.0, 13.0, 26.0] {
52        tile_ctx.move_to(offset, 0.0);
53        tile_ctx.line_to(offset - 26.0, 26.0);
54    }
55    tile_ctx.stroke().ok()?;
56    PALETTE.block().apply_alpha(&tile_ctx, Alpha::new(0.6));
57    for (x, y) in [(6.5, 6.5), (19.5, 19.5)] {
58        tile_ctx.arc(x, y, 1.4, 0.0, TAU);
59        tile_ctx.fill().ok()?;
60    }
61    Some(tile)
62}

69    fn draw_water(&self, ctx: &Context, points: &[Vec2<Millimeters>], px_per_mm: PixelsPerMm, to_px: ToPx<'_>) {
70        // Two glowing bank lines around a dark river body - the
71        // Seine reads as a band, not a stroke. The 7 mm band width
72        // is map-space, so the river zooms with the map.
73        let banks = WaterBanks::new(points, WATER_HALF_WIDTH);
74        if banks.left.is_empty() {
75            return;
76        }
77        let left_bank: Vec<PxPos> = to_px.points(&banks.left);
78        let right_bank: Vec<PxPos> = to_px.points(&banks.right);
79        ctx.set_line_cap(LineCap::Round);
80        ctx.band_path(&left_bank, &right_bank);
81        PALETTE.water_fill().apply(ctx);
82        ctx.fill().ok();
83        for bank in [&left_bank, &right_bank] {
84            ctx.px_polyline_path(bank);
85            ctx.set_line_width(px_per_mm.scale(Millimeters::new(1.8)).max(Pixels::new(0.5)).as_f64());
86            PALETTE.water().apply_alpha(ctx, Alpha::new(0.25));
87            ctx.stroke_preserve().ok();
88            ctx.set_line_width(px_per_mm.scale(Millimeters::new(0.9)).max(Pixels::new(0.4)).as_f64());
89            PALETTE.water().apply(ctx);
90            ctx.stroke().ok();
91        }
92    }

94    fn draw_bridge(&self, ctx: &Context, from: PxPos, to: PxPos, px_per_mm: PixelsPerMm) {
95        ctx.set_line_cap(LineCap::Round);
96        ctx.move_to(from.x, from.y);
97        ctx.line_to(to.x, to.y);
98        ctx.set_line_width(px_per_mm.scale(Millimeters::new(5.0)).max(Pixels::new(0.5)).as_f64());
99        PALETTE.paper().apply(ctx);
100        ctx.stroke_preserve().ok();
101        ctx.set_line_width(px_per_mm.scale(Millimeters::new(2.2)).max(Pixels::new(0.4)).as_f64());
102        PALETTE.station().apply(ctx);
103        ctx.stroke().ok();
104    }

106    fn draw_outer(&self, ctx: &Context, spec: &OuterSpec<'_>, to_px: ToPx<'_>) {
107        if let Some(tile) = outer_tile() {
108            let pin = to_px.pos(spec.pin);
109            ctx.tile_fill(&tile, &pin, spec.width.as_f64(), spec.height.as_f64());
110        }
111        ctx.fill_ground(spec.ground, &PALETTE.paper(), to_px);
112    }

114    fn draw_block(&self, ctx: &Context, block: &BlockSpec<'_>, to_px: ToPx<'_>) {
115        ctx.set_line_join(cairo::LineJoin::Round);
116        // Parks first: green blobs under the houses (which avoid
117        // them anyway); the names draw later, over the streets.
118        for park in block.parks {
119            let blob: Vec<PxPos> = to_px.points(&park.polygon);
120            ctx.px_polyline_path(&blob);
121            ctx.close_path();
122            PALETTE.park().apply(ctx);
123            ctx.fill_preserve().ok();
124            PALETTE.block().apply_alpha(ctx, Alpha::new(0.4));
125            ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(0.4)).max(Pixels::new(0.2)).as_f64());
126            ctx.stroke().ok();
127        }
128        // Houses: raised fill over the paper, roof details in the
129        // `roof` tone (contrast against the fill), block-ink
130        // outline. Fill shade varies with width and variant.
131        ctx.draw_houses(block, &PALETTE, to_px);
132        // Decorative streets crossing the block interior - muted
133        // slate, distinct from the underground orange, under the
134        // captions.
135        ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(1.0)).max(Pixels::new(0.3)).as_f64());
136        for street in block.streets {
137            let path: Vec<PxPos> = to_px.points(street);
138            ctx.px_polyline_path(&path);
139            PALETTE.street().apply_alpha(ctx, Alpha::new(0.7));
140            ctx.stroke().ok();
141        }
142        // Park names over the streets, then the district caption.
143        for park in block.parks {
144            ctx.park_label(
145                park.name.as_str(),
146                park.polygon.centroid(),
147                to_px,
148                block.px_per_mm,
149                &CaptionStyle::park(PALETTE.paper(), PALETTE.label(), block.park_orientation),
150            );
151        }
152        if let Some(label) = &block.label {
153            ctx.district_label(
154                label,
155                to_px,
156                block.px_per_mm,
157                &CaptionStyle::district(PALETTE.paper(), PALETTE.label(), block.district_orientation),
158            );
159        }
160        // Block outline on top.
161        if block.outline.len() >= 3 {
162            let outline: Vec<PxPos> = to_px.points(block.outline);
163            ctx.px_polyline_path(&outline);
164            ctx.close_path();
165            ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(0.6)).max(Pixels::new(0.3)).as_f64());
166            PALETTE.block().apply(ctx);
167            ctx.stroke().ok();
168        }
169    }

171    fn draw_edge(&self, ctx: &Context, edge: &EdgeSpec<'_>) {
172        let px = edge.px_per_mm;
173        let color = if edge.highlight {
174            PALETTE.transport_color(edge.transport, edge.line, edge.map)
175        } else {
176            PALETTE.edge()
177        };
178        ctx.set_line_cap(LineCap::Round);
179        if edge.wavy {
180            // Boat wake: a dark rim under the wave keeps the cyan
181            // swell readable over the water band.
182            ctx.edge_path(edge);
183            ctx.set_line_width(px.scale(Millimeters::new(2.2)).max(Pixels::new(0.5)).as_f64());
184            PALETTE.paper().apply(ctx);
185            ctx.stroke_preserve().ok();
186            ctx.set_line_width(px.scale(Millimeters::new(1.3)).max(Pixels::new(0.4)).as_f64());
187            color.apply(ctx);
188            ctx.stroke().ok();
189            return;
190        }
191        if edge.dashed {
192            // Metro lane in the combined overlay: a dashed
193            // line-color stroke with a dark rim separates it from
194            // lookalike transport lanes (M1 vs taxi). Dark line
195            // colors lift toward white, or M12-vanish-into-paper.
196            let color = color.with_min_luminance(Luminance::new(0.5));
197            ctx.set_dash(&[px.scale(Millimeters::new(4.0)).as_f64(), px.scale(Millimeters::new(2.5)).as_f64()], 0.0);
198            ctx.edge_path(edge);
199            ctx.set_line_width(px.scale(Millimeters::new(2.4)).max(Pixels::new(0.5)).as_f64());
200            PALETTE.paper().apply(ctx);
201            ctx.stroke_preserve().ok();
202            ctx.set_line_width(px.scale(Millimeters::new(1.4)).max(Pixels::new(0.4)).as_f64());
203            color.apply(ctx);
204            ctx.stroke().ok();
205            ctx.set_dash(&[], 0.0);
206            return;
207        }
208        ctx.set_line_width(
209            px.scale(Millimeters::new(if edge.highlight { 3.6 } else { 1.6 }))
210                .max(Pixels::new(0.5))
211                .as_f64(),
212        );
213        color.apply_alpha(ctx, Alpha::new(if edge.highlight { 0.28 } else { 0.9 }));
214        ctx.edge_path(edge);
215        ctx.stroke_preserve().ok();
216        ctx.set_line_width(px.scale(Millimeters::new(1.2)).max(Pixels::new(0.4)).as_f64());
217        color.apply(ctx);
218        ctx.stroke().ok();
219    }

221    fn draw_landmark(&self, ctx: &Context, pos: PxPos, scale: ScaleFactor, px_per_mm: PixelsPerMm) {
222        ctx.new_path();
223        PALETTE.accent().apply_alpha(ctx, Alpha::new(0.5));
224        ctx.set_line_width(1.0);
225        ctx.arc(pos.x, pos.y, px_per_mm.scale(Millimeters::new(2.0 * scale.get())).max(Pixels::new(0.5)).as_f64(), 0.0, TAU);
226        ctx.stroke().ok();
227    }

229    fn draw_station(&self, ctx: &Context, spec: &StationSpec<'_>) {
230        // Labels leave a current point; `arc` would otherwise join it.
231        ctx.new_path();
232        let pos = spec.pos;
233        let px_per_mm = spec.px_per_mm;
234        let size = match spec.importance {
235            Some(ImportanceClass::Minor) => 0.75,
236            Some(ImportanceClass::Hub) => 1.35,
237            _ => 1.0,
238        };
239        let radius = px_per_mm
240            .scale(Millimeters::new(2.4 * size * spec.station_scale.get()))
241            .max(Pixels::new(0.4))
242            .as_f64();
243        if spec.metro_overlay && !spec.lines.is_empty() {
244            let sectors = spec.lines.len() as f64;
245            for (i, line) in spec.lines.iter().enumerate() {
246                let start = -std::f64::consts::FRAC_PI_2 + i as f64 * TAU / sectors;
247                ctx.move_to(pos.x, pos.y);
248                ctx.arc(pos.x, pos.y, radius, start, start + TAU / sectors);
249                ctx.close_path();
250                PALETTE.transport_color(Transport::Metro, Some(line), spec.map).apply(ctx);
251                ctx.fill().ok();
252            }
253        } else {
254            ctx.arc(pos.x, pos.y, radius, 0.0, TAU);
255            PALETTE.station().apply(ctx);
256            ctx.fill_preserve().ok();
257            ctx.set_line_width(0.8);
258            PALETTE.paper().apply(ctx);
259            ctx.stroke().ok();
260        }
261        for (i, transport) in spec.served.iter().enumerate() {
262            ctx.arc(
263                pos.x,
264                pos.y,
265                radius
266                    + px_per_mm
267                        .scale(Millimeters::new(0.9 * spec.ring_width.get() * (spec.served.len() - i) as f64))
268                        .max(Pixels::new(0.4))
269                        .as_f64(),
270                0.0,
271                TAU,
272            );
273            PALETTE.base_color(*transport).apply(ctx);
274            ctx.set_line_width(0.8 * spec.ring_width.get());
275            ctx.stroke().ok();
276        }
277        if spec.selected {
278            ctx.arc(pos.x, pos.y, radius + px_per_mm.scale(Millimeters::new(2.6)).max(Pixels::new(0.4)).as_f64(), 0.0, TAU);
279            PALETTE.select().apply(ctx);
280            ctx.set_line_width(1.6);
281            ctx.stroke().ok();
282        }
283    }

285    fn draw_label(&self, ctx: &Context, spec: &LabelSpec<'_>) {
286        let pos = spec.pos;
287        ctx.select_font_face("monospace", FontSlant::Normal, FontWeight::Normal);
288        ctx.set_font_size(
289            spec.px_per_mm
290                .scale(Millimeters::new(if spec.inside { 3.0 } else { 3.4 } * spec.label_scale.get()))
291                .max(Pixels::new(7.0))
292                .as_f64(),
293        );
294        let Ok(extents) = ctx.text_extents(spec.text) else { return };
295        let _saved = ctx.save();
296        let angle = spec.orientation.angle(pos);
297        if !angle.is_zero() {
298            ctx.translate(pos.x, pos.y);
299            ctx.rotate(angle.as_f64());
300            ctx.translate(-pos.x, -pos.y);
301        }
302        let y = if spec.inside {
303            pos.y + extents.height() / 2.0
304        } else {
305            pos.y + spec.radius.as_f64() + spec.px_per_mm.scale(Millimeters::new(5.0)).max(Pixels::new(9.0)).as_f64()
306        };
307        ctx.new_path();
308        ctx.move_to(pos.x - extents.width() / 2.0, y);
309        // In-disc numbers draw in paper ink - `label` is cyan glow.
310        if spec.inside {
311            PALETTE.paper().apply(ctx);
312        } else {
313            PALETTE.label().apply_alpha(ctx, Alpha::new(0.85));
314        }
315        ctx.show_text(spec.text).ok();
316        let _restored = ctx.restore();
317    }

319    fn draw_anchor(&self, ctx: &Context, pos: PxPos, tracked: bool, px_per_mm: PixelsPerMm) {
320        ctx.new_path();
321        let radius = px_per_mm.scale(Millimeters::new(5.0)).max(Pixels::new(0.5)).as_f64();
322        let color = if tracked { PALETTE.accent() } else { PALETTE.dim() };
323        ctx.arc(pos.x, pos.y, radius, 0.0, TAU);
324        color.apply_alpha(ctx, Alpha::new(0.35));
325        ctx.set_line_width(4.0);
326        ctx.stroke_preserve().ok();
327        color.apply(ctx);
328        ctx.set_line_width(1.2);
329        ctx.stroke().ok();
330    }

332    fn draw_hint(&self, ctx: &Context, text: &str, width: Pixels, height: Pixels, bottom: Pixels) {
333        ctx.select_font_face("monospace", FontSlant::Normal, FontWeight::Bold);
334        ctx.set_font_size(16.0);
335        let Ok(extents) = ctx.text_extents(text) else { return };
336        let (pad_x, pad_y) = (14.0, 10.0);
337        let (box_w, box_h) = (extents.width() + 2.0 * pad_x, extents.height() + 2.0 * pad_y);
338        let (x, y) = ((width.as_f64() - box_w) / 2.0, height.as_f64() - box_h - 24.0 - bottom.as_f64());
339        PALETTE.panel().apply_alpha(ctx, Alpha::new(0.85));
340        ctx.rectangle(x, y, box_w, box_h);
341        ctx.fill().ok();
342        PALETTE.accent().apply(ctx);
343        ctx.move_to(x + pad_x - extents.x_bearing(), y + pad_y - extents.y_bearing());
344        ctx.show_text(text).ok();
345    }
