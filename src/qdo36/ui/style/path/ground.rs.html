1//! Ground and texture helpers shared by the style renderers.
2
3use crate::ui::style::PxPos;
4use crate::ui::style::ToPx;
5use crate::ui::style::path::edge::EdgePathExt;
6use cairo::Context;
7use cairo::ImageSurface;
8use qdo36_map::Ground;
9use smearor_commons_color::Color;
10
11/// Cairo ground fills: `ctx.fill_ground(..)` floods the city's
12/// ground mask, `ctx.tile_fill(..)` repeats the outer texture.
13pub trait GroundExt {
14    /// Fills every ground polygon with `color` - the city rests
15    /// on plain paper while the countryside keeps its texture.
16    fn fill_ground(&self, ground: &Ground, color: &Color<f64>, to_px: ToPx<'_>);
17    /// Fills the whole widget with `tile` repeated, anchored so
18    /// the tile origin sits at `pin` - the texture repeats
19    /// infinitely and pans with the map.
20    fn tile_fill(&self, tile: &ImageSurface, pin: &PxPos, width: f64, height: f64);
21}
22
23impl GroundExt for Context {
24    fn fill_ground(&self, ground: &Ground, color: &Color<f64>, to_px: ToPx<'_>) {
25        color.apply(self);
26        for polygon in ground {
27            if polygon.len() < 3 {
28                continue;
29            }
30            let points: Vec<PxPos> = to_px.points(polygon);
31            self.px_polyline_path(&points);
32            self.close_path();
33            self.fill().ok();
34        }
35    }
36
37    fn tile_fill(&self, tile: &ImageSurface, pin: &PxPos, width: f64, height: f64) {
38        let pattern = cairo::SurfacePattern::create(tile);
39        pattern.set_extend(cairo::Extend::Repeat);
40        pattern.set_matrix(cairo::Matrix::new(1.0, 0.0, 0.0, 1.0, -pin.x, -pin.y));
41        self.set_source(&pattern).ok();
42        self.rectangle(0.0, 0.0, width, height);
43        self.fill().ok();
44    }
45}
