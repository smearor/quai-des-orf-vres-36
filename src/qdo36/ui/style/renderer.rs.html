1//! `StyleRenderer` - the draw interface a map style implements.
2
3use cairo::Context;
4use tangible_objects_model::Millimeters;
5use tangible_objects_model::Pixels;
6use tangible_objects_model::PixelsPerMm;
7use tangible_objects_model::Vec2;
8
9use crate::ui::style::EdgeSpec;
10use crate::ui::style::RgbColor;
11
12/// A position in widget pixels - `to_px` produces these, the style
13/// draws them.
14pub type PxPos = Vec2<Pixels>;
15
16/// How the scene's elements are drawn.
17///
18/// All coordinates arrive in widget px - the stage (`MapTransform` +
19/// px/mm) already applied. `px_per_mm` scales stroke widths.
20pub trait StyleRenderer {
21    /// The paper color the scene is painted on.
22    fn paper(&self) -> RgbColor;
23
24    /// A water polyline (river, canal).
25    fn draw_water(&self, ctx: &Context, points: &[Vec2<Millimeters>], px_per_mm: PixelsPerMm, to_px: &dyn Fn(Vec2<Millimeters>) -> PxPos);
26
27    /// A station pair's edge.
28    fn draw_edge(&self, ctx: &Context, edge: &EdgeSpec<'_>);
29
30    /// A backdrop landmark marker.
31    fn draw_landmark(&self, ctx: &Context, pos: PxPos, px_per_mm: PixelsPerMm);
32
33    /// A station node (fill + outline + selection ring).
34    fn draw_station(&self, ctx: &Context, pos: PxPos, selected: bool, px_per_mm: PixelsPerMm);
35
36    /// A station name label below the node.
37    fn draw_label(&self, ctx: &Context, name: &str, pos: PxPos, radius: Pixels, px_per_mm: PixelsPerMm);
38
39    /// The anchor marker (filled = tracked, hollow = virtual).
40    fn draw_anchor(&self, ctx: &Context, pos: PxPos, tracked: bool, px_per_mm: PixelsPerMm);
41
42    /// The overlay hint badge (tower missing etc).
43    fn draw_hint(&self, ctx: &Context, text: &str, width: Pixels, height: Pixels);
44}
