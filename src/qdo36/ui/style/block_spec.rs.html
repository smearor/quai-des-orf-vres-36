1//! `BlockSpec` - one city block the scene draws.
2
3use crate::ui::style::DistrictLabel;
4use crate::ui::style::HouseMode;
5use crate::ui::style::LabelOrientation;
6use crate::ui::style::ParkSpot;
7use crate::ui::style::PxPos;
8use crate::ui::style::ToPx;
9use qdo36_map::House;
10use smearor_commons_geometry::Pixels;
11use smearor_commons_geometry::PixelsPerMm;
12use std::f64::consts::FRAC_1_SQRT_2;
13
14/// One city block: its inset outline plus the procedural houses
15/// inside, all in map mm.
16pub struct BlockSpec<'a> {
17    /// The block outline, already inset from its bounding streets,
18    /// river and border.
19    pub outline: &'a qdo36_map::Polygon,
20    /// Houses inside the block - quads fronting the outline ring
21    /// plus rounded corner footprints.
22    pub buildings: &'a [House],
23    /// Decorative streets crossing the block interior (visual
24    /// only - the houses stay out of their corridors).
25    pub streets: &'a [qdo36_map::Polyline],
26    /// Park blobs inside the block, kept free of houses.
27    pub parks: &'a [ParkSpot],
28    /// The district name the block carries, drawn at the deepest
29    /// inside point.
30    pub label: Option<DistrictLabel>,
31    /// District caption orientation (radial faces the nearest
32    /// table edge, screen stays upright).
33    pub district_orientation: LabelOrientation,
34    /// Park caption orientation.
35    pub park_orientation: LabelOrientation,
36    /// How the houses render (top-down footprints, fixed or
37    /// radial isometric extrusion).
38    pub house_mode: HouseMode,
39    /// Map center in widget px - the radial extrusion pivot.
40    pub center: PxPos,
41    /// Pixel scale for stroke widths.
42    pub px_per_mm: PixelsPerMm,
43}
44
45impl BlockSpec<'_> {
46    /// The block's house extrusion shift for `height` (widget
47    /// px): flat top-down blocks don't shift, classic iso shifts
48    /// up-left, radial iso shifts each block's outline centroid
49    /// toward `center` so walls face the nearest table edge.
50    pub fn extrusion_shift(&self, height: Pixels, to_px: ToPx<'_>) -> PxPos {
51        match self.house_mode {
52            HouseMode::TopDown => PxPos::ZERO,
53            HouseMode::Iso => {
54                let d = -height * FRAC_1_SQRT_2;
55                PxPos::from_px(d, d)
56            }
57            HouseMode::RadialIso => self.radial_shift(height, to_px),
58        }
59    }
60
61    /// The radial extrusion: this block's outline centroid to the
62    /// map center, scaled to `height` - walls spill outward toward
63    /// the nearest table edge. A degenerate block at the center
64    /// falls back to the fixed iso direction.
65    fn radial_shift(&self, height: Pixels, to_px: ToPx<'_>) -> PxPos {
66        let c = to_px.pos(self.outline.centroid());
67        let (dx, dy) = (self.center.x - c.x, self.center.y - c.y);
68        let len = dx.hypot(dy);
69        if len < 1.0 {
70            let d = -height * FRAC_1_SQRT_2;
71            return PxPos::from_px(d, d);
72        }
73        PxPos::from_px(height * (dx / len), height * (dy / len))
74    }
75}
