1//! `StyleColor` - the semantic color slots a map style declares.
2//!
3//! The slot names mirror the SVG style-book tokens (`ARTWORK.md`
4//! §2: bg, ink, accent, water, taxi, bus, batobus, station,
5//! highlight, panel_bg) so the cairo renderers speak the same
6//! vocabulary as the generated artwork. Metro edges are
7//! deliberately absent - their colors come from `map.meta().lines`,
8//! not from the style.
9
10use smearor_commons_color::palette_definition;
11
12palette_definition! {
13    /// The semantic color slots of a map style - the key type of
14    /// [`crate::ui::style::StylePalette`].
15    pub enum StyleColor {
16        /// The field the scene paints on.
17        Paper => "paper",
18        /// Primary foreground ink: station outlines, labels, hairlines.
19        Ink => "ink",
20        /// Signature accent: gold, neon glow, comic ink-pop.
21        Accent => "accent",
22        /// Dimmed secondary: virtual anchor, muted elements.
23        Dim => "dim",
24        /// Water accent - the river's two bank lines.
25        Water => "water",
26        /// The river body between the bank lines - darker than
27        /// `Water`, distinct from `Paper`.
28        WaterFill => "water_fill",
29        /// Selection highlight ring.
30        Select => "select",
31        /// Station node fill.
32        Station => "station",
33        /// Label and hint text.
34        Label => "label",
35        /// Hint/overlay box fill.
36        Panel => "panel",
37        /// Taxi route color.
38        Taxi => "taxi",
39        /// Bus route color.
40        Bus => "bus",
41        /// Batobus route color.
42        Batobus => "batobus",
43        /// Underground secret-route color.
44        Underground => "underground",
45        /// Generic metro color for unbranded metro marks (station
46        /// rings); edges use the per-line `[map.lines]` color
47        /// instead.
48        Metro => "metro",
49        /// Neutral edge ink - the default web color when a
50        /// transport layer is not highlighted (standard view).
51        Edge => "edge",
52        /// City-block outline - the free areas between streets,
53        /// river and border.
54        Block => "block",
55        /// Procedural building fill inside a block - slightly
56        /// raised from `Paper`.
57        Building => "building",
58        /// Park green inside a block - the dataset's named green
59        /// spots.
60        Park => "park",
61        /// Roof detail lines inside a house footprint - contrast
62        /// against `Building`, not `Block`.
63        Roof => "roof",
64        /// Decorative streets crossing a block interior - muted,
65        /// below the transport colors.
66        Street => "street"
67    }
68}
