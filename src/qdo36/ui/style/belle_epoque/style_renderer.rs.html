34/// The Belle-Époque look: parchment field, engraving-style double
35/// strokes in sepia ink, gold accents and serif labels (ARTWORK §2
36/// Belle Époque - the same voice as the generated masters).
37#[derive(Debug, Clone, Copy)]
38pub struct BelleEpoqueStyle;

40/// The countryside texture tile: fine diagonal engraving hatches
41/// on the parchment - a printed field, infinitely repeatable.
42fn outer_tile() -> Option<ImageSurface> {
43    let tile = ImageSurface::create(cairo::Format::ARgb32, 20, 20).ok()?;
44    let tile_ctx = Context::new(&tile).ok()?;
45    PALETTE.paper().apply(&tile_ctx);
46    tile_ctx.paint().ok()?;
47    // Engraved field: diagonal hatching coarse enough to read at
48    // map-fit zoom, with a second fainter pass for texture depth.
49    PALETTE.ink().apply_alpha(&tile_ctx, Alpha::new(0.16));
50    tile_ctx.set_line_width(0.8);
51    for offset in [0.0, 10.0, 20.0] {
52        tile_ctx.move_to(offset, 0.0);
53        tile_ctx.line_to(offset - 20.0, 20.0);
54    }
55    tile_ctx.stroke().ok()?;
56    PALETTE.ink().apply_alpha(&tile_ctx, Alpha::new(0.07));
57    tile_ctx.set_line_width(0.5);
58    for offset in [5.0, 15.0] {
59        tile_ctx.move_to(offset, 0.0);
60        tile_ctx.line_to(offset + 20.0, 20.0);
61    }
62    tile_ctx.stroke().ok()?;
63    Some(tile)
64}

71    fn draw_water(&self, ctx: &Context, points: &[Vec2<Millimeters>], px_per_mm: PixelsPerMm, to_px: ToPx<'_>) {
72        // Engraved river: a pale wash band framed by thin sepia
73        // bank hairlines, like the rivers on period maps. The 7 mm
74        // band width is map-space, so the river zooms with the map.
75        let banks = WaterBanks::new(points, WATER_HALF_WIDTH);
76        if banks.left.is_empty() {
77            return;
78        }
79        let left_bank: Vec<PxPos> = to_px.points(&banks.left);
80        let right_bank: Vec<PxPos> = to_px.points(&banks.right);
81        ctx.set_line_cap(LineCap::Round);
82        ctx.band_path(&left_bank, &right_bank);
83        PALETTE.water_fill().apply(ctx);
84        ctx.fill().ok();
85        ctx.set_line_width(px_per_mm.scale(Millimeters::new(0.6)).max(Pixels::new(0.3)).as_f64());
86        PALETTE.ink().apply_alpha(ctx, Alpha::new(0.55));
87        for bank in [&left_bank, &right_bank] {
88            ctx.px_polyline_path(bank);
89            ctx.stroke().ok();
90        }
91    }

93    fn draw_bridge(&self, ctx: &Context, from: PxPos, to: PxPos, px_per_mm: PixelsPerMm) {
94        ctx.set_line_cap(LineCap::Round);
95        ctx.move_to(from.x, from.y);
96        ctx.line_to(to.x, to.y);
97        ctx.set_line_width(px_per_mm.scale(Millimeters::new(5.0)).max(Pixels::new(0.5)).as_f64());
98        PALETTE.paper().apply(ctx);
99        ctx.stroke_preserve().ok();
100        ctx.set_line_width(px_per_mm.scale(Millimeters::new(2.2)).max(Pixels::new(0.4)).as_f64());
101        PALETTE.station().apply(ctx);
102        ctx.stroke().ok();
103    }

105    fn draw_outer(&self, ctx: &Context, spec: &OuterSpec<'_>, to_px: ToPx<'_>) {
106        if let Some(tile) = outer_tile() {
107            let pin = to_px.pos(spec.pin);
108            ctx.tile_fill(&tile, &pin, spec.width.as_f64(), spec.height.as_f64());
109        }
110        ctx.fill_ground(spec.ground, &PALETTE.paper(), to_px);
111    }

113    fn draw_block(&self, ctx: &Context, block: &BlockSpec<'_>, to_px: ToPx<'_>) {
114        ctx.set_line_join(cairo::LineJoin::Round);
115        // Parks first: green blobs under the houses (which avoid
116        // them anyway); the names draw later, over the streets.
117        for park in block.parks {
118            let blob: Vec<PxPos> = to_px.points(&park.polygon);
119            ctx.px_polyline_path(&blob);
120            ctx.close_path();
121            PALETTE.park().apply(ctx);
122            ctx.fill_preserve().ok();
123            PALETTE.block().apply_alpha(ctx, Alpha::new(0.4));
124            ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(0.4)).max(Pixels::new(0.2)).as_f64());
125            ctx.stroke().ok();
126        }
127        // Houses: raised fill over the paper, roof details in the
128        // `roof` tone (contrast against the fill), block-ink
129        // outline. Fill shade varies with width and variant.
130        ctx.draw_houses(block, &PALETTE, to_px);
131        // Decorative streets crossing the block interior - muted,
132        // under the captions.
133        ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(1.0)).max(Pixels::new(0.3)).as_f64());
134        for street in block.streets {
135            let path: Vec<PxPos> = to_px.points(street);
136            ctx.px_polyline_path(&path);
137            PALETTE.street().apply_alpha(ctx, Alpha::new(0.7));
138            ctx.stroke().ok();
139        }
140        // Park names over the streets, then the district caption -
141        // dark ink with a paper halo for legibility.
142        for park in block.parks {
143            ctx.park_label(
144                park.name.as_str(),
145                park.polygon.centroid(),
146                to_px,
147                block.px_per_mm,
148                &CaptionStyle::park(PALETTE.paper(), PALETTE.ink(), block.park_orientation),
149            );
150        }
151        if let Some(label) = &block.label {
152            ctx.district_label(
153                label,
154                to_px,
155                block.px_per_mm,
156                &CaptionStyle::district(PALETTE.paper(), PALETTE.ink(), block.district_orientation),
157            );
158        }
159        // Block outline on top.
160        if block.outline.len() >= 3 {
161            let outline: Vec<PxPos> = to_px.points(block.outline);
162            ctx.px_polyline_path(&outline);
163            ctx.close_path();
164            ctx.set_line_width(block.px_per_mm.scale(Millimeters::new(0.5)).max(Pixels::new(0.3)).as_f64());
165            PALETTE.block().apply(ctx);
166            ctx.stroke().ok();
167        }
168    }

170    fn draw_edge(&self, ctx: &Context, edge: &EdgeSpec<'_>) {
171        let px = edge.px_per_mm;
172        ctx.set_line_cap(LineCap::Round);
173        ctx.edge_path(edge);
174        let color = if edge.highlight {
175            PALETTE.transport_color(edge.transport, edge.line, edge.map)
176        } else {
177            PALETTE.edge()
178        };
179        if edge.transport == Transport::Underground {
180            ctx.set_dash(&[px.scale(Millimeters::new(3.2)).as_f64(), px.scale(Millimeters::new(2.4)).as_f64()], 0.0);
181            ctx.set_line_width(px.scale(Millimeters::new(1.2)).max(Pixels::new(0.4)).as_f64());
182            color.apply_alpha(ctx, Alpha::new(if edge.highlight { 0.6 } else { 0.9 }));
183            ctx.stroke().ok();
184            ctx.set_dash(&[], 0.0);
185            return;
186        }
187        if edge.dashed {
188            // Metro lane in the combined overlay: dashes with the
189            // sepia frame keep lookalike colors apart (M1 vs taxi).
190            ctx.set_dash(&[px.scale(Millimeters::new(4.0)).as_f64(), px.scale(Millimeters::new(2.5)).as_f64()], 0.0);
191        }
192        // Engraving frame: a fine sepia hairline under the route color,
193        // so every line reads like a plated double-rule - also the
194        // boat wake's rim.
195        ctx.set_line_width(px.scale(Millimeters::new(if edge.wavy { 2.2 } else { 2.6 })).max(Pixels::new(0.6)).as_f64());
196        PALETTE.ink().apply_alpha(ctx, Alpha::new(if edge.highlight { 0.45 } else { 0.25 }));
197        ctx.stroke_preserve().ok();
198        ctx.set_line_width(px.scale(Millimeters::new(1.6)).max(Pixels::new(0.4)).as_f64());
199        color.apply(ctx);
200        ctx.stroke().ok();
201        ctx.set_dash(&[], 0.0);
202    }

204    fn draw_landmark(&self, ctx: &Context, pos: PxPos, scale: ScaleFactor, px_per_mm: PixelsPerMm) {
205        ctx.new_path();
206        let radius = px_per_mm.scale(Millimeters::new(2.2 * scale.get())).max(Pixels::new(0.5)).as_f64();
207        PALETTE.ink().apply_alpha(ctx, Alpha::new(0.75));
208        ctx.set_line_width(1.0);
209        ctx.arc(pos.x, pos.y, radius, 0.0, TAU);
210        ctx.stroke().ok();
211        PALETTE.accent().apply_alpha(ctx, Alpha::new(0.6));
212        ctx.arc(pos.x, pos.y, radius * 0.6, 0.0, TAU);
213        ctx.stroke().ok();
214    }

216    fn draw_station(&self, ctx: &Context, spec: &StationSpec<'_>) {
217        // Labels leave a current point; `arc` would otherwise join it.
218        ctx.new_path();
219        let pos = spec.pos;
220        let px_per_mm = spec.px_per_mm;
221        let size = match spec.importance {
222            Some(ImportanceClass::Minor) => 0.75,
223            Some(ImportanceClass::Hub) => 1.35,
224            _ => 1.0,
225        };
226        let radius = px_per_mm
227            .scale(Millimeters::new(2.6 * size * spec.station_scale.get()))
228            .max(Pixels::new(0.4))
229            .as_f64();
230        if spec.metro_overlay && !spec.lines.is_empty() {
231            let sectors = spec.lines.len() as f64;
232            for (i, line) in spec.lines.iter().enumerate() {
233                let start = -std::f64::consts::FRAC_PI_2 + i as f64 * TAU / sectors;
234                ctx.move_to(pos.x, pos.y);
235                ctx.arc(pos.x, pos.y, radius, start, start + TAU / sectors);
236                ctx.close_path();
237                PALETTE.transport_color(Transport::Metro, Some(line), spec.map).apply(ctx);
238                ctx.fill().ok();
239            }
240        } else {
241            ctx.arc(pos.x, pos.y, radius, 0.0, TAU);
242            PALETTE.station().apply(ctx);
243            ctx.fill_preserve().ok();
244            ctx.set_line_width(1.1);
245            PALETTE.ink().apply(ctx);
246            ctx.stroke().ok();
247        }
248        for (i, transport) in spec.served.iter().enumerate() {
249            ctx.arc(
250                pos.x,
251                pos.y,
252                radius
253                    + px_per_mm
254                        .scale(Millimeters::new(0.8 * spec.ring_width.get() * (spec.served.len() - i) as f64))
255                        .max(Pixels::new(0.4))
256                        .as_f64(),
257                0.0,
258                TAU,
259            );
260            PALETTE.base_color(*transport).apply(ctx);
261            ctx.set_line_width(0.7 * spec.ring_width.get());
262            ctx.stroke().ok();
263        }
264        // Ornamental inner ring, like the plated station badges.
265        ctx.arc(pos.x, pos.y, radius * 0.68, 0.0, TAU);
266        PALETTE.accent().apply_alpha(ctx, Alpha::new(0.8));
267        ctx.set_line_width(0.6);
268        ctx.stroke().ok();
269        if spec.selected {
270            ctx.arc(pos.x, pos.y, radius + px_per_mm.scale(Millimeters::new(2.4)).max(Pixels::new(0.4)).as_f64(), 0.0, TAU);
271            PALETTE.select().apply(ctx);
272            ctx.set_line_width(1.6);
273            ctx.stroke_preserve().ok();
274            PALETTE.ink().apply_alpha(ctx, Alpha::new(0.4));
275            ctx.set_line_width(0.7);
276            ctx.stroke().ok();
277        }
278    }

280    fn draw_label(&self, ctx: &Context, spec: &LabelSpec<'_>) {
281        let pos = spec.pos;
282        ctx.select_font_face("Cormorant Garamond", FontSlant::Normal, FontWeight::Bold);
283        ctx.set_font_size(
284            spec.px_per_mm
285                .scale(Millimeters::new(if spec.inside { 3.2 } else { 3.6 } * spec.label_scale.get()))
286                .max(Pixels::new(8.0))
287                .as_f64(),
288        );
289        let Ok(extents) = ctx.text_extents(spec.text) else { return };
290        let _saved = ctx.save();
291        let angle = spec.orientation.angle(pos);
292        if !angle.is_zero() {
293            ctx.translate(pos.x, pos.y);
294            ctx.rotate(angle.as_f64());
295            ctx.translate(-pos.x, -pos.y);
296        }
297        let y = if spec.inside {
298            pos.y + extents.height() / 2.0
299        } else {
300            pos.y + spec.radius.as_f64() + spec.px_per_mm.scale(Millimeters::new(5.2)).max(Pixels::new(9.5)).as_f64()
301        };
302        ctx.new_path();
303        ctx.move_to(pos.x - extents.width() / 2.0, y);
304        // In-disc numbers draw in ink - `label` is paper-colored.
305        if spec.inside {
306            PALETTE.ink().apply(ctx);
307        } else {
308            PALETTE.label().apply(ctx);
309        }
310        ctx.show_text(spec.text).ok();
311        let _restored = ctx.restore();
312    }

314    fn draw_anchor(&self, ctx: &Context, pos: PxPos, tracked: bool, px_per_mm: PixelsPerMm) {
315        ctx.new_path();
316        let radius = px_per_mm.scale(Millimeters::new(5.5)).max(Pixels::new(0.5)).as_f64();
317        ctx.arc(pos.x, pos.y, radius, 0.0, TAU);
318        if tracked {
319            PALETTE.ink().apply(ctx);
320            ctx.fill_preserve().ok();
321            PALETTE.accent().apply(ctx);
322        } else {
323            PALETTE.dim().apply_alpha(ctx, Alpha::new(0.7));
324        }
325        ctx.set_line_width(1.6);
326        ctx.stroke().ok();
327    }

329    fn draw_hint(&self, ctx: &Context, text: &str, width: Pixels, height: Pixels, bottom: Pixels) {
330        ctx.select_font_face("Cormorant Garamond", FontSlant::Normal, FontWeight::Bold);
331        ctx.set_font_size(17.0);
332        let Ok(extents) = ctx.text_extents(text) else { return };
333        let (pad_x, pad_y) = (16.0, 11.0);
334        let (box_w, box_h) = (extents.width() + 2.0 * pad_x, extents.height() + 2.0 * pad_y);
335        let (x, y) = ((width.as_f64() - box_w) / 2.0, height.as_f64() - box_h - 24.0 - bottom.as_f64());
336        PALETTE.panel().apply_alpha(ctx, Alpha::new(0.94));
337        ctx.rectangle(x, y, box_w, box_h);
338        ctx.fill().ok();
339        PALETTE.accent().apply(ctx);
340        ctx.rectangle(x, y, box_w, box_h);
341        ctx.set_line_width(1.5);
342        ctx.stroke().ok();
343        ctx.rectangle(x + 2.5, y + 2.5, box_w - 5.0, box_h - 5.0);
344        ctx.set_line_width(0.7);
345        ctx.stroke().ok();
346        PALETTE.label().apply(ctx);
347        ctx.move_to(x + pad_x - extents.x_bearing(), y + pad_y - extents.y_bearing());
348        ctx.show_text(text).ok();
349    }
