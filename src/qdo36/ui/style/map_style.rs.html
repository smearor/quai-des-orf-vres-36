3use crate::ui::style::BelleEpoqueStyle;

5use crate::ui::style::NeonStyle;

9use crate::ui::style::comic;

11use qdo36_mcp::StyleName;
12use serde::Deserialize;

14
15/// The map look. `Comic` is the default; the other styles are seams
16/// for `ARTWORK.md` §2's palettes.
17#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Hash, Serialize, Deserialize)]
18#[serde(rename_all = "lowercase")]
19pub enum MapStyle {
20    /// Flat fills, bold outlines, halftone water (default).
21    #[default]
22    Comic,
23    /// Sepia hatching, serif labels - Belle Époque.
24    BelleEpoque,
25    /// Dark field + glow strokes - Neon.
26    Neon,
27}
28
29impl MapStyle {
30    /// The artwork namespace (`art/{family}/paris/{namespace}/`).
31    pub fn namespace(&self) -> &'static str {
32        match self {
33            Self::Comic => "comic",
34            Self::BelleEpoque => "belle-epoque",
35            Self::Neon => "neon",
36        }
37    }
38
39    /// The renderer for this style.
40    pub fn renderer(&self) -> Box<dyn StyleRenderer> {
41        match self {
42            Self::Comic => Box::new(ComicStyle),
43            Self::BelleEpoque => Box::new(BelleEpoqueStyle),
44            Self::Neon => Box::new(NeonStyle),
45        }
46    }
47
48    /// The style's semantic palette (menu previews etc).
49    pub fn palette(&self) -> &'static StylePalette {
50        match self {
51            Self::Comic => &comic::palette::PALETTE,
52            Self::BelleEpoque => &belle_epoque::palette::PALETTE,
53            Self::Neon => &neon::palette::PALETTE,
54        }
55    }
56
57    /// The CSS family class the overlay menu carries so its panel
58    /// follows the map look (`style-*` rules in `style.css`).
59    pub fn controls_class(&self) -> &'static str {
60        match self {
61            Self::Comic => "style-comic",
62            Self::BelleEpoque => "style-belle",
63            Self::Neon => "style-neon",
64        }
65    }
66}
67
68impl From<StyleName> for MapStyle {
69    fn from(name: StyleName) -> Self {
70        match name {
71            StyleName::Comic => Self::Comic,
72            StyleName::BelleEpoque => Self::BelleEpoque,
73            StyleName::Neon => Self::Neon,
74        }
75    }
76}
