1//! `EdgeSpec` - one rendered edge as the `StyleRenderer` sees it.
2
3use qdo36_map::GameMap;
4use qdo36_map::LineName;
5use qdo36_map::Transport;
6use tangible_objects_model::Millimeters;
7use tangible_objects_model::Pixels;
8use tangible_objects_model::PixelsPerMm;
9
10use crate::ui::style::PxPos;
11
12/// One rendered edge - endpoints already in widget px, plus the
13/// domain data the style may color by.
14pub struct EdgeSpec<'a> {
15    /// From-station position (px).
16    pub from: PxPos,
17    /// To-station position (px).
18    pub to: PxPos,
19    /// The edge's transport mode.
20    pub transport: Transport,
21    /// Transit line (metro only; colored per `[map.lines]`).
22    pub line: Option<&'a LineName>,
23    /// The map - line-color lookup.
24    pub map: &'a GameMap,
25    /// The drawn polyline in px (routed through intermediate stations);
26    /// empty means a straight `from` -> `to` segment.
27    pub route: Vec<PxPos>,
28    /// Perpendicular pixel offset per drawn segment - parallel
29    /// metro lines shift sideways so every line keeps its own
30    /// color; one entry per drawn segment, empty means centered.
31    pub offsets: Vec<Pixels>,
32    /// Draw the `route` as a smooth spline through its interior
33    /// points (Catmull-Rom) with straight first/last hops - the
34    /// Batobus boat course curves along the river, the pier stubs
35    /// stay straight.
36    pub smooth: bool,
37    /// Draw the edge dashed with a dark rim - metro lanes in the
38    /// combined parallel+metro overlay, so a line color matching a
39    /// neighbor lane (M1 vs taxi, M6/M12 vs bus) stays readable.
40    pub dashed: bool,
41    /// Draw the edge as a sine wave along its route - the Batobus
42    /// boat wake on the water.
43    pub wavy: bool,
44    /// Arc-length phase offset of the wave, anchored to the
45    /// water course - edges sharing a stretch trace identical
46    /// waves instead of two crossing ones.
47    pub wave_phase: Millimeters,
48    /// Whether the edge draws in its transport/line color; `false`
49    /// draws the neutral `edge` ink (standard view).
50    pub highlight: bool,
51    /// Pixels per surface millimeter.
52    pub px_per_mm: PixelsPerMm,
53}
