1//! `StationSpec` - everything a style needs to draw one station.
2
3use crate::ui::style::PxPos;
4use qdo36_map::GameMap;
5use qdo36_map::ImportanceClass;
6use qdo36_map::LineName;
7use qdo36_map::Transport;
8use qdo36_tangibles::ScaleFactor;
9use tangible_objects_model::PixelsPerMm;
10
11/// A station node under the current view: base disc, one ring per
12/// served transport and - inside the metro overlay - a pie-split
13/// fill in the colors of the lines serving it.
14#[derive(Debug)]
15pub struct StationSpec<'a> {
16    /// Station center in widget px.
17    pub pos: PxPos,
18    /// Transports serving the station (drives the rings).
19    pub served: &'a [Transport],
20    /// Metro lines serving the station (drives the pie fill).
21    pub lines: &'a [LineName],
22    /// The map - line-color lookup for the pie fill.
23    pub map: &'a GameMap,
24    /// Metro overlay active: the disc fills in line colors, split
25    /// into `lines.len()` sectors when several lines share it.
26    pub metro_overlay: bool,
27    /// Strategic importance class - scales the disc radius
28    /// (`Minor` smaller, `Typical` base, `Hub` larger).
29    pub importance: Option<ImportanceClass>,
30    /// Selection ring drawn when true.
31    pub selected: bool,
32    /// Global disc radius factor (`RenderParams.station_scale`).
33    pub station_scale: ScaleFactor,
34    /// Global transport-ring width factor (`RenderParams.ring_width`).
35    pub ring_width: ScaleFactor,
36    /// Pixels per surface millimeter.
37    pub px_per_mm: PixelsPerMm,
38}
