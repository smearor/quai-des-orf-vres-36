24/// Snap radius for tap hit-testing, in map millimeters.
25const SNAP_RADIUS_MM: f64 = 12.0;

31    /// State held by the widget.
32    pub struct MapSceneImp {
33        /// The current scene state.
34        pub state: RefCell<Option<Rc<SceneState>>>,
35        /// Interpolates the rendered anchor pose toward the target in

45            }

58    impl WidgetImpl for MapSceneImp {
59        fn snapshot(&self, snapshot: &gtk4::Snapshot) {
60            let widget = self.obj();
61            let Some(state) = self.state.borrow().clone() else {
62                return;
63            };
64            let width = widget.width() as f64;
65            let height = widget.height() as f64;
66            if width <= 0.0 || height <= 0.0 {
67                return;
68            }
69            let rect = gtk4::graphene::Rect::new(0.0, 0.0, width as f32, height as f32);
70            let ctx = gtk4::prelude::SnapshotExt::append_cairo(snapshot, &rect);
71            let pose = self.animator.borrow_mut().pose_at(Instant::now());
72            super::super::render(&ctx, &state, pose, Pixels::new(width), Pixels::new(height));
73        }
74    }

77glib::wrapper! {
78    /// The map scene widget.
79    pub struct MapScene(ObjectSubclass<imp::MapSceneImp>)
80        @extends gtk4::Widget,
81        @implements gtk4::Buildable, gtk4::ConstraintTarget, gtk4::Accessible;
82}
83
84impl Default for MapScene {
85    fn default() -> Self {
86        Self::new()
87    }
88}
89
90impl MapScene {
91    /// Creates the scene widget.
92    pub fn new() -> Self {
93        let scene: Self = glib::Object::new();
94        // Frame driver: redraw only while the pose animation runs.
95        scene.add_tick_callback(|scene, _clock| {
96            if scene.imp().animator.borrow().is_animating() {
97                scene.queue_draw();
98            }
99            glib::ControlFlow::Continue
100        });
101        scene
102    }
103
104    /// Replaces the scene state, retargets the pose animation and
105    /// queues a redraw.
106    pub fn update(&self, state: Rc<SceneState>) {
107        self.imp().animator.borrow_mut().set_target(state.anchor.pose(), Instant::now());
108        *self.imp().state.borrow_mut() = Some(state);
109        self.queue_draw();
110    }
111
112    /// The station nearest to a widget-px tap (snap radius ~12 mm).
113    pub fn station_at(&self, state: &SceneState, pos: PxPos) -> Option<StationId> {
114        let pose = self.imp().animator.borrow_mut().pose_at(Instant::now());
115        let stage = RenderStage::new(state, pose, Pixels::new(self.width() as f64), Pixels::new(self.height() as f64));
116        state
117            .map
118            .nearest_station(stage.to_map(pos))
119            .filter(|nearest| nearest.distance.as_f64() <= SNAP_RADIUS_MM)
120            .map(|nearest| nearest.station)
121    }
122
123    /// The view pan keeping the map point under `from` (widget px)
124    /// beneath `to` (widget px) once the map transform becomes
125    /// `pose`/`scale` - the centroid anchor for the two-finger
126    /// pan/pinch/rotate updates (`from` == `to` pins the rotate
127    /// pivot). Computed in state space (the animator's eased pose is
128    /// render smoothing only). Keeps the current pan while the fit
129    /// has no scale.
130    pub fn anchored_pan(&self, state: &SceneState, from: PxPos, to: PxPos, pose: AnchorPose, scale: MapScale) -> Vec2<Millimeters> {
131        let stage = RenderStage::new(state, state.anchor.pose(), Pixels::new(self.width() as f64), Pixels::new(self.height() as f64));
132        if stage.fit.scale.as_f64() <= 0.0 {
133            return state.pan;
134        }
135        let anchored = stage.to_map(from);
136        let transform = MapTransform::new(pose, &state.config, scale);
137        let target = stage.fit.unmap(to);
138        let shifted = transform.map_to_surface(anchored);
139        Vec2::new(target.x - shifted.x, target.y - shifted.y)
140    }
