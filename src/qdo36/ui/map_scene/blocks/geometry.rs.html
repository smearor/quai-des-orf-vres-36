30/// The mode-dependent block render data: drawable inner blocks
31/// plus the ground mask covering the whole city - everything
32/// outside it takes the style's outer texture instead.
33pub struct BlocksGeometry {
34    /// Drawable blocks: inner faces with a valid inset, without
35    /// the faces underground lines suppress.
36    pub shapes: Vec<BlockShapes>,
37    /// Ground mask covering the city: every inner face's raw
38    /// outline expanded by each edge's setback, so the fill reaches
39    /// exactly the outer faces' inset edge - streets, corridors
40    /// and suppressed faces all rest on ground, never on texture.
41    pub ground: Ground,
42}
43
44/// The cache key for the mode-dependent block geometry.
45pub fn blocks_key(parallel: bool, metro_overlay: bool) -> u8 {
46    u8::from(parallel) | (u8::from(metro_overlay) << 1)
47}
48
49impl BlocksGeometry {
50    /// Computes every face's inset outline, houses and the city
51    /// ground mask for the current view mode.
52    pub fn compute(layout: &BlockLayout, parallel: bool, metro_overlay: bool) -> Self {
53        let margin = |cut: &BlockCut| layout.setback(cut, parallel, metro_overlay).as_f64();
54        let params = BuildingParams::default();
55        let mut shapes = Vec::new();
56        let mut ground = Ground::default();
57        for (index, face) in layout.faces.iter().enumerate() {
58            if face.is_outer() {
59                continue;
60            }
61            ground.0.push(face.outset(&margin));
62            if let Some(outline) = face.inset(&margin) {
63                // Obstacles the houses stay out of: tunnel corridors,
64                // decorative street corridors, park blobs.
65                let mut obstacles = Obstacles(face.tunnels.iter().map(|tunnel| tunnel.corridor(Millimeters::new(0.8))).collect());
66                let streets = Street::of_block(&outline, index as u64);
67                for street in &streets {
68                    for pair in street.windows(2) {
69                        obstacles.0.push(Segment::new(pair[0], pair[1]).corridor(Millimeters::new(1.6)));
70                    }
71                }
72                let parks: Vec<ParkSpot> = face
73                    .parks
74                    .iter()
75                    .filter_map(|park| {
76                        park.polygon(&outline, index as u64).map(|polygon| ParkSpot {
77                            name: park.name.clone(),
78                            polygon,
79                        })
80                    })
81                    .collect();
82                for park in &parks {
83                    obstacles.0.push(park.polygon.clone());
84                }
85                // Landmark volumes: a landmark inside the block becomes
86                // a tall octagonal house - its pin stays the roof plate.
87                let landmark_footprints: Vec<Polygon> = face.landmarks.iter().filter_map(|landmark| landmark.footprint(&outline)).collect();
88                obstacles.0.extend(landmark_footprints.iter().cloned());
89                // The district caption dodges the park blobs so it
90                // never lands on a park name.
91                let avoid: Vec<Polygon> = parks.iter().map(|park| park.polygon.clone()).collect();
92                let label = face.district.as_ref().and_then(|name| {
93                    outline.deepest_point_avoiding(&avoid).map(|pos| DistrictLabel {
94                        name: Rc::from(name.as_str()),
95                        pos,
96                    })
97                });
98                let mut buildings = params.houses(&outline, index as u64, &obstacles);
99                buildings.extend(landmark_footprints.into_iter().map(|footprint| House {
100                    footprint,
101                    units: Units::ONE,
102                    variant: Variant::LANDMARK,
103                    depth: Depth::LANDMARK,
104                    height: Height::LANDMARK,
105                }));
106                shapes.push(BlockShapes {
107                    buildings,
108                    outline,
109                    streets: streets.into_iter().map(|street| street.0).collect(),
110                    parks,
111                    label,
112                });
113            }
114        }
115        Self { shapes, ground }
116    }
117
118    /// Draws every inner block's inset outline and houses. `layers`
119    /// picks the caption orientation per district/park label;
120    /// `center` (widget px) is the radial pivot.
121    pub fn draw(&self, ctx: &Context, renderer: &dyn StyleRenderer, layers: &SceneLayers, center: PxPos, px_per_mm: PixelsPerMm, to_px: ToPx<'_>) {
122        for block in &self.shapes {
123            renderer.draw_block(
124                ctx,
125                &BlockSpec {
126                    outline: &block.outline,
127                    buildings: &block.buildings,
128                    streets: &block.streets,
129                    parks: &block.parks,
130                    label: block.label.clone(),
131                    district_orientation: layers.district_mode.orientation(center),
132                    park_orientation: layers.park_mode.orientation(center),
133                    house_mode: layers.house_mode,
134                    center,
135                    px_per_mm,
136                },
137                to_px,
138            );
139        }
140    }
