1//! `WaterRoute` - a route drawn along a water course.
2
3use qdo36_map::Polyline;
4use smearor_commons_geometry::Millimeters;
5use smearor_commons_geometry::Vec2;
6
7/// A boat route along a water course: the drawn route polyline
8/// and the arc length at which it enters the course - the entry
9/// arc anchors the wave phase so edges sharing a water stretch
10/// trace matching waves.
11pub struct WaterRoute {
12    /// The drawn route polyline in map mm.
13    pub route: Polyline,
14    /// The entry point's arc length along the course in map mm.
15    pub entry_arc: Millimeters,
16}
17
18impl WaterRoute {
19    /// The pier-to-pier course a Batobus edge draws: both station
20    /// positions are projected onto the closest water course and
21    /// the route follows the waterline between the two projected
22    /// points - the boat hugs the river instead of cutting corners
23    /// over land. `water` carries the same smoothed courses the
24    /// bands draw from. Empty when the map ships no water.
25    pub fn along(water: &[Polyline], from: Vec2<Millimeters>, to: Vec2<Millimeters>) -> Self {
26        // The closest point on a polyline: (position along the line
27        // as segment index + fraction, projected point, approach
28        // distance).
29        let closest = |points: &Polyline, pos: Vec2<Millimeters>| {
30            points
31                .nearest_segment(&pos)
32                .map(|(index, projection)| (index as f64 + projection.fraction, projection.point, pos.distance(&projection.point)))
33        };
34        let Some((points, entry, exit)) = water
35            .iter()
36            .filter_map(|points| {
37                let (Some(entry), Some(exit)) = (closest(points, from), closest(points, to)) else {
38                    return None;
39                };
40                Some((points, entry, exit))
41            })
42            .min_by(|(_, a_entry, a_exit), (_, b_entry, b_exit)| (a_entry.2 + a_exit.2).total_cmp(&(b_entry.2 + b_exit.2)))
43        else {
44            return Self {
45                route: Polyline::default(),
46                entry_arc: Millimeters::ZERO,
47            };
48        };
49        // Follow the course from `entry` to `exit` - the inner course
50        // vertices run in polyline order when `entry` is upstream, in
51        // reverse otherwise.
52        let (low, high) = if entry.0 <= exit.0 { (entry, exit) } else { (exit, entry) };
53        let i0 = low.0.floor() as usize + 1;
54        let i1 = high.0.floor() as usize;
55        let inner: &[Vec2<Millimeters>] = if i0 <= i1 { &points[i0..=i1] } else { &[] };
56        let mut route = vec![from, entry.1];
57        if entry.0 <= exit.0 {
58            route.extend_from_slice(inner);
59        } else {
60            route.extend(inner.iter().rev().copied());
61        }
62        route.extend_from_slice(&[exit.1, to]);
63        // The entry point's arc length along the course anchors the
64        // wave phase: edges sharing a stretch draw matching waves.
65        let entry_arc = points.arc_length(entry.0);
66        Self {
67            route: Polyline::new(route),
68            entry_arc,
69        }
70    }
71}
