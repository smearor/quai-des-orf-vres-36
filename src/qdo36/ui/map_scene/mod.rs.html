57/// Full scene render: paper, backdrop, edges, stations, anchor,
58/// overlay - each element delegated to the active `StyleRenderer`.
59/// `pose` is the interpolated anchor pose for this frame; `width`x
60/// `height` is the widget size in px.
61pub fn render(ctx: &Context, state: &SceneState, pose: AnchorPose, width: Pixels, height: Pixels) {
62    let map = state.map.as_ref();
63    let stage = RenderStage::new(state, pose, width, height);
64    let to_px = |pos| stage.to_px(pos);
65    let px_per_mm = stage.fit.scale;
66    let renderer = state.style.renderer();
67
68    // Paper.
69    renderer.paper().apply(ctx);
70    ctx.paint().ok();
71
72    // Backdrop: water + landmarks (gameplay-inert dressing).
73    let backdrop = map.backdrop();
74    if state.layers.water {
75        for polyline in &backdrop.water {
76            renderer.draw_water(ctx, polyline.points(), px_per_mm, &to_px);
77        }
78    }
79    if state.layers.landmarks {
80        for landmark in &backdrop.landmarks {
81            let pos = to_px(landmark.pos);
82            let spec = ArtSpec {
83                family: ArtFamily::Landmarks,
84                stem: AssetStem::for_landmark(landmark.name.as_str()),
85                pos,
86                height: Millimeters::new(14.0 * state.params.landmark_scale.get()),
87                anchor: AnchorFrac::PIN_TIP,
88                alpha: Alpha::OPAQUE,
89            };
90            if !state.assets.draw(ctx, state.style, &spec, px_per_mm) {
91                renderer.draw_landmark(ctx, pos, state.params.landmark_scale, px_per_mm);
92            }
93        }
94    }
95
96    // Bridge decks: where a visible taxi/bus edge crosses the
97    // water, a pale plank under the road line (decorative).
98    if state.layers.water {
99        let water_segments: Vec<Segment<Millimeters>> = backdrop
100            .water
101            .iter()
102            .flat_map(|polyline| polyline.points().windows(2).map(|pair| Segment::new(pair[0], pair[1])))
103            .collect();
104        for edge in map.edges() {
105            if !matches!(edge.transport, Transport::Taxi | Transport::Bus) || !state.layers.transports.visible(edge.transport) {
106                continue;
107            }
108            let (Some(from), Some(to)) = (map.station(edge.from), map.station(edge.to)) else {
109                continue;
110            };
111            // Strict crossing only - an edge merely ending on the
112            // waterline is a pier access, not a bridge (Q13 rule).
113            let road = Segment::new(from.pos, to.pos);
114            let Some(cross) = water_segments
115                .iter()
116                .find_map(|water| if road.crosses(water) { road.intersection(water) } else { None })
117            else {
118                continue;
119            };
120            let (dx, dy) = (to.pos.x - from.pos.x, to.pos.y - from.pos.y);
121            if dx == 0.0 && dy == 0.0 {
122                continue;
123            }
124            let half = 6.0 / dx.hypot(dy);
125            renderer.draw_bridge(
126                ctx,
127                to_px(Vec2::new(cross.x - dx * half, cross.y - dy * half)),
128                to_px(Vec2::new(cross.x + dx * half, cross.y + dy * half)),
129                px_per_mm,
130            );
131        }
132    }
133
134    // Per-station served transports drive the service rings.
135    let mut served: HashMap<StationId, Vec<Transport>> = HashMap::new();
136    for edge in map.edges() {
137        for id in [edge.from, edge.to] {
138            let list = served.entry(id).or_default();
139            if !list.contains(&edge.transport) {
140                list.push(edge.transport);
141            }
142        }
143    }
144
145    // Edges: transport layer on + the line not hidden by name;
146    // highlighted transports draw in their own colors, everything
147    // else in neutral ink. Metro edges route through the street mesh
148    // (or their declared `via`) as pass-through lines; hops shared
149    // by distinct lines draw parallel offsets in each line's color.
150    let line_hidden = |edge: &Edge| edge.line.as_ref().is_some_and(|line| state.layers.hidden_lines.contains(line));
151
152    // Resolve every metro edge's station path once (via > street
153    // > chord) and collect the distinct lines sharing each hop.
154    let mut metro_paths: Vec<(usize, Vec<StationId>)> = Vec::new();
155    let mut sharing: HashMap<(StationId, StationId), BTreeSet<Option<LineName>>> = HashMap::new();
156    for (edge_i, edge) in map.edges().iter().enumerate() {
157        if edge.transport != Transport::Metro || line_hidden(edge) || !state.layers.transports.visible(edge.transport) {
158            continue;
159        }
160        let path = if !edge.via.is_empty() {
161            let mut path = vec![edge.from];
162            path.extend_from_slice(&edge.via);
163            path.push(edge.to);
164            path
165        } else if edge.direct {
166            vec![edge.from, edge.to]
167        } else {
168            map.route_on_surface(edge.from, edge.to).unwrap_or_else(|| vec![edge.from, edge.to])
169        };
170        for w in path.windows(2) {
171            let key = if w[0] < w[1] { (w[0], w[1]) } else { (w[1], w[0]) };
172            sharing.entry(key).or_default().insert(edge.line.clone());
173        }
174        metro_paths.push((edge_i, path));
175    }
176
177    for (edge_i, edge) in map.edges().iter().enumerate() {
178        if !state.layers.transports.visible(edge.transport) || line_hidden(edge) {
179            continue;
180        }
181        let (Some(from), Some(to)) = (map.station(edge.from), map.station(edge.to)) else {
182            continue;
183        };
184        // `via` waypoints draw any edge's route as a bent polyline
185        // (metro additionally offsets hops shared by other lines).
186        let (route, offsets) = if edge.transport == Transport::Metro {
187            let path = metro_paths
188                .iter()
189                .find(|(e_i, _)| *e_i == edge_i)
190                .map(|(_, path)| path.as_slice())
191                .unwrap_or(&[]);
192            let spacing = px_per_mm.scale(1.3);
193            let offsets: Vec<f64> = path
194                .windows(2)
195                .map(|w| {
196                    let key = if w[0] < w[1] { (w[0], w[1]) } else { (w[1], w[0]) };
197                    let (count, rank) = sharing
198                        .get(&key)
199                        .map(|lines| (lines.len(), lines.iter().position(|l| *l == edge.line).unwrap_or(0)))
200                        .unwrap_or((1, 0));
201                    (rank as f64 - (count as f64 - 1.0) / 2.0) * spacing
202                })
203                .collect();
204            let route: Vec<_> = path.iter().filter_map(|id| map.station(*id)).map(|s| to_px(s.pos)).collect();
205            (route, offsets)
206        } else if !edge.via.is_empty() {
207            let route: Vec<PxPos> = [edge.from]
208                .into_iter()
209                .chain(edge.via.iter().copied())
210                .chain([edge.to])
211                .filter_map(|id| map.station(id))
212                .map(|s| to_px(s.pos))
213                .collect();
214            (route, Vec::new())
215        } else {
216            (Vec::new(), Vec::new())
217        };
218        renderer.draw_edge(
219            ctx,
220            &EdgeSpec {
221                from: to_px(from.pos),
222                to: to_px(to.pos),
223                transport: edge.transport,
224                line: edge.line.as_ref(),
225                map,
226                route: if route.len() > 2 { route } else { Vec::new() },
227                offsets,
228                highlight: state.layers.highlight.contains(&edge.transport),
229                px_per_mm,
230            },
231        );
232    }
233
234    // Per-station served metro lines drive the overlay pie fill.
235    let mut served_lines: HashMap<StationId, Vec<LineName>> = HashMap::new();
236    for edge in map.edges() {
237        if edge.transport == Transport::Metro
238            && !line_hidden(edge)
239            && let Some(line) = &edge.line
240        {
241            for id in [edge.from, edge.to] {
242                let list = served_lines.entry(id).or_default();
243                if !list.contains(line) {
244                    list.push(line.clone());
245                }
246            }
247        }
248    }
249    let metro_overlay = state.layers.highlight.contains(&Transport::Metro);
250
251    // Metro line badges on the routed paths: on the two edges
252    // carrying the line's end stations plus every third segment.
253    if metro_overlay {
254        let mut lines: BTreeMap<LineName, Vec<Vec<StationId>>> = BTreeMap::new();
255        for (edge_i, path) in &metro_paths {
256            if let Some(line) = map.edges().get(*edge_i).and_then(|e| e.line.clone()) {
257                lines.entry(line).or_default().push(path.clone());
258            }
259        }
260        for (line, paths) in &lines {
261            // Chain the line's paths into one station sequence by
262            // shared endpoints.
263            let mut chain = paths.first().cloned().unwrap_or_default();
264            let mut rest: Vec<Vec<qdo36_map::StationId>> = paths.iter().skip(1).cloned().collect();
265            while let Some(idx) = rest.iter().position(|p| p.first() == chain.last() || p.last() == chain.last()) {
266                let mut piece = rest.swap_remove(idx);
267                if piece.first() == chain.last() {
268                    chain.extend(piece.into_iter().skip(1));
269                } else {
270                    piece.reverse();
271                    chain.extend(piece.into_iter().skip(1));
272                }
273            }
274            let segments = chain.len().saturating_sub(1);
275            for i in 0..segments {
276                if i != 0 && i != segments - 1 && i % 3 != 0 {
277                    continue;
278                }
279                let (Some(a), Some(b)) = (map.station(chain[i]), map.station(chain[i + 1])) else {
280                    continue;
281                };
282                let mid = Segment::new(to_px(a.pos), to_px(b.pos)).midpoint();
283                let spec = ArtSpec {
284                    family: ArtFamily::Ligne,
285                    stem: AssetStem::for_line(line),
286                    pos: mid,
287                    height: Millimeters::new(6.0 * state.params.metro_scale.get()),
288                    anchor: AnchorFrac::CENTER,
289                    alpha: Alpha::OPAQUE,
290                };
291                state.assets.draw(ctx, state.style, &spec, px_per_mm);
292            }
293        }
294    }
295
296    // Stations + labels + numbers + scores.
297    let importance = map.importance();
298    // Radial labels face the nearest *screen* edge - the player's
299    // seat side - not the map center (which pans off-viewport).
300    let center = PxPos::new(width.as_f64() / 2.0, height.as_f64() / 2.0);
301    let orientation = match state.layers.label_mode {
302        LabelMode::Radial => LabelOrientation::Radial { center },
303        LabelMode::Screen => LabelOrientation::Screen,
304    };
305    let radius = Pixels::new(px_per_mm.scale(3.0).max(0.4));
306    for station in map.stations() {
307        let pos = to_px(station.pos);
308        let station_importance = importance.get(&station.id);
309        if state.layers.stations {
310            renderer.draw_station(
311                ctx,
312                &StationSpec {
313                    pos,
314                    served: served.get(&station.id).map(Vec::as_slice).unwrap_or_default(),
315                    lines: served_lines.get(&station.id).map(Vec::as_slice).unwrap_or_default(),
316                    map,
317                    metro_overlay,
318                    importance: station_importance.map(|i| i.class),
319                    selected: state.selected == Some(station.id),
320                    station_scale: state.params.station_scale,
321                    ring_width: state.params.ring_width,
322                    px_per_mm,
323                },
324            );
325        }
326        if state.layers.labels {
327            renderer.draw_label(
328                ctx,
329                &LabelSpec {
330                    text: station.name.as_str(),
331                    pos,
332                    radius,
333                    orientation,
334                    inside: false,
335                    label_scale: state.params.label_scale,
336                    px_per_mm,
337                },
338            );
339        }
340        if state.layers.numbers {
341            renderer.draw_label(
342                ctx,
343                &LabelSpec {
344                    text: &station.id.to_string(),
345                    pos,
346                    radius,
347                    orientation,
348                    inside: true,
349                    label_scale: state.params.label_scale,
350                    px_per_mm,
351                },
352            );
353        }
354        if state.layers.scores
355            && let Some(info) = station_importance
356        {
357            // Pushed further out than the name label.
358            renderer.draw_label(
359                ctx,
360                &LabelSpec {
361                    text: &info.score.0.to_string(),
362                    pos,
363                    radius: Pixels::new(radius.as_f64() * 2.4),
364                    orientation,
365                    inside: false,
366                    label_scale: state.params.label_scale,
367                    px_per_mm,
368                },
369            );
370        }
371    }
372
373    // Dataset-debug layer: ring every `bridge` station.
374    if state.layers.bridges {
375        for station in map.stations() {
376            if station.is_bridge() {
377                renderer.draw_landmark(ctx, to_px(station.pos), ScaleFactor::new(2.0), px_per_mm);
378            }
379        }
380    }
381
382    // Anchor marker: the Eiffel pin artwork where the style ships one,
383    // else the primitive ring.
384    let tracked = matches!(state.anchor, AnchorState::Tracked(_));
385    let anchor_pos = to_px(state.config.anchor_map_pos);
386    let anchor_spec = ArtSpec {
387        family: ArtFamily::Landmarks,
388        stem: AssetStem::new("eiffel-anchor"),
389        pos: anchor_pos,
390        height: Millimeters::new(16.0),
391        anchor: AnchorFrac::PIN_TIP,
392        alpha: if tracked { Alpha::OPAQUE } else { Alpha::new(0.4) },
393    };
394    if !state.assets.draw(ctx, state.style, &anchor_spec, px_per_mm) {
395        renderer.draw_anchor(ctx, anchor_pos, tracked, px_per_mm);
396    }
397
398    // Compass rose: rotates with the map, so it always reads map-north.
399    if state.layers.compass {
400        let compass_pos = to_px(state.compass_pos);
401        let compass_spec = ArtSpec {
402            family: ArtFamily::Map,
403            stem: AssetStem::new("compass"),
404            pos: compass_pos,
405            height: Millimeters::new(26.0),
406            anchor: AnchorFrac::CENTER,
407            alpha: Alpha::new(0.9),
408        };
409        state.assets.draw(ctx, state.style, &compass_spec, px_per_mm);
410    }
411
412    // "Tower missing" hint - only while virtual.
413    if let (Some(hint), AnchorState::Virtual(_)) = (&state.anchor_hint, state.anchor) {
414        renderer.draw_hint(ctx, hint, width, height);
415    }
416}
