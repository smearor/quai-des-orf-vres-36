87/// Full scene render: paper, backdrop, edges, stations, anchor,
88/// overlay - each element delegated to the active `StyleRenderer`.
89/// `pose` is the interpolated anchor pose for this frame; `width`x
90/// `height` is the widget size in px.
91pub fn render(ctx: &Context, state: &SceneState, pose: AnchorPose, width: Pixels, height: Pixels) {
92    let map = state.map.as_ref();
93    let stage = RenderStage::new(state, pose, width, height);
94    let to_px = |pos| stage.to_px(pos);
95    let px_per_mm = stage.fit.scale;
96    let renderer = state.style.renderer();
97
98    // Paper.
99    renderer.paper().apply(ctx);
100    ctx.paint().ok();
101
102    // Backdrop: water + landmarks (gameplay-inert dressing). The
103    // Chaikin-smoothed courses feed the river bands, the Batobus
104    // routes and the block cuts, so band, boat line and block
105    // outline can never diverge. Water vertices that sit on a
106    // `bridge` station stay pinned: the river keeps flowing right
107    // under those markers.
108    let backdrop = map.backdrop();
109    let water_courses = state.water_courses();
110
111    // Countryside texture + city ground: the style's infinite
112    // outer texture covers everything the ground polygons don't -
113    // drawn under the Seine so the river washes its textured
114    // banks too, and cached per view mode like the blocks.
115    let blocks_geometry = if state.layers.blocks {
116        let metro_overlay = state.layers.highlight.contains(&Transport::Metro);
117        let parallel = state.layers.parallel_transports;
118        let key = blocks::blocks_key(parallel, metro_overlay);
119        let cached = state.block_shapes.borrow();
120        let geometry = match cached.as_ref().filter(|(k, _)| *k == key) {
121            Some((_, geometry)) => geometry.clone(),
122            None => {
123                drop(cached);
124                let computed = Rc::new(blocks::BlocksGeometry::compute(state.block_layout(), parallel, metro_overlay));
125                *state.block_shapes.borrow_mut() = Some((key, computed.clone()));
126                computed
127            }
128        };
129        renderer.draw_outer(
130            ctx,
131            &OuterSpec {
132                ground: &geometry.ground,
133                pin: Vec2::ZERO,
134                width,
135                height,
136                px_per_mm,
137            },
138            ToPx(&to_px),
139        );
140        Some(geometry)
141    } else {
142        None
143    };
144
145    if state.layers.water {
146        for course in water_courses {
147            renderer.draw_water(ctx, course, px_per_mm, ToPx(&to_px));
148        }
149    }
150
151    // City blocks: the free areas the streets, river and border
152    // cut the map into. Drawn under everything else - lines,
153    // labels and landmarks stay on top. The insets widen with the
154    // street corridors in the parallel/metro overlays.
155    if let Some(geometry) = &blocks_geometry {
156        geometry.draw(ctx, renderer.as_ref(), &state.layers, PxPos::from_px(width / 2.0, height / 2.0), px_per_mm, ToPx(&to_px));
157    }
158    // Bridge-station decks for docking and drawing.
159    let decks = state.decks();
160    // Smoothed water segments drive the bridge crossing tests.
161    let water_segments = state.water_segments();
162    let line_hidden = |edge: &Edge| edge.line.as_ref().is_some_and(|line| state.layers.hidden_lines.contains(line));
163    // Radial orientation faces away from the *screen* center - the
164    // player's seat side - like the radial labels below.
165    let center = PxPos::from_px(width / 2.0, height / 2.0);
166    if state.layers.landmarks {
167        // Obstacle geometry the pins dodge: every station disc and
168        // each edge's station-chain chords (px space).
169        let station_discs = station_discs(state, &stage);
170        let edge_segments: Vec<(PxPos, PxPos)> = map
171            .edges()
172            .iter()
173            .flat_map(|edge| {
174                let ids: Vec<StationId> = edge.station_chain().collect();
175                ids.windows(2)
176                    .filter_map(|w| {
177                        let (Some(a), Some(b)) = (map.station(w[0]), map.station(w[1])) else {
178                            return None;
179                        };
180                        Some((to_px(a.pos), to_px(b.pos)))
181                    })
182                    .collect::<Vec<_>>()
183            })
184            .collect();
185        for landmark in &backdrop.landmarks {
186            let pos = to_px(landmark.pos);
187            let height_mm = 14.0 * state.params.landmark_scale.get();
188            let preferred = state.layers.landmark_mode.preferred(pos, center);
189            let angle = landmark_rotation(
190                pos,
191                preferred,
192                px_per_mm.scale(Millimeters::new(height_mm * 0.5)).as_f64(),
193                px_per_mm.scale(Millimeters::new(height_mm * 0.4)).as_f64(),
194                &station_discs,
195                &edge_segments,
196            );
197            let spec = ArtSpec {
198                family: ArtFamily::Landmarks,
199                stem: AssetStem::for_landmark(landmark.name.as_str()),
200                pos,
201                height: Millimeters::new(height_mm),
202                anchor: AnchorFrac::PIN_TIP,
203                rotation: angle + RotationRad::new(FRAC_PI_2),
204                alpha: Alpha::OPAQUE,
205            };
206            if !state.assets.draw(ctx, state.style, &spec, px_per_mm) {
207                renderer.draw_landmark(ctx, pos, state.params.landmark_scale, px_per_mm);
208            }
209        }
210    }
211
212    // Resolve every metro/underground edge's station path once
213    // (metro: via > street mesh > chord, tunnels: via > chord) and
214    // collect who shares each hop: metro by distinct line,
215    // underground by edge - shared hops draw as parallel offsets
216    // instead of piling identical strokes on top of each other.
217    let mut routed_paths: Vec<(usize, EdgePath)> = Vec::new();
218    let mut sharing: Sharing<Vec<HopShare>> = Sharing::default();
219    for (edge_i, edge) in map.edges().iter().enumerate() {
220        if !matches!(edge.transport, Transport::Metro | Transport::Underground) || line_hidden(edge) || !state.layers.transports.visible(edge.transport) {
221            continue;
222        }
223        let path = map.edge_path(edge);
224        let share = HopShare::of(edge, edge_i);
225        for w in path.hops() {
226            let bucket = sharing.entry(w).or_default();
227            if !bucket.contains(&share) {
228                bucket.push(share.clone());
229            }
230        }
231        routed_paths.push((edge_i, path));
232    }
233
234    // Parallel transports: hops shared by several lanes draw side
235    // by side, each lane in its own color (metro-line style). A
236    // lane is a whole transport - taxi, bus; also metro while its
237    // overlay is off - or, with the metro overlay on, a single
238    // metro line in its line color.
239    let metro_overlay = state.layers.highlight.contains(&Transport::Metro);
240    let mut surface_sharing: Sharing<BTreeSet<ParallelShare>> = Sharing::default();
241    if state.layers.parallel_transports {
242        for edge in map.edges() {
243            if !matches!(edge.transport, Transport::Taxi | Transport::Bus) {
244                continue;
245            }
246            let ids: Vec<StationId> = edge.station_chain().collect();
247            for hop in ids.windows(2) {
248                let key = StationPair::new(hop[0], hop[1]);
249                surface_sharing.entry(key).or_default().insert(ParallelShare::Transport(edge.transport));
250            }
251        }
252        for (edge_i, path) in &routed_paths {
253            let Some(edge) = map.edges().get(*edge_i).filter(|edge| edge.is_transport(Transport::Metro)) else {
254                continue;
255            };
256            let share = ParallelShare::for_metro(edge.line.as_ref(), metro_overlay);
257            for hop in path.hops() {
258                surface_sharing.entry(hop).or_default().insert(share.clone());
259            }
260        }
261    }
262    // Lane side per hop: the rank ordering on a hop may be
263    // mirrored for all lanes at once (the lanes stay distinct);
264    // the sign is chosen so lanes keep their side through via
265    // vertices where possible. Without it, a hop traversed against
266    // its canonical direction forces every lane to cross the
267    // corridor at the vertex.
268    let hop_normal = |key: StationPair| -> Option<Vec2<Millimeters>> {
269        let (Some(a), Some(b)) = (map.station(key.a()), map.station(key.b())) else {
270            return None;
271        };
272        let direction = b.pos - a.pos;
273        (direction.magnitude() > Millimeters::new(1e-9)).then(|| direction.normalized().perpendicular())
274    };
275    // One continuity vote per via vertex: side(v) should equal
276    // side(u) * c, with c = sign(rank_u * rank_v * (n_u . n_v)).
277    let mut constraints: Vec<LaneConstraint> = Vec::new();
278    if state.layers.parallel_transports {
279        for edge in map.edges() {
280            if !matches!(edge.transport, Transport::Taxi | Transport::Bus) {
281                continue;
282            }
283            let hops: Vec<StationPair> = edge.chain().hops().collect();
284            let share = ParallelShare::Transport(edge.transport);
285            constraints.extend(LaneConstraint::chain(&hops, &surface_sharing.ranks(&hops, &share), hop_normal));
286        }
287        for (edge_i, path) in &routed_paths {
288            let Some(edge) = map.edges().get(*edge_i).filter(|edge| edge.is_transport(Transport::Metro)) else {
289                continue;
290            };
291            let share = ParallelShare::for_metro(edge.line.as_ref(), metro_overlay);
292            let hops: Vec<StationPair> = path.hops().collect();
293            constraints.extend(LaneConstraint::chain(&hops, &surface_sharing.ranks(&hops, &share), hop_normal));
294        }
295    } else {
296        for (edge_i, path) in &routed_paths {
297            let hops: Vec<StationPair> = path.hops().collect();
298            let Some(edge) = map.edges().get(*edge_i) else { continue };
299            let share = HopShare::of(edge, *edge_i);
300            constraints.extend(LaneConstraint::chain(&hops, &sharing.ranks(&hops, &share), hop_normal));
301        }
302    }
303    let hop_side = LaneSides::solve(&constraints);
304    // The hop's parallel offset for a lane: its rank among the
305    // lanes serving the hop, centered - relative to the canonical
306    // a < b direction, so opposite-direction traversals land on
307    // different lanes instead of collapsing onto one - times the
308    // hop's lane side (keeps lanes on their side at via vertices).
309    let spacing = px_per_mm.scale(Millimeters::new(2.6)).as_f64();
310    let surface_offset = |share: &ParallelShare, a: StationId, b: StationId| -> f64 { share.offset(&surface_sharing, &hop_side, a, b, spacing) };
311    let metro_in_parallel = |edge: &Edge| edge.is_transport(Transport::Metro) && state.layers.parallel_transports && !metro_overlay;
312
313    // Metro + underground draw under the surface traffic - the
314    // boats and roads ride over the tunnels.
315    let edge_route = |edge_i: usize, edge: &Edge| -> Option<RoutePlan> {
316        let (Some(_), Some(_)) = (map.station(edge.from), map.station(edge.to)) else {
317            return None;
318        };
319        // `via` waypoints draw any edge's route as a bent polyline
320        // (metro/tunnel hops shared with others draw parallel
321        // offsets). Batobus edges already drew under the bridges.
322        if matches!(edge.transport, Transport::Metro | Transport::Underground) {
323            let path = routed_paths
324                .iter()
325                .find(|(e_i, _)| *e_i == edge_i)
326                .map(|(_, path)| path.as_slice())
327                .unwrap_or(&[]);
328            let share = HopShare::of(edge, edge_i);
329            let offsets: Vec<Pixels> = if edge.is_transport(Transport::Metro) && state.layers.parallel_transports {
330                let share = ParallelShare::for_metro(edge.line.as_ref(), metro_overlay);
331                path.windows(2).map(|w| Pixels::new(surface_offset(&share, w[0], w[1]))).collect()
332            } else {
333                path.windows(2)
334                    .map(|w| {
335                        let key = StationPair::new(w[0], w[1]);
336                        let dir = key.direction(w[0]);
337                        let side = hop_side.of(&key);
338                        Pixels::new(sharing.offset(&key, &share, spacing, dir, side))
339                    })
340                    .collect()
341            };
342            let route_mm: Polyline = path.iter().filter_map(|id| map.station(*id)).map(|s| s.pos).collect();
343            let route: Vec<PxPos> = route_mm.iter().map(|pos| to_px(*pos)).collect();
344            Some(RoutePlan { route, offsets, route_mm })
345        } else {
346            // Above-ground edges dock at bridge-station deck ends;
347            // everything else follows its `via` stations (or the
348            // straight chord when `via` is empty). Hops shared by
349            // several transports draw parallel offsets in the
350            // edge's own color (parallel-transports overlay).
351            let ids: Vec<StationId> = edge.station_chain().collect();
352            let route_plan = map.road_route(&ids, decks, water_segments);
353            let route_mm = &route_plan.polyline;
354            let route: Vec<PxPos> = route_mm.iter().map(|pos| to_px(*pos)).collect();
355            let offsets: Vec<Pixels> = if state.layers.parallel_transports {
356                (0..route.len().saturating_sub(1))
357                    .map(|segment| {
358                        let hop = route_plan.hops[segment];
359                        Pixels::new(surface_offset(&ParallelShare::Transport(edge.transport), hop.a(), hop.b()))
360                    })
361                    .collect()
362            } else {
363                Vec::new()
364            };
365            Some(RoutePlan {
366                route,
367                offsets,
368                route_mm: route_plan.polyline,
369            })
370        }
371    };
372    let draw_surface_edge = |edge_i: usize, edge: &Edge| {
373        let (Some(from), Some(to)) = (map.station(edge.from), map.station(edge.to)) else {
374            return;
375        };
376        let Some(RoutePlan { route, offsets, .. }) = edge_route(edge_i, edge) else {
377            return;
378        };
379        renderer.draw_edge(
380            ctx,
381            &EdgeSpec {
382                from: to_px(from.pos),
383                to: to_px(to.pos),
384                transport: edge.transport,
385                // A metro in parallel mode draws the generic metro
386                // color, not per-line colors (those are the metro
387                // overlay's job).
388                line: if metro_in_parallel(edge) { None } else { edge.line.as_ref() },
389                map,
390                route,
391                offsets,
392                smooth: false,
393                wavy: false,
394                wave_phase: Millimeters::ZERO,
395                // A metro lane dashes while both overlays run, so
396                // a line color matching a neighboring transport
397                // lane stays readable.
398                dashed: edge.is_transport(Transport::Metro) && state.layers.parallel_transports && metro_overlay,
399                highlight: state.layers.highlight.contains(&edge.transport)
400                    || (state.layers.parallel_transports && matches!(edge.transport, Transport::Taxi | Transport::Bus))
401                    || metro_in_parallel(edge),
402                px_per_mm,
403            },
404        );
405    };
406
407    // Dim regions for the tunnel lines: the river bands and every
408    // park blob - where a metro/underground line crosses one, it
409    // fades under a paper wash.
410    let mut dim_regions: Vec<Polygon> = Vec::new();
411    for course in water_courses {
412        let banks = qdo36_map::WaterBanks::new(course, WATER_HALF_WIDTH);
413        if banks.left.len() >= 2 {
414            let mut band = banks.left;
415            band.0.extend(banks.right.iter().rev().copied());
416            dim_regions.push(Polygon::new(band.into()));
417        }
418    }
419    if let Some(geometry) = &blocks_geometry {
420        for block in &geometry.shapes {
421            for park in &block.parks {
422                dim_regions.push(park.polygon.clone());
423            }
424        }
425    }
426    for (edge_i, edge) in map.edges().iter().enumerate() {
427        if !matches!(edge.transport, Transport::Metro | Transport::Underground) || !state.layers.transports.visible(edge.transport) || line_hidden(edge) {
428            continue;
429        }
430        let Some(RoutePlan { route, route_mm, .. }) = edge_route(edge_i, edge) else {
431            continue;
432        };
433        draw_surface_edge(edge_i, edge);
434        // The tunnel fades inside every dim region it crosses.
435        for region in &dim_regions {
436            let crosses = route_mm.windows(2).any(|hop| region.crossed(&Segment::new(hop[0], hop[1])));
437            if !crosses {
438                continue;
439            }
440            let region_px: Vec<PxPos> = region.iter().map(|pos| to_px(*pos)).collect();
441            ctx.save().ok();
442            ctx.px_polyline_path(&region_px);
443            ctx.close_path();
444            ctx.clip();
445            renderer.paper().apply_alpha(ctx, Alpha::new(0.55));
446            ctx.set_line_width(px_per_mm.scale(Millimeters::new(2.6)).max(Pixels::new(0.5)).as_f64());
447            ctx.px_polyline_path(&route);
448            ctx.stroke().ok();
449            ctx.restore().ok();
450        }
451    }
452
453    // Batobus edges first: the boat course draws under the bridge
454    // decks - boats pass under bridges, not over them.
455    for edge in map.edges() {
456        if edge.transport != Transport::Batobus || line_hidden(edge) || !state.layers.transports.visible(edge.transport) {
457            continue;
458        }
459        let (Some(from), Some(to)) = (map.station(edge.from), map.station(edge.to)) else {
460            continue;
461        };
462        // A bridgehead pier docks at its bank-side deck end - the
463        // boat bends to the bank under its bridge instead of
464        // cutting onto land. On-water piers keep mid-river.
465        let dock = |id: StationId| {
466            map.station(id).map(|station| {
467                decks
468                    .get(&id)
469                    .filter(|deck| !deck.on_water)
470                    .map_or(station.pos, |deck| deck.end_on_side(station.pos))
471            })
472        };
473        let WaterRoute {
474            route: mut route_mm,
475            entry_arc,
476        } = if edge.via.is_empty() {
477            match (dock(edge.from), dock(edge.to)) {
478                (Some(a), Some(b)) => WaterRoute::along(water_courses, a, b),
479                _ => WaterRoute {
480                    route: Polyline::new(vec![from.pos, to.pos]),
481                    entry_arc: Millimeters::ZERO,
482                },
483            }
484        } else {
485            WaterRoute {
486                route: edge.station_chain().filter_map(dock).collect(),
487                entry_arc: Millimeters::ZERO,
488            }
489        };
490        // The wave phase anchored to the course: the pier stub
491        // shifts the local arc, so edges sharing a water stretch
492        // trace the same wave instead of two crossing ones.
493        let wave_phase = match (route_mm.first(), route_mm.get(1)) {
494            (Some(a), Some(b)) => entry_arc - Millimeters::new((b.x - a.x).hypot(b.y - a.y)),
495            _ => Millimeters::ZERO,
496        };
497        // Bridgehead docks bend into an arc: the water-course stub
498        // inside the bend zone is trimmed, then two waypoints -
499        // upstream of the deck center, part way toward the bank
500        // end - turn the line in over open water before it slips
501        // under the bridge.
502        if route_mm.len() >= 2
503            && let Some(deck) = decks.get(&edge.from).filter(|d| !d.on_water)
504        {
505            deck.dock_start(&mut route_mm);
506        }
507        if route_mm.len() >= 2
508            && let Some(deck) = decks.get(&edge.to).filter(|d| !d.on_water)
509        {
510            deck.dock_end(&mut route_mm);
511        }
512        let route: Vec<PxPos> = route_mm.iter().map(|pos| to_px(*pos)).collect();
513        renderer.draw_edge(
514            ctx,
515            &EdgeSpec {
516                from: to_px(from.pos),
517                to: to_px(to.pos),
518                transport: edge.transport,
519                line: edge.line.as_ref(),
520                map,
521                route,
522                offsets: Vec::new(),
523                smooth: true,
524                dashed: false,
525                wavy: true,
526                wave_phase,
527                highlight: state.layers.highlight.contains(&edge.transport) || state.layers.parallel_transports,
528                px_per_mm,
529            },
530        );
531    }
532
533    // Bridge decks: where a visible taxi/bus edge crosses the
534    // water, a pale plank under the road line (decorative). The
535    // drawn route runs from -> via... -> to, so every hop segment
536    // gets its own crossing test - `via` detours would otherwise
537    // put the plank on the straight chord, off the drawn road.
538    if state.layers.water {
539        for edge in map.edges() {
540            if !matches!(edge.transport, Transport::Taxi | Transport::Bus) || !state.layers.transports.visible(edge.transport) {
541                continue;
542            }
543            let hop_ids: Vec<StationId> = edge.station_chain().collect();
544            let hops: Vec<Vec2<Millimeters>> = hop_ids.iter().filter_map(|id| map.station(*id)).map(|station| station.pos).collect();
545            if hops.len() != hop_ids.len() {
546                continue;
547            }
548            for (index, hop) in hops.windows(2).enumerate() {
549                // A hop touching a bridge station rides on that
550                // station's deck (on-water) or is bent onto it
551                // (bridgehead) - no extra plank there.
552                if decks.contains_key(&hop_ids[index]) || decks.contains_key(&hop_ids[index + 1]) {
553                    continue;
554                }
555                let road = Segment::new(hop[0], hop[1]);
556                // Strict crossing only - an edge merely ending on
557                // the waterline is a pier access, not a bridge
558                // (Q13 rule).
559                let Some(cross) = water_segments.iter().find_map(|water| road.crossing_point(water)) else {
560                    continue;
561                };
562                let direction = road.to - road.from;
563                if direction.magnitude() <= Millimeters::new(1e-9) {
564                    continue;
565                }
566                let unit = direction.normalized();
567                renderer.draw_bridge(
568                    ctx,
569                    to_px(cross - unit * BRIDGE_HALF_SPAN.as_f64()),
570                    to_px(cross + unit * BRIDGE_HALF_SPAN.as_f64()),
571                    px_per_mm,
572                );
573            }
574        }
575        // Bridge-station decks always draw - perpendicular to the
576        // river, bank to bank, station marker mid-deck.
577        for deck in decks.values() {
578            let (from, to) = deck.ends();
579            renderer.draw_bridge(ctx, to_px(from), to_px(to), px_per_mm);
580        }
581    }
582
583    // Per-station served transports drive the service rings.
584    let served_transports = map.served_transports();
585
586    // Taxi + bus draw over the tunnels and the boat courses -
587    // surface traffic rides on top.
588    for (edge_i, edge) in map.edges().iter().enumerate() {
589        if !matches!(edge.transport, Transport::Taxi | Transport::Bus) || !state.layers.transports.visible(edge.transport) || line_hidden(edge) {
590            continue;
591        }
592        draw_surface_edge(edge_i, edge);
593    }
594
595    // Per-station served metro lines drive the overlay pie fill.
596    let served_lines = map.served_lines(&state.layers.hidden_lines);
597    // Metro line badges on the routed paths: on the two edges
598    // carrying the line's end stations plus every third segment.
599    if metro_overlay {
600        let mut lines: BTreeMap<LineName, Vec<EdgePath>> = BTreeMap::new();
601        for (edge_i, path) in &routed_paths {
602            if let Some(line) = map.edges().get(*edge_i).and_then(|e| e.line.clone()) {
603                lines.entry(line).or_default().push(path.clone());
604            }
605        }
606        for (line, paths) in &lines {
607            // Chain the line's paths into one station sequence by
608            // shared endpoints.
609            let chain = EdgePath::joined(paths.clone());
610            let segments = chain.len().saturating_sub(1);
611            for i in 0..segments {
612                if i != 0 && i != segments - 1 && i % 3 != 0 {
613                    continue;
614                }
615                let (Some(a), Some(b)) = (map.station(chain[i]), map.station(chain[i + 1])) else {
616                    continue;
617                };
618                // The badge rides on its own lane - the offset the
619                // edge draws at - so stacked lines keep their
620                // markers apart instead of printing on each other.
621                let offset = if state.layers.parallel_transports {
622                    surface_offset(&ParallelShare::for_metro(Some(line), metro_overlay), chain[i], chain[i + 1])
623                } else {
624                    let key = StationPair::new(chain[i], chain[i + 1]);
625                    let dir = key.direction(chain[i]);
626                    let side = hop_side.of(&key);
627                    sharing.offset(&key, &HopShare::Metro(line.clone()), spacing, dir, side)
628                };
629                let (a_px, b_px) = (to_px(a.pos), to_px(b.pos));
630                let normal = (b_px - a_px).normalized().perpendicular();
631                let mid = (a_px + b_px) * 0.5 + normal * offset;
632                let spec = ArtSpec {
633                    family: ArtFamily::Ligne,
634                    stem: AssetStem::for_line(line),
635                    pos: mid,
636                    height: LIGNE_BADGE_HEIGHT * state.params.metro_scale.get(),
637                    anchor: AnchorFrac::CENTER,
638                    rotation: RotationRad::new(0.0),
639                    alpha: Alpha::OPAQUE,
640                };
641                state.assets.draw(ctx, state.style, &spec, px_per_mm);
642            }
643        }
644    }
645
646    // Stations + labels + numbers + scores. Radial labels face the
647    // nearest *screen* edge - not the map center (which pans off-
648    // viewport) - the same `center` the radial landmarks use.
649    let importance = map.importance();
650    let orientation = state.layers.label_mode.orientation(center);
651    let radius = px_per_mm.scale(STATION_RADIUS_MM).max(Pixels::new(0.4));
652    for station in map.stations() {
653        let pos = to_px(station.pos);
654        let station_importance = importance.get(&station.id);
655        if state.layers.stations {
656            renderer.draw_station(
657                ctx,
658                &StationSpec {
659                    pos,
660                    served: served_transports.get(&station.id).map(Vec::as_slice).unwrap_or_default(),
661                    lines: served_lines.get(&station.id).map(Vec::as_slice).unwrap_or_default(),
662                    map,
663                    metro_overlay,
664                    importance: station_importance.map(|i| i.class),
665                    selected: state.selected == Some(station.id),
666                    station_scale: state.params.station_scale,
667                    ring_width: state.params.ring_width,
668                    px_per_mm,
669                },
670            );
671        }
672        if state.layers.labels {
673            renderer.draw_label(
674                ctx,
675                &LabelSpec {
676                    text: station.name.as_str(),
677                    pos,
678                    radius,
679                    orientation,
680                    inside: false,
681                    label_scale: state.params.label_scale,
682                    px_per_mm,
683                },
684            );
685        }
686        if state.layers.numbers {
687            renderer.draw_label(
688                ctx,
689                &LabelSpec {
690                    text: &station.id.to_string(),
691                    pos,
692                    radius,
693                    orientation,
694                    inside: true,
695                    label_scale: state.params.label_scale,
696                    px_per_mm,
697                },
698            );
699        }
700        if state.layers.scores
701            && let Some(info) = station_importance
702        {
703            // Pushed further out than the name label.
704            renderer.draw_label(
705                ctx,
706                &LabelSpec {
707                    text: &info.score.0.to_string(),
708                    pos,
709                    radius: radius * 2.4,
710                    orientation,
711                    inside: false,
712                    label_scale: state.params.label_scale,
713                    px_per_mm,
714                },
715            );
716        }
717    }
718
719    // Dataset-debug layer: ring every `bridge` station.
720    if state.layers.bridges {
721        for station in map.stations_of_kind(StationKind::Bridge) {
722            renderer.draw_landmark(ctx, to_px(station.pos), ScaleFactor::new(2.0), px_per_mm);
723        }
724    }
725
726    // Anchor marker: the Eiffel pin artwork where the style ships one,
727    // else the primitive ring.
728    let tracked = matches!(state.anchor, AnchorState::Tracked(_));
729    let anchor_pos = to_px(state.config.anchor_map_pos);
730    let anchor_spec = ArtSpec {
731        family: ArtFamily::Landmarks,
732        stem: AssetStem::new("eiffel-anchor"),
733        pos: anchor_pos,
734        height: ANCHOR_HEIGHT,
735        anchor: AnchorFrac::PIN_TIP,
736        rotation: RotationRad::new(0.0),
737        alpha: if tracked { Alpha::OPAQUE } else { Alpha::new(0.4) },
738    };
739    if !state.assets.draw(ctx, state.style, &anchor_spec, px_per_mm) {
740        renderer.draw_anchor(ctx, anchor_pos, tracked, px_per_mm);
741    }
742
743    // Compass rose: rotates with the map, so it always reads map-north.
744    if state.layers.compass {
745        let compass_pos = to_px(state.compass_pos);
746        let compass_spec = ArtSpec {
747            family: ArtFamily::Map,
748            stem: AssetStem::new("compass"),
749            pos: compass_pos,
750            height: COMPASS_HEIGHT,
751            anchor: AnchorFrac::CENTER,
752            rotation: RotationRad::new(0.0),
753            alpha: Alpha::new(0.9),
754        };
755        state.assets.draw(ctx, state.style, &compass_spec, px_per_mm);
756    }
757
758    // "Tower missing" hint - only while virtual; floats above
759    // the menu strip at the screen edge.
760    if let (Some(hint), AnchorState::Virtual(_)) = (&state.anchor_hint, state.anchor) {
761        renderer.draw_hint(ctx, hint, width, height, Pixels::new(controls::EDGE_RESERVE));
762    }
763}

765/// The station obstacle discs the landmark pins dodge - position
766/// and radius in px (the radius pads the nominal station disc by
767/// ~10 %).
768fn station_discs(state: &SceneState, stage: &RenderStage) -> Vec<(PxPos, Pixels)> {
769    state
770        .map
771        .stations()
772        .map(|station| {
773            let radius = STATION_RADIUS_MM * (1.1 * state.params.station_scale.get());
774            (stage.to_px(station.pos), stage.fit.scale.scale(radius))
775        })
776        .collect()
777}

779/// The pin direction (screen radians) a landmark artwork takes:
780/// the candidate direction least covered by station discs and
781/// edges, staying closest to `preferred` when ties allow. The pin
782/// body sits `body_offset` px out from the tip and covers
783/// `body_radius`.
784fn landmark_rotation(
785    pos: PxPos,
786    preferred: RotationRad,
787    body_offset: f64,
788    body_radius: f64,
789    stations: &[(PxPos, Pixels)],
790    edges: &[(PxPos, PxPos)],
791) -> RotationRad {
792    const STEPS: usize = 16;
793    let segment_distance = |px: f64, py: f64, a: PxPos, b: PxPos| {
794        let (vx, vy) = (b.x - a.x, b.y - a.y);
795        let len_sq = vx * vx + vy * vy;
796        let t = if len_sq > 1e-9 {
797            (((px - a.x) * vx + (py - a.y) * vy) / len_sq).clamp(0.0, 1.0)
798        } else {
799            0.0
800        };
801        (px - a.x - t * vx).hypot(py - a.y - t * vy)
802    };
803    let mut best = (f64::INFINITY, f64::INFINITY, preferred);
804    for i in 0..STEPS {
805        let angle = RotationRad::new(i as f64 * TAU / STEPS as f64);
806        let (cx, cy) = (pos.x + angle.as_f64().cos() * body_offset, pos.y + angle.as_f64().sin() * body_offset);
807        let station_cover: f64 = stations
808            .iter()
809            .map(|(s, r)| (r.as_f64() + body_radius - (cx - s.x).hypot(cy - s.y)).max(0.0))
810            .sum();
811        let edge_cover: f64 = edges.iter().map(|(a, b)| (body_radius - segment_distance(cx, cy, *a, *b)).max(0.0)).sum();
812        let occlusion = station_cover + edge_cover;
813        let deviation = angle.signed_delta_to(preferred).as_f64().abs();
814        if occlusion < best.0 - 1e-9 || (occlusion <= best.0 + 1e-9 && deviation < best.1) {
815            best = (occlusion, deviation, angle);
816        }
817    }
818    best.2
819}
