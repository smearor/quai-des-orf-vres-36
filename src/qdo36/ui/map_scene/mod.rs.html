20/// Full scene render: paper, backdrop, edges, stations, anchor,
21/// overlay - each element delegated to the active `StyleRenderer`.
22/// `pose` is the interpolated anchor pose for this frame; `width`x
23/// `height` is the widget size in px.
24pub fn render(ctx: &cairo::Context, state: &SceneState, pose: AnchorPose, width: Pixels, height: Pixels) {
25    let map = state.map.as_ref();
26    let stage = RenderStage::new(state, pose, width, height);
27    let to_px = |pos| stage.to_px(pos);
28    let px_per_mm = stage.fit.scale;
29    let renderer = state.style.renderer();
30
31    // Paper.
32    renderer.paper().apply(ctx);
33    ctx.paint().ok();
34
35    // Backdrop: water + landmarks (gameplay-inert dressing).
36    let backdrop = map.backdrop();
37    for polyline in &backdrop.water {
38        renderer.draw_water(ctx, polyline.points(), px_per_mm, &to_px);
39    }
40    for landmark in &backdrop.landmarks {
41        renderer.draw_landmark(ctx, to_px(landmark.pos), px_per_mm);
42    }
43
44    // Edges.
45    for edge in map.edges() {
46        let (Some(from), Some(to)) = (map.station(edge.from), map.station(edge.to)) else {
47            continue;
48        };
49        renderer.draw_edge(
50            ctx,
51            &EdgeSpec {
52                from: to_px(from.pos),
53                to: to_px(to.pos),
54                transport: edge.transport,
55                line: edge.line.as_ref(),
56                map,
57                px_per_mm,
58            },
59        );
60    }
61
62    // Stations + labels.
63    for station in map.stations() {
64        let pos = to_px(station.pos);
65        renderer.draw_station(ctx, pos, state.selected == Some(station.id), px_per_mm);
66        let radius = Pixels::new(px_per_mm.scale(3.0).max(0.4));
67        renderer.draw_label(ctx, station.name.as_str(), pos, radius, px_per_mm);
68    }
69
70    // Anchor marker.
71    renderer.draw_anchor(ctx, to_px(state.config.anchor_map_pos), matches!(state.anchor, AnchorState::Tracked(_)), px_per_mm);
72
73    // "Tower missing" hint - only while virtual.
74    if let (Some(hint), AnchorState::Virtual(_)) = (&state.anchor_hint, state.anchor) {
75        renderer.draw_hint(ctx, hint, width, height);
76    }
77}
