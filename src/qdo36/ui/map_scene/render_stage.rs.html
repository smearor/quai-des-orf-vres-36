1//! `RenderStage` - the transform/pixel plumbing for one render frame.
2
3use qdo36_tangibles::AnchorPose;
4use qdo36_tangibles::MapTransform;
5use qdo36_tangibles::SurfaceMm;
6use smearor_commons_geometry::Millimeters;
7use smearor_commons_geometry::Pixels;
8use smearor_commons_geometry::Vec2;
9use tangible_objects_model::SurfaceFit;
10
11use crate::ui::map_scene::SceneState;
12use crate::ui::style::PxPos;
13
14/// The shared transform/pixel plumbing for one render frame (or one
15/// inverse query): `MapTransform` (map-mm -> surface-mm) plus the
16/// upstream `SurfaceFit` (surface-mm -> widget-px, incl. display
17/// rotation + letterbox).
18pub struct RenderStage {
19    /// Uniform surface fit: rotation + scale + letterbox offset.
20    pub fit: SurfaceFit,
21    /// Map-mm -> surface-mm transform.
22    transform: MapTransform,
23    /// View pan offset in surface-mm (post-transform).
24    pan: Vec2<Millimeters>,
25}
26
27impl RenderStage {
28    /// Builds the stage for `state` at widget size `width`x`height`
29    /// (widget px), anchoring at `pose` (the interpolated pose for
30    /// this frame).
31    pub fn new(state: &SceneState, pose: AnchorPose, width: Pixels, height: Pixels) -> Self {
32        let bounds = state.map.bounds();
33        let surface = state.surface.unwrap_or_else(|| SurfaceMm {
34            width: Millimeters::new(width.as_f64().max(bounds.width().as_f64())),
35            height: Millimeters::new(height.as_f64().max(bounds.height().as_f64())),
36        });
37        Self {
38            fit: SurfaceFit::new(Vec2::from_mm(surface.width, surface.height), Vec2::from_px(width, height), state.display_rotation),
39            transform: MapTransform::new(pose, &state.config, state.scale),
40            pan: state.pan,
41        }
42    }
43
44    /// map-mm -> widget-px (transform, then the view pan).
45    pub fn to_px(&self, pos: Vec2<Millimeters>) -> PxPos {
46        let surface = self.transform.map_to_surface(pos);
47        self.fit.map(Vec2::new(surface.x + self.pan.x, surface.y + self.pan.y))
48    }
49
50    /// widget-px -> map-mm (exact inverse of `to_px`).
51    pub fn to_map(&self, pos: PxPos) -> Vec2<Millimeters> {
52        let surface = self.fit.unmap(pos);
53        self.transform.surface_to_map(Vec2::new(surface.x - self.pan.x, surface.y - self.pan.y))
54    }
55
56    /// The view pan that puts map position `target` on widget
57    /// pixel `point` under the current transform (the `goto_*`
58    /// tools' centering math).
59    pub fn pan_for(&self, target: Vec2<Millimeters>, point: PxPos) -> Vec2<Millimeters> {
60        let surface = self.transform.map_to_surface(target);
61        let want = self.fit.unmap(point);
62        Vec2::new(want.x - surface.x, want.y - surface.y)
63    }
64}
