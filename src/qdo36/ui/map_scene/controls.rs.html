3
4use crate::mcp_bridge;
5use crate::ui::map_scene::LayerName;

7use gtk4::Align;
8use gtk4::Orientation;

11/// Builds the vertical control bar anchored to the screen edge.
12/// The handlers mutate the scene through the same `mcp_bridge`
13/// helpers the MCP commands use - always in sync.
14pub fn controls() -> gtk4::Box {
15    let bar = gtk4::Box::builder()
16        .orientation(Orientation::Vertical)
17        .spacing(8)
18        .css_classes(["map-controls"])
19        .halign(Align::End)
20        .valign(Align::Center)
21        .margin_end(12)
22        .build();
23
24    let zoom_in = gtk4::Button::from_icon_name("zoom-in-symbolic");
25    zoom_in.connect_clicked(|_| mcp_bridge::zoom_by(1.25));
26    bar.append(&zoom_in);
27
28    let zoom_reset = gtk4::Button::from_icon_name("zoom-original-symbolic");
29    zoom_reset.connect_clicked(|_| mcp_bridge::reset_view());
30    bar.append(&zoom_reset);
31
32    let zoom_out = gtk4::Button::from_icon_name("zoom-out-symbolic");
33    zoom_out.connect_clicked(|_| mcp_bridge::zoom_by(0.8));
34    bar.append(&zoom_out);
35
36    let label_mode = gtk4::Button::from_icon_name("object-rotate-right-symbolic");
37    label_mode.set_tooltip_text(Some("labels: radial / screen"));
38    label_mode.connect_clicked(|_| mcp_bridge::toggle_label_mode());
39    bar.append(&label_mode);
40
41    bar.append(&layer_menu());
42    bar
43}

45/// The overlay popover: checkboxes per scene layer.
46fn layer_menu() -> gtk4::MenuButton {
47    let button = gtk4::MenuButton::builder().icon_name("view-list-symbolic").build();
48    let popover = gtk4::Popover::new();
49    let list = gtk4::Box::builder()
50        .orientation(Orientation::Vertical)
51        .spacing(4)
52        .margin_top(8)
53        .margin_bottom(8)
54        .margin_start(10)
55        .margin_end(10)
56        .build();
57    let initial = mcp_bridge::current_layers().unwrap_or_default();
58    for row in layer_toggles(&initial) {
59        let check = gtk4::CheckButton::with_label(row.label.as_str());
60        check.set_active(row.active);
61        check.connect_toggled(move |check| (row.toggle)(check.is_active()));
62        list.append(&check);
63    }
64    popover.set_child(Some(&list));
65    button.set_popover(Some(&popover));
66    button
67}
68
69/// Layer-toggle closures + the initial checkbox state, boxed so
70/// the checkbox list stays generic.
71fn layer_toggles(initial: &crate::ui::map_scene::SceneLayers) -> Vec<LayerToggle> {
72    LayerName::ALL
73        .iter()
74        .map(|&name| LayerToggle {
75            label: name,
76            toggle: Box::new(move |on| mcp_bridge::mutate_layers(move |l| l.set(name, on))),
77            active: initial.get(name),
78        })
79        .collect()
80}
