1//! `LaneConstraint` - a lane-continuity vote between two hops.
2
3use crate::ui::map_scene::lane::SideRelation;
4use qdo36_map::StationPair;
5use smearor_commons_geometry::Millimeters;
6use smearor_commons_geometry::Vec2;
7
8/// A lane-continuity constraint between two hops sharing a vertex:
9/// `relation` is the sign `side(to)` should take relative to
10/// `side(from)` so the lane keeps its spatial side through the
11/// vertex (`+1` same side, `-1` mirrored hop ordering).
12pub struct LaneConstraint {
13    /// The hop the edge arrives on.
14    pub from: StationPair,
15    /// The hop the edge departs on.
16    pub to: StationPair,
17    /// The desired side relation: `side(to) = side(from) * relation`.
18    pub relation: SideRelation,
19}
20
21impl LaneConstraint {
22    /// The constraints along `hops`' junctions given each hop's
23    /// lane `rank` and `normal_of`: `relation` is the sign of
24    /// `rank_u * rank_v * dot(n_u, n_v)` - junctions whose
25    /// normals are missing or whose product vanishes contribute
26    /// nothing.
27    pub fn chain(hops: &[StationPair], ranks: &[f64], normal_of: impl Fn(StationPair) -> Option<Vec2<Millimeters>>) -> Vec<LaneConstraint> {
28        let mut constraints = Vec::new();
29        for i in 1..hops.len() {
30            let (from, to) = (hops[i - 1], hops[i]);
31            let (Some(n_from), Some(n_to)) = (normal_of(from), normal_of(to)) else {
32                continue;
33            };
34            let relation = ranks[i - 1] * ranks[i] * n_from.dot::<f64>(&n_to);
35            if relation.abs() > 1e-9 {
36                constraints.push(Self {
37                    from,
38                    to,
39                    relation: SideRelation::of(relation),
40                });
41            }
42        }
43        constraints
44    }
45}
