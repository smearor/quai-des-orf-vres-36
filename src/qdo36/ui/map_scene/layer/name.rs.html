1//! `LayerName` - the identifier of one scene layer.
2
3use std::fmt;
4
5/// A scene layer - the MCP layer keys, the overlay checkbox
6/// captions and the `SceneLayers` fields share this identifier.
7#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
8pub enum LayerName {
9    /// Taxi edges.
10    Taxi,
11    /// Bus edges.
12    Bus,
13    /// Metro edges (+ the metro overlay extras).
14    Metro,
15    /// The metro overlay itself - highlighted lines, parallel
16    /// offsets, badges and pie-filled transfer stations.
17    MetroOverlay,
18    /// Underground edges.
19    Underground,
20    /// Batobus edges.
21    Batobus,
22    /// Parallel transports: taxi/bus hops shared by several
23    /// transports draw side by side in each transport's color.
24    ParallelTransports,
25    /// Water backdrop.
26    Water,
27    /// City-block outlines: the free areas enclosed by streets,
28    /// river and border.
29    Blocks,
30    /// Landmark markers.
31    Landmarks,
32    /// Station discs + service rings.
33    Stations,
34    /// Station name labels.
35    Labels,
36    /// In-disc station numbers.
37    Numbers,
38    /// Strategic importance scores.
39    Scores,
40    /// The compass rose.
41    Compass,
42    /// Bridge-station markers - dataset debugging only, toggled
43    /// via MCP (kept out of `ALL`, so no overlay checkbox).
44    Bridges,
45}
46
47impl LayerName {
48    /// Every toggleable layer, in overlay-menu order.
49    pub const ALL: &[LayerName] = &[
50        Self::Taxi,
51        Self::Bus,
52        Self::Metro,
53        Self::MetroOverlay,
54        Self::Underground,
55        Self::Batobus,
56        Self::ParallelTransports,
57        Self::Water,
58        Self::Blocks,
59        Self::Landmarks,
60        Self::Stations,
61        Self::Labels,
62        Self::Numbers,
63        Self::Scores,
64        Self::Compass,
65    ];
66
67    /// The layer's stable key (`"taxi"`, `"labels"`, ...) - MCP
68    /// field names and checkbox captions derive from it.
69    pub const fn as_str(self) -> &'static str {
70        match self {
71            Self::Taxi => "taxi",
72            Self::Bus => "bus",
73            Self::Metro => "metro",
74            Self::MetroOverlay => "metro overlay",
75            Self::Underground => "underground",
76            Self::Batobus => "batobus",
77            Self::ParallelTransports => "parallel transports",
78            Self::Water => "water",
79            Self::Blocks => "blocks",
80            Self::Landmarks => "landmarks",
81            Self::Stations => "stations",
82            Self::Labels => "labels",
83            Self::Numbers => "numbers",
84            Self::Scores => "scores",
85            Self::Compass => "compass",
86            Self::Bridges => "bridges",
87        }
88    }
89}
90
91impl fmt::Display for LayerName {
92    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
93        f.write_str(self.as_str())
94    }
95}
