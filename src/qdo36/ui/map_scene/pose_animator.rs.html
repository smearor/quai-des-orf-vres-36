1//! `PoseAnimator` - smooth transitions between anchor poses.
2//!
3//! Sits between the logical `AnchorState` (tracked / virtual) and the
4//! rendered pose: small per-event deltas are followed with a
5//! frame-rate independent exponential blend (tracking mode), while
6//! large jumps - re-placement elsewhere or the first pose after the
7//! virtual fallback - run a timed ease (teleport mode) so the map
8//! visibly slides and rotates into place.
9
10use qdo36_tangibles::AnchorPose;
11use std::f64::consts::PI;
12use std::time::Duration;
13use std::time::Instant;
14use tangible_objects_model::Millimeters;
15
16/// Duration of the visible ease for teleport jumps.
17const TRANSITION_DURATION: Duration = Duration::from_millis(300);
18
19/// Center jump beyond which a pose change counts as teleport.
20const TELEPORT_DISTANCE_MILLIMETERS: f64 = 80.0;
21
22/// Rotation jump beyond which a pose change counts as teleport (~25°).
23const TELEPORT_ROTATION_RAD: f64 = 25.0 * PI / 180.0;
24
25/// Tracking follow time constant: the rendered pose closes ~63% of
26/// the remaining gap per this interval.
27const TRACKING_TAU: Duration = Duration::from_millis(80);
28
29/// Residual center gap below which tracking snaps to the target.
30const SETTLE_DISTANCE_MILLIMETERS: f64 = 0.5;
31
32/// Residual rotation gap below which tracking snaps to the target.
33const SETTLE_ROTATION_RAD: f64 = 0.005;
34
35/// Interpolates the rendered anchor pose toward the latest target.
36///
37/// The widget calls [`Self::set_target`] on every incoming pose and
38/// [`Self::pose_at`] once per frame; [`Self::is_animating`] tells it
39/// whether further frames are due.
40#[derive(Debug)]
41pub struct PoseAnimator {
42    /// The pose rendered this frame.
43    rendered: AnchorPose,
44    /// The latest pose reported by the tangible stream.
45    target: AnchorPose,
46    /// Timed teleport ease: start pose and start instant; `target` is
47    /// the destination. `None` in tracking mode.
48    transition: Option<(AnchorPose, Instant)>,
49    /// When `pose_at` last ran - the tracking follow's frame delta.
50    last_frame: Instant,
51}
52
53impl PoseAnimator {
54    /// An animator holding `pose`, without an initial transition.
55    pub fn new(pose: AnchorPose, now: Instant) -> Self {
56        Self {
57            rendered: pose,
58            target: pose,
59            transition: None,
60            last_frame: now,
61        }
62    }
63
64    /// Records a new target pose.
65    ///
66    /// A jump beyond the teleport thresholds (measured against the
67    /// previous target, so poses arriving mid-ease do not restart it)
68    /// opens a timed ease from the rendered pose; smaller deltas are
69    /// left to the tracking follow.
70    pub fn set_target(&mut self, target: AnchorPose, now: Instant) {
71        let teleported = self.target.center.distance(&target.center) >= Millimeters::new(TELEPORT_DISTANCE_MILLIMETERS)
72            || self.target.rotation_rad.signed_delta_to(target.rotation_rad).as_f64().abs() >= TELEPORT_ROTATION_RAD;
73        if teleported {
74            self.transition = Some((self.rendered, now));
75        }
76        self.target = target;
77    }
78
79    /// The pose to render at `now` - advances the teleport ease or the
80    /// tracking follow toward the target.
81    pub fn pose_at(&mut self, now: Instant) -> AnchorPose {
82        match self.transition {
83            Some((from, started)) => {
84                let progress = (now.saturating_duration_since(started).as_secs_f64() / TRANSITION_DURATION.as_secs_f64()).min(1.0);
85                if progress >= 1.0 {
86                    self.transition = None;
87                    self.rendered = self.target;
88                } else {
89                    self.rendered = from.lerp(self.target, ease_out_cubic(progress));
90                }
91            }
92            None => {
93                let alpha = 1.0 - (-now.saturating_duration_since(self.last_frame).as_secs_f64() / TRACKING_TAU.as_secs_f64()).exp();
94                self.rendered = self.rendered.lerp(self.target, alpha);
95                if self.settled() {
96                    self.rendered = self.target;
97                }
98            }
99        }
100        self.last_frame = now;
101        self.rendered
102    }
103
104    /// Whether the rendered pose still differs from the target - the
105    /// widget redraws while animating.
106    pub fn is_animating(&self) -> bool {
107        self.transition.is_some() || !self.settled()
108    }
109
110    /// Whether the rendered pose has reached the target within the
111    /// settle tolerances.
112    fn settled(&self) -> bool {
113        self.rendered.center.distance(&self.target.center) <= Millimeters::new(SETTLE_DISTANCE_MILLIMETERS)
114            && self.rendered.rotation_rad.signed_delta_to(self.target.rotation_rad).as_f64().abs() <= SETTLE_ROTATION_RAD
115    }
116}
117
118/// Cubic ease-out: fast start, gentle settle.
119fn ease_out_cubic(t: f64) -> f64 {
120    1.0 - (1.0 - t).powi(3)
121}

123#[cfg(test)]
124mod tests {
125    use super::*;
126    use tangible_objects_model::RotationRad;
127    use tangible_objects_model::Vec2;
128
129    fn pose(x: f64, y: f64, rotation_rad: f64) -> AnchorPose {
130        AnchorPose {
131            center: Vec2::new(x, y),
132            rotation_rad: RotationRad::new(rotation_rad),
133        }
134    }
135
136    #[test]
137    fn small_moves_track_toward_the_target() {
138        let t0 = Instant::now();
139        let mut animator = PoseAnimator::new(pose(0.0, 0.0, 0.0), t0);
140        animator.set_target(pose(10.0, 0.0, 0.0), t0);
141
142        // One 16 ms frame: ~18% of the gap closes (EMA, tau = 80 ms).
143        let rendered = animator.pose_at(t0 + Duration::from_millis(16));
144        assert!(rendered.center.x > 0.5 && rendered.center.x < 5.0);
145        assert!(animator.is_animating());
146
147        // Long enough later the pose settles on the target and idles.
148        let rendered = animator.pose_at(t0 + Duration::from_secs(1));
149        assert_eq!(rendered.center.x, 10.0);
150        assert!(!animator.is_animating());
151    }
152
153    #[test]
154    fn large_jumps_run_a_timed_ease() {
155        let t0 = Instant::now();
156        let mut animator = PoseAnimator::new(pose(0.0, 0.0, 0.0), t0);
157        animator.set_target(pose(500.0, 0.0, 0.0), t0);
158
159        // Start of the ease: still the rendered pose.
160        assert_eq!(animator.pose_at(t0).center.x, 0.0);
161
162        // Half-way through the 300 ms ease: past the midpoint
163        // (ease-out cubic reaches 87.5%).
164        let rendered = animator.pose_at(t0 + Duration::from_millis(150));
165        assert!(rendered.center.x > 250.0 && rendered.center.x < 500.0);
166        assert!(animator.is_animating());
167
168        // Done: exactly the target, no further redraws.
169        let rendered = animator.pose_at(t0 + TRANSITION_DURATION);
170        assert_eq!(rendered.center.x, 500.0);
171        assert!(!animator.is_animating());
172    }
173
174    #[test]
175    fn mid_ease_targets_do_not_restart_the_transition() {
176        let t0 = Instant::now();
177        let mut animator = PoseAnimator::new(pose(0.0, 0.0, 0.0), t0);
178        animator.set_target(pose(500.0, 0.0, 0.0), t0);
179        animator.pose_at(t0 + Duration::from_millis(150));
180
181        // Small deltas near the new target keep the ease running and
182        // bend it toward the latest pose instead of restarting it.
183        animator.set_target(pose(510.0, 0.0, 0.0), t0 + Duration::from_millis(200));
184        let rendered = animator.pose_at(t0 + TRANSITION_DURATION);
185        assert_eq!(rendered.center.x, 510.0);
186        assert!(!animator.is_animating());
187    }
188
189    #[test]
190    fn rotation_takes_the_shortest_arc() {
191        let t0 = Instant::now();
192        let mut animator = PoseAnimator::new(pose(0.0, 0.0, 2.9), t0);
193        // Wrapped delta: +0.48 rad across ±PI, not -5.8 the long way.
194        animator.set_target(pose(0.0, 0.0, -2.9), t0);
195
196        let rendered = animator.pose_at(t0 + Duration::from_millis(150));
197        assert!(rendered.rotation_rad.as_f64() > 2.9);
198    }
199
200    #[test]
201    fn retarget_during_ease_continues_smoothly() {
202        let t0 = Instant::now();
203        let mut animator = PoseAnimator::new(pose(0.0, 0.0, 0.0), t0);
204        animator.set_target(pose(500.0, 0.0, 0.0), t0);
205        let midway = animator.pose_at(t0 + Duration::from_millis(150));
206
207        // A second teleport re-eases from the rendered pose, not the
208        // stale first target.
209        animator.set_target(pose(700.0, 0.0, 0.0), t0 + Duration::from_millis(150));
210        let rendered = animator.pose_at(t0 + Duration::from_millis(150));
211        assert_eq!(rendered.center.x, midway.center.x);
212        let rendered = animator.pose_at(t0 + Duration::from_millis(450));
213        assert_eq!(rendered.center.x, 700.0);
214    }
