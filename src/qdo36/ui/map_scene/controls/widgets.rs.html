11/// Removes all children of a box (the drawer's single section).
12pub(crate) fn clear(box_: &gtk4::Box) {
13    while let Some(child) = box_.first_child() {
14        box_.remove(&child);
15    }
16}

18/// A bar toggle opening its section inside the inline `drawer`
19/// row - unlike a popover, the row lives in the rotated widget
20/// tree, so it faces the edge's player too. `build` constructs a
21/// fresh section on each open so its controls mirror live state.
22pub(crate) fn drawer_button(
23    icon: &str,
24    tip: &str,
25    drawer: &gtk4::Box,
26    open_toggle: &Rc<RefCell<Option<gtk4::ToggleButton>>>,
27    build: impl Fn() -> gtk4::Widget + 'static,
28) -> gtk4::ToggleButton {
29    let button = gtk4::ToggleButton::builder().icon_name(icon).build();
30    if !tip.is_empty() {
31        button.set_tooltip_text(Some(tip));
32    }
33    let drawer = drawer.clone();
34    let open_toggle = open_toggle.clone();
35    button.connect_toggled(move |button| {
36        if button.is_active() {
37            // Close the previously open section's toggle first -
38            // the borrow must end before `set_active` re-enters
39            // this handler.
40            let prev = open_toggle.borrow_mut().replace(button.clone());
41            if let Some(prev) = prev
42                && prev != *button
43            {
44                prev.set_active(false);
45            }
46            clear(&drawer);
47            drawer.append(&build());
48            drawer.set_visible(true);
49        } else if open_toggle.borrow().as_ref() == Some(button) {
50            open_toggle.borrow_mut().take();
51            clear(&drawer);
52            drawer.set_visible(false);
53        }
54    });
55    button
56}

58/// A section caption inside an overlay popover.
59pub(super) fn menu_heading(text: String) -> gtk4::Label {
60    gtk4::Label::builder()
61        .label(text)
62        .halign(Align::Start)
63        .css_classes(["heading"])
64        .margin_top(4)
65        .build()
66}

68/// Appends a radio row per option; `current` picks the initial
69/// selection and `apply` publishes the picked mode.
70pub(super) fn mode_radios<T: Copy + PartialEq + 'static>(list: &gtk4::Box, l10n: &L10n, current: T, options: &[ModeOption<T>], apply: fn(T)) {
71    let mut group: Option<gtk4::CheckButton> = None;
72    for option in options {
73        let mode = option.mode;
74        let check = gtk4::CheckButton::with_label(&l10n.msg_key(option.label));
75        check.set_active(current == mode);
76        if let Some(first) = &group {
77            check.set_group(Some(first));
78        } else {
79            group = Some(check.clone());
80        }
81        check.connect_toggled(move |check| {
82            if check.is_active() {
83                apply(mode);
84            }
85        });
86        list.append(&check);
87    }
88}
