1//! `tools` - the tools drawer's screenshot actions.
2
3use crate::l10n::L10n;
4use crate::l10n::LabelKey;
5use crate::mcp_bridge;
6use crate::ui::map_scene::controls::widgets::action_row;
7use gtk4::Orientation;
8use gtk4::prelude::*;
9use tracing::debug;
10
11/// The tools drawer content: screenshot actions - the full window
12/// straight away, or an armed drag for the area selection.
13pub(crate) fn tools_list(l10n: &L10n) -> gtk4::Box {
14    let list = gtk4::Box::builder()
15        .orientation(Orientation::Horizontal)
16        .spacing(8)
17        .margin_top(8)
18        .margin_bottom(8)
19        .margin_start(10)
20        .margin_end(10)
21        .build();
22    let shot_all = action_row(&l10n.msg_key(LabelKey("controls-shot-all")));
23    shot_all.connect_clicked(move |_| {
24        debug!("tools menu: take-screenshot clicked");
25        // Schedules on after-paint: the menu toggle just
26        // invalidated the window's cached render node.
27        mcp_bridge::save_shot(None);
28    });
29    list.append(&shot_all);
30    let shot_area = action_row(&l10n.msg_key(LabelKey("controls-shot-area")));
31    shot_area.connect_clicked(move |_| {
32        debug!("tools menu: select-area clicked, arming");
33        mcp_bridge::arm_area_shot();
34    });
35    list.append(&shot_area);
36    list
37}
