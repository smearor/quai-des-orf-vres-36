15/// The layer drawer content: checkboxes per scene layer, grouped
16/// by topic so the row stays scannable.
17pub(crate) fn layer_list(l10n: &L10n) -> gtk4::Widget {
18    let list = gtk4::Box::builder()
19        .orientation(Orientation::Horizontal)
20        .spacing(16)
21        .margin_top(8)
22        .margin_bottom(8)
23        .margin_start(10)
24        .margin_end(10)
25        .build();
26    let initial = mcp_bridge::current_layers().unwrap_or_default();
27    let mut toggles: HashMap<LayerName, LayerToggle> = layer_toggles(&initial).into_iter().map(|row| (row.label, row)).collect();
28    for group in LayerGroup::ALL {
29        let column = gtk4::Box::builder().orientation(Orientation::Vertical).spacing(4).build();
30        column.append(&menu_heading(l10n.msg_key(group.label_key())));
31        for &name in group.layer_names() {
32            let Some(row) = toggles.remove(&name) else { continue };
33            let check = gtk4::CheckButton::with_label(row.label.as_str());
34            check.set_active(row.active);
35            check.connect_toggled(move |check| (row.toggle)(check.is_active()));
36            column.append(&check);
37        }
38        list.append(&column);
39    }
40    list.upcast()
41}
42
43/// Layer-toggle closures + the initial checkbox state, boxed so
44/// the checkbox list stays generic.
45fn layer_toggles(initial: &SceneLayers) -> Vec<LayerToggle> {
46    LayerName::ALL
47        .iter()
48        .map(|&name| LayerToggle {
49            label: name,
50            toggle: Box::new(move |on| mcp_bridge::mutate_layers(move |layers| layers.set(name, on))),
51            active: initial.get(name),
52        })
53        .collect()
54}
