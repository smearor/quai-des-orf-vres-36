10/// The style drawer content: one button per artwork style showing
11/// a live miniature of its palette plus the style's name.
12pub(crate) fn style_list() -> gtk4::Widget {
13    let list = gtk4::Box::builder()
14        .orientation(Orientation::Horizontal)
15        .spacing(12)
16        .margin_top(8)
17        .margin_bottom(8)
18        .margin_start(10)
19        .margin_end(10)
20        .build();
21    let styles = [("Comic", MapStyle::Comic), ("Belle Époque", MapStyle::BelleEpoque), ("Neon", MapStyle::Neon)];
22    let current = mcp_bridge::current_style().unwrap_or_default();
23    for (label, style) in styles {
24        let row = gtk4::Box::builder().orientation(Orientation::Vertical).spacing(4).build();
25        row.append(&style_preview(style));
26        row.append(&gtk4::Label::new(Some(label)));
27        let button = gtk4::Button::builder().child(&row).build();
28        if style == current {
29            button.add_css_class("checked");
30        }
31        button.connect_clicked(move |_| mcp_bridge::set_map_style(style));
32        list.append(&button);
33    }
34    list.upcast()
35}

37/// A miniature scene painted in the style's palette - paper,
38/// houses, a water band, an accent route and a station dot.
39fn style_preview(style: MapStyle) -> gtk4::DrawingArea {
40    let palette = style.palette();
41    let area = gtk4::DrawingArea::builder()
42        .content_width(64)
43        .content_height(40)
44        .hexpand(false)
45        .halign(Align::Center)
46        .build();
47    area.set_draw_func(move |_, ctx, w, h| {
48        let (w, h) = (f64::from(w), f64::from(h));
49        palette.paper.apply(ctx);
50        ctx.rectangle(0.0, 0.0, w, h);
51        ctx.fill().ok();
52        // Houses on the paper field.
53        palette.building.apply(ctx);
54        for (x, y, hw, hh) in [(0.10, 0.14, 0.22, 0.26), (0.40, 0.08, 0.26, 0.34), (0.72, 0.18, 0.20, 0.24)] {
55            ctx.rectangle(w * x, h * y, w * hw, h * hh);
56        }
57        ctx.fill().ok();
58        // The river band crossing the lower third.
59        palette.water_fill.apply(ctx);
60        ctx.move_to(0.0, h * 0.72);
61        ctx.line_to(w, h * 0.58);
62        ctx.line_to(w, h * 0.86);
63        ctx.line_to(0.0, h * 0.98);
64        ctx.close_path();
65        ctx.fill().ok();
66        // An accent route plus a station disc.
67        ctx.set_line_width(2.2);
68        palette.accent.apply(ctx);
69        ctx.move_to(w * 0.08, h * 0.62);
70        ctx.line_to(w * 0.92, h * 0.30);
71        ctx.stroke().ok();
72        palette.station.apply(ctx);
73        ctx.arc(w * 0.5, h * 0.46, 4.0, 0.0, TAU);
74        ctx.fill().ok();
75        ctx.set_line_width(1.4);
76        palette.ink.apply(ctx);
77        ctx.stroke().ok();
78    });
79    area
80}
