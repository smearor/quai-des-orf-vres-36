1//! `SceneLayers` - which scene sections the renderer draws.
2//!
3//! Toggled live via the MCP `set_layers` command for visual
4//! debugging; defaults to everything on.
5
6use crate::ui::map_scene::LabelMode;
7use crate::ui::map_scene::LayerName;
8use crate::ui::map_scene::TransportLayers;
9use crate::ui::style::HouseMode;
10use qdo36_map::LineName;
11use qdo36_map::Transport;
12use std::collections::BTreeSet;

14/// Which scene sections render.
15#[derive(Debug, Clone, PartialEq, Eq)]
16pub struct SceneLayers {
17    /// Water polylines (the Seine).
18    pub water: bool,
19    /// City-block outlines: the free areas enclosed by streets,
20    /// river and border.
21    pub blocks: bool,
22    /// Backdrop landmark pins.
23    pub landmarks: bool,
24    /// Per-transport edge visibility.
25    pub transports: TransportLayers,
26    /// Transports drawn in their own colors; the rest render in the
27    /// neutral `edge` ink (standard view).
28    pub highlight: BTreeSet<Transport>,
29    /// Metro lines hidden by name (e.g. `"M4"`) - on top of the
30    /// `metro` transport flag.
31    pub hidden_lines: BTreeSet<LineName>,
32    /// Taxi/bus hops shared by several transports draw side by
33    /// side in each transport's color (metro-line style).
34    pub parallel_transports: bool,
35    /// Station discs.
36    pub stations: bool,
37    /// Station name labels.
38    pub labels: bool,
39    /// Station number labels inside the disc.
40    pub numbers: bool,
41    /// Strategic-importance score labels beside stations.
42    pub scores: bool,
43    /// Label orientation for tabletop play.
44    pub label_mode: LabelMode,
45    /// Landmark artwork orientation: upright pins (`Screen`) or
46    /// pins facing the nearest table edge (`Radial`); each pin
47    /// additionally dodges stations and edges.
48    pub landmark_mode: LabelMode,
49    /// District caption orientation (`Radial` default like the
50    /// station labels).
51    pub district_mode: LabelMode,
52    /// Park caption orientation (`Radial` default like the
53    /// station labels).
54    pub park_mode: LabelMode,
55    /// House rendering mode (`RadialIso` default: each block's
56    /// houses extrude toward its nearest table edge).
57    pub house_mode: HouseMode,
58    /// The compass rose.
59    pub compass: bool,
60    /// Bridge-station markers (dataset debugging, MCP-only layer).
61    pub bridges: bool,
62}
63
64impl SceneLayers {
65    /// The named layer's visibility flag.
66    pub fn get(&self, name: LayerName) -> bool {
67        match name {
68            LayerName::Taxi => self.transports.taxi,
69            LayerName::Bus => self.transports.bus,
70            LayerName::Metro => self.transports.metro,
71            LayerName::MetroOverlay => self.highlight.contains(&Transport::Metro),
72            LayerName::Underground => self.transports.underground,
73            LayerName::Batobus => self.transports.batobus,
74            LayerName::ParallelTransports => self.parallel_transports,
75            LayerName::Water => self.water,
76            LayerName::Blocks => self.blocks,
77            LayerName::Landmarks => self.landmarks,
78            LayerName::Stations => self.stations,
79            LayerName::Labels => self.labels,
80            LayerName::Numbers => self.numbers,
81            LayerName::Scores => self.scores,
82            LayerName::Compass => self.compass,
83            LayerName::Bridges => self.bridges,
84        }
85    }
86
87    /// Sets the named layer's visibility flag.
88    pub fn set(&mut self, name: LayerName, on: bool) {
89        if name == LayerName::MetroOverlay {
90            if on {
91                self.highlight.insert(Transport::Metro);
92            } else {
93                self.highlight.remove(&Transport::Metro);
94            }
95            return;
96        }
97        if let Some(flag) = self.flag_mut(name) {
98            *flag = on;
99        }
100    }
101
102    /// The bool flag a layer name maps to - `None` for layers
103    /// that are not a plain flag (the metro overlay highlight).
104    fn flag_mut(&mut self, name: LayerName) -> Option<&mut bool> {
105        let flag = match name {
106            LayerName::Taxi => &mut self.transports.taxi,
107            LayerName::Bus => &mut self.transports.bus,
108            LayerName::Metro => &mut self.transports.metro,
109            LayerName::Underground => &mut self.transports.underground,
110            LayerName::Batobus => &mut self.transports.batobus,
111            LayerName::ParallelTransports => &mut self.parallel_transports,
112            LayerName::Water => &mut self.water,
113            LayerName::Blocks => &mut self.blocks,
114            LayerName::Landmarks => &mut self.landmarks,
115            LayerName::Stations => &mut self.stations,
116            LayerName::Labels => &mut self.labels,
117            LayerName::Numbers => &mut self.numbers,
118            LayerName::Scores => &mut self.scores,
119            LayerName::Compass => &mut self.compass,
120            LayerName::Bridges => &mut self.bridges,
121            LayerName::MetroOverlay => return None,
122        };
123        Some(flag)
124    }
125}
