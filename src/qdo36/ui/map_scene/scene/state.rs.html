30/// Everything the scene draws - the map, anchor, surface and view
31/// state. Passed in whole via [`crate::ui::map_scene::MapScene::update`].
32#[derive(Clone)]
33pub struct SceneState {
34    /// The loaded city map.
35    pub map: Rc<GameMap>,
36    /// Anchor calibration (anchor station + heading offset).
37    pub config: AnchorConfig,
38    /// Current anchor state (tracked or virtual pose).
39    pub anchor: AnchorState,
40    /// Physical surface size in mm (None: dev fallback px == mm).
41    pub surface: Option<SurfaceMm>,
42    /// View zoom.
43    pub scale: MapScale,
44    /// Highlighted station (tap selection).
45    pub selected: Option<StationId>,
46    /// Display rotation applied to surface-mm positions before the
47    /// px scale (rotated desktop panels).
48    pub display_rotation: DisplayRotation,
49    /// The map look (comic / neon / belle epoque).
50    pub style: MapStyle,
51    /// The generated SVG artwork (landmark pins, anchor, compass).
52    pub assets: Rc<ArtAssets>,
53    /// Overlay hint shown while the anchor is virtual (tower lifted
54    /// / not yet placed).
55    pub anchor_hint: Option<String>,
56    /// Which scene sections render (MCP `set_layers`).
57    pub layers: SceneLayers,
58    /// View pan offset in surface-mm (MCP `set_view`).
59    pub pan: Vec2<Millimeters>,
60    /// Compass rose position in map-mm (MCP `set_view compass_mm`).
61    pub compass_pos: Vec2<Millimeters>,
62    /// Global render factors (MCP `set_params` / UI controls).
63    pub params: RenderParams,
64    /// The map's city-block division - computed once on first use;
65    /// the map is immutable so the layout never goes stale.
66    pub block_layout: OnceCell<Rc<BlockLayout>>,
67    /// The rendered block geometry (inset outlines + houses +
68    /// ground masks) for the current view mode - recomputed when
69    /// the mode key changes, since the overlays widen the street
70    /// corridors.
71    pub block_shapes: RefCell<Option<(u8, Rc<BlocksGeometry>)>>,
72    /// The bridge-station decks - computed once on first use; the
73    /// map is immutable so they never go stale.
74    pub decks: OnceCell<HashMap<StationId, BridgeDeck>>,
75    /// The smoothed water courses - computed once on first use;
76    /// the map is immutable so they never go stale.
77    pub water_courses: OnceCell<Vec<Polyline>>,
78    /// The water courses' segments for bridge crossing tests -
79    /// computed once on first use.
80    pub water_segments: OnceCell<Vec<Segment<Millimeters>>>,
81}
82
83impl SceneState {
84    /// The map's city-block division (lazily computed).
85    pub fn block_layout(&self) -> &Rc<BlockLayout> {
86        self.block_layout.get_or_init(|| Rc::new(self.map.block_layout(WATER_HALF_WIDTH)))
87    }
88
89    /// Bridge-station decks: one per `bridge` station, perpendicular
90    /// to the smoothed course, station marker mid-deck. Empty while
91    /// the water layer is off - an invisible deck must not reroute
92    /// the roads either (lazily computed).
93    pub fn decks(&self) -> &HashMap<StationId, BridgeDeck> {
94        /// No decks while the water layer is off.
95        static EMPTY: LazyLock<HashMap<StationId, BridgeDeck>> = LazyLock::new(HashMap::new);
96        if self.layers.water {
97            self.decks.get_or_init(|| self.map.bridge_decks(WATER_HALF_WIDTH))
98        } else {
99            &EMPTY
100        }
101    }
102
103    /// The Chaikin-smoothed water courses - the same courses feed
104    /// the river bands, the Batobus routes and the bridge decks
105    /// (lazily computed).
106    pub fn water_courses(&self) -> &[Polyline] {
107        self.water_courses.get_or_init(|| self.map.smoothed_water())
108    }
109
110    /// The water courses' segments - the bridge crossing tests
111    /// run on these (lazily computed).
112    pub fn water_segments(&self) -> &[Segment<Millimeters>] {
113        self.water_segments
114            .get_or_init(|| self.water_courses().iter().flat_map(|course| course.segments()).collect())
115    }
116}
