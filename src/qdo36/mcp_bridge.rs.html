1//! MCP bridge: publishes the app state snapshot the `qdo36-mcp`

4//!
5//! The GTK side *writes* a typed document into a `watch` channel and
6//! *executes* `GameRequest`s on the main loop; the MCP thread reads
7//! the document and forwards commands - no GTK access from the

67    /// and its subscribers; installed by [`attach`].

98/// Registers one edge's overlay menu so `set_map_style` can
99/// restyle its panels.
100pub(crate) fn register_menu(panels: &[gtk4::Widget]) {
101    MENU.with(|slot| {
102        slot.borrow_mut().push(MenuParts { panels: panels.to_vec() });
103    });
104}

106/// Swaps the menus' `style-*` family class so their panels follow
107/// the map look.
108fn menu_apply_style(style: MapStyle) {
109    use gtk4::prelude::WidgetExt;
110    MENU.with(|slot| {
111        for menu in slot.borrow().iter() {
112            for panel in &menu.panels {
113                for class in [MapStyle::Comic, MapStyle::BelleEpoque, MapStyle::Neon].map(|style| style.controls_class()) {
114                    panel.remove_css_class(class);
115                }
116                panel.add_css_class(style.controls_class());
117            }
118        }
119    });
120}

170/// Starts the MCP Streamable-HTTP endpoint on `addr`, installs the
171/// state channel and drains `GameRequest`s on the GTK main loop.
172/// No-op friendly: the app is unchanged when `--mcp-listen` is not
173/// passed.
174pub fn attach(addr: SocketAddr) {
175    let (watch_tx, watch_rx) = channel(StateSnapshot::default());
176    let (command_tx, command_rx) = unbounded::<GameRequest>();
177    MCP_STATE_WATCH.with(|slot| *slot.borrow_mut() = Some(watch_tx));
178    GAME_HOST.with(|slot| *slot.borrow_mut() = Some(GameHost::with_rng(StdRng::from_rng(&mut rand::rng()))));
179    spawn_http(command_tx, watch_rx, addr);
180    info!(%addr, "MCP endpoint attached");
181    MainContext::default().spawn_local(async move {
182        while let Ok(request) = command_rx.recv().await {
183            // View/scene commands are app concerns, not session ones -
184            // the game host never sees them.
185            let reply = match &request.command {
186                CommandKind::Screenshot => mcp_screenshot(),
187                CommandKind::SceneLayout => scene_layout(),
188                CommandKind::SetView { zoom, rotate, pan, compass } => set_view(*zoom, *rotate, *pan, *compass),
189                CommandKind::Goto { stations, zoom } => goto(stations, *zoom),
190                CommandKind::SetLayers { mask } => set_layers(mask),
191                CommandKind::SetStyle { style } => set_style(*style),
192                CommandKind::SetAnchor { center, rotation_rad } => set_anchor(*center, *rotation_rad),
193                CommandKind::ReloadMap => reload_map(),
194                CommandKind::SetParams {
195                    station_scale,
196                    landmark_scale,
197                    metro_scale,
198                    ring_width,
199                    label_scale,
200                } => set_params(*station_scale, *landmark_scale, *metro_scale, *ring_width, *label_scale),
201                _ => with_game_host(|host| host.handle(&request.command)).unwrap_or(Err(McpError::ChannelClosed)),
202            };
203            // Publish after every command - mutations become visible
204            // to move_log/game_config/game_state subscribers.
205            with_game_host(|host| publish_game(host.session_cloned()));
206            let _ = request.reply.send(reply);
207        }
208    });
209}

217/// Renders the registered window into a PNG texture, cropped to
218/// `viewport` when given; `None` captures the full window. `None`
219/// when no window or surface is available.
220fn window_texture(viewport: Option<gtk4::graphene::Rect>) -> Option<gdk4::Texture> {
221    use gtk4::prelude::*;
222
223    let Some(window) = APP_WINDOW.with(|slot| slot.borrow().clone()) else {
224        warn!("screenshot: no app window registered");
225        return None;
226    };
227    let Some(surface) = window.surface() else {
228        warn!("screenshot: window has no surface");
229        return None;
230    };
231    let snapshot = gtk4::Snapshot::new();
232    gtk4::WidgetPaintable::new(Some(&window)).snapshot(&snapshot, f64::from(window.width()), f64::from(window.height()));
233    let Some(node) = snapshot.to_node() else {
234        warn!(
235            visible = window.is_visible(),
236            realized = window.is_realized(),
237            mapped = window.is_mapped(),
238            width = window.width(),
239            height = window.height(),
240            "screenshot: snapshot produced no render node"
241        );
242        return None;
243    };
244    let Some(renderer) = gtk4::gsk::Renderer::for_surface(&surface) else {
245        warn!("screenshot: no renderer for surface");
246        return None;
247    };
248    let viewport = viewport.unwrap_or_else(|| gtk4::graphene::Rect::new(0.0, 0.0, window.width() as f32, window.height() as f32));
249    let texture = renderer.render_texture(&node, Some(&viewport));
250    // A renderer may only be disposed while unrealized - dropping it
251    // while realized on the surface asserts in GSK.
252    renderer.unrealize();
253    Some(texture)
254}

274/// Schedules a screenshot of the window (or the `viewport` sub-rect
275/// for the area shot) into the user's screenshot directory. The
276/// widget paintable renders the *cached* render node, which is NULL
277/// between an invalidation and the following paint - so the capture
278/// runs on `after-paint` of the next frame (forced via
279/// `queue_draw`).
280pub(crate) fn save_shot(viewport: Option<gtk4::graphene::Rect>) {
281    use gtk4::prelude::*;
282
283    let Some(window) = APP_WINDOW.with(|slot| slot.borrow().clone()) else {
284        warn!("screenshot: no app window registered");
285        return;
286    };
287    let Some(surface) = window.surface() else {
288        warn!("screenshot: window has no surface");
289        return;
290    };
291    let handler_id = Rc::new(RefCell::new(None));
292    let id_slot = handler_id.clone();
293    let id = surface.frame_clock().connect_after_paint(move |clock| {
294        if let Some(id) = id_slot.borrow_mut().take() {
295            glib::object::ObjectExt::disconnect(clock, id);
296        }
297        write_shot(viewport);
298    });
299    handler_id.replace(Some(id));
300    window.queue_draw();
301}

303/// Writes the window PNG into the screenshot dir; returns the path.
304fn write_shot(viewport: Option<gtk4::graphene::Rect>) -> Option<PathBuf> {
305    use gdk4::prelude::TextureExt;
306
307    let Some(texture) = window_texture(viewport) else {
308        warn!("screenshot: render failed");
309        return None;
310    };
311    let dir = glib::user_special_dir(glib::UserDirectory::Pictures)
312        .unwrap_or_else(|| glib::home_dir().join("Pictures"))
313        .join("Screenshots");
314    if let Err(err) = std::fs::create_dir_all(&dir) {
315        warn!(%err, dir = %dir.display(), "screenshot: create dir failed");
316        return None;
317    }
318    let Some(now) = glib::DateTime::now_local().ok() else {
319        warn!("screenshot: local time unavailable");
320        return None;
321    };
322    let Some(stamp) = now.format("%Y-%m-%d_%H-%M-%S").ok() else {
323        warn!("screenshot: timestamp format failed");
324        return None;
325    };
326    let path = dir.join(format!("qdo36-{stamp}-{:03}.png", now.microsecond() / 1000));
327    match std::fs::write(&path, texture.save_to_png_bytes().as_ref()) {
328        Ok(()) => {
329            info!(path = %path.display(), "screenshot saved");
330            Some(path)
331        }
332        Err(err) => {
333            warn!(%err, path = %path.display(), "screenshot: write failed");
334            None
335        }
336    }
337}

339/// The `screenshot` command: PNG of the window, base64-encoded.
340fn mcp_screenshot() -> CommandReply {
341    use gdk4::prelude::TextureExt;
342
343    let Some(texture) = window_texture(None) else {
344        return Err(McpError::StateUnavailable);
345    };
346    let bytes = texture.save_to_png_bytes();
347    Ok(CommandResult::Screenshot(ScreenshotReply {
348        png_base64: glib::base64_encode(bytes.as_ref()).to_string(),
349        width: Pixels::new(f64::from(texture.width())),
350        height: Pixels::new(f64::from(texture.height())),
351    }))
352}

360/// The `set_anchor` command: virtual anchor pose on the surface +
361/// lift the map page over the setup page.
362fn set_anchor(center: Vec2<Millimeters>, rotation_rad: Option<RotationRad>) -> CommandReply {
363    let Some((state, scene, show_map)) = SCENE.with(|slot| slot.borrow().clone()) else {
364        return Err(McpError::ChannelClosed);
365    };
366    {
367        let mut state = state.borrow_mut();
368        state.anchor = AnchorState::Virtual(AnchorPose {
369            center,
370            rotation_rad: rotation_rad.unwrap_or(RotationRad::new(0.0)),
371        });
372        scene.update(Rc::new(state.clone()));
373    }
374    show_map();
375    Ok(CommandResult::AnchorSet)
376}

410/// The `reload_map` command: re-reads the city dataset from disk
411/// (dev checkout) or the bundled copy, swaps the scene map and
412/// hands it to the game host.
413fn reload_map() -> CommandReply {
414    with_scene(|state, scene| {
415        let map = {
416            let state = state.borrow();
417            let city: City = state
418                .map
419                .meta()
420                .name
421                .as_str()
422                .parse()
423                .map_err(|e| McpError::InvalidParams(format!("unknown city: {e}")))?;
424            let text = city
425                .dataset_path()
426                .and_then(|path| std::fs::read_to_string(&path).ok())
427                .unwrap_or_else(|| city.dataset().to_string());
428            Rc::new(
429                text.parse::<GameMap>()
430                    .map_err(|e| McpError::InvalidParams(format!("dataset reload failed: {e}")))?,
431            )
432        };
433        set_game_map(map.clone());
434        publish_map(MapInfo::from_map(&map, Some(state.borrow().config.anchor_map_pos)));
435        state.borrow_mut().map = map;
436        scene.update(Rc::new(state.borrow().clone()));
437        Ok(CommandResult::MapReloaded)
438    })
439    .unwrap_or(Err(McpError::StateUnavailable))
440}

442/// The `scene_layout` command: every station and landmark with
443/// map-mm and widget-px positions under the current transform.
444fn scene_layout() -> CommandReply {
445    use gtk4::prelude::WidgetExt;
446
447    let Some((entities,)) = with_scene(|state, scene| {
448        let state = state.borrow();
449        // The scene fills the window; fall back to its size when the
450        // widget reports no allocation yet.
451        let (width, height) = match (scene.width(), scene.height()) {
452            (w, h) if w > 0 && h > 0 => (w, h),
453            _ => APP_WINDOW
454                .with(|slot| slot.borrow().as_ref().map(|w| (w.width(), w.height())))
455                .unwrap_or((0, 0)),
456        };
457        let stage = RenderStage::new(&state, state.anchor.pose(), Pixels::new(width as f64), Pixels::new(height as f64));
458        let to_px = |pos| stage.to_px(pos);
459        let stations = state.map.stations().map(|s| SceneEntity {
460            kind: EntityKind::Station,
461            name: s.name.to_string(),
462            station: Some(s.id),
463            mm: s.pos,
464            px: to_px(s.pos),
465        });
466        let landmarks = state.map.backdrop().landmarks.iter().map(|l| SceneEntity {
467            kind: EntityKind::Landmark,
468            name: l.name.to_string(),
469            station: None,
470            mm: l.pos,
471            px: to_px(l.pos),
472        });
473        (stations.chain(landmarks).collect::<Vec<_>>(),)
474    }) else {
475        return Err(McpError::StateUnavailable);
476    };
477    Ok(CommandResult::Layout(qdo36_mcp::SceneLayout { entities }))
478}

504/// On-screen control helpers - the same scene mutation path the
505/// MCP commands use, callable from GTK callbacks (main thread).
506/// Multiplies the view zoom by `factor` (zoom buttons: 1.25 / 0.8).
507pub(crate) fn zoom_by(factor: f64) {
508    with_scene(|state, scene| {
509        {
510            let mut state = state.borrow_mut();
511            let zoom = (state.scale.factor() * factor).clamp(MapScale::MIN.factor(), MapScale::MAX.factor());
512            if let Ok(scale) = MapScale::try_new(zoom) {
513                state.scale = scale;
514            }
515        }
516        scene.update(Rc::new(state.borrow().clone()));
517    });
518}

520/// Resets zoom to `MapScale::ONE` and clears the pan.
521/// The `goto_*` commands: centers the view on the stations'
522/// midpoint. Two stations auto-zoom so the hop fills roughly 55%
523/// of the window's smaller side; one station keeps the zoom unless
524/// `zoom` overrides.
525fn goto(stations: &[StationId], zoom: Option<MapScale>) -> CommandReply {
526    use gtk4::prelude::WidgetExt;
527
528    with_scene(|state, scene| {
529        {
530            let mut state = state.borrow_mut();
531            let positions: Vec<Vec2<Millimeters>> = stations.iter().filter_map(|id| state.map.station(*id)).map(|s| s.pos).collect();
532            if positions.is_empty() {
533                return Err(McpError::InvalidParams(format!("unknown station(s) {stations:?}")));
534            }
535            let (width, height) = match (scene.width(), scene.height()) {
536                (w, h) if w > 0 && h > 0 => (w, h),
537                _ => APP_WINDOW
538                    .with(|slot| slot.borrow().as_ref().map(|w| (w.width(), w.height())))
539                    .unwrap_or((0, 0)),
540            };
541            let stage = RenderStage::new(&state, state.anchor.pose(), Pixels::new(width as f64), Pixels::new(height as f64));
542            let center = Vec2::new(
543                positions.iter().map(|pos| pos.x).sum::<f64>() / positions.len() as f64,
544                positions.iter().map(|pos| pos.y).sum::<f64>() / positions.len() as f64,
545            );
546            state.pan = stage.pan_for(center, PxPos::new(width as f64 / 2.0, height as f64 / 2.0));
547            // Auto-fit: widget px per map-mm is zoom * fit.scale, so
548            // the zoom landing the station distance on 55% of the
549            // smaller side is target_px / (distance * fit.scale).
550            let scale = zoom.or_else(|| {
551                let [a, b] = positions.as_slice() else { return None };
552                let distance = (b.x - a.x).hypot(b.y - a.y);
553                let fit = 0.55 * (width.min(height) as f64) / (distance * stage.fit.scale.as_f64());
554                MapScale::try_new(fit.clamp(MapScale::MIN.factor(), MapScale::MAX.factor())).ok()
555            });
556            if let Some(scale) = scale {
557                state.scale = scale;
558            }
559        }
560        scene.update(Rc::new(state.borrow().clone()));
561        Ok(CommandResult::ViewApplied)
562    })
563    .unwrap_or(Err(McpError::StateUnavailable))
564}

566pub(crate) fn reset_view() {
567    with_scene(|state, scene| {
568        {
569            let mut state = state.borrow_mut();
570            state.scale = MapScale::ONE;
571            state.pan = Vec2::ZERO;
572        }

620/// The `set_layers` command: layer-visibility toggles.
621fn set_layers(mask: &LayerMask) -> CommandReply {
622    with_scene(|state, scene| {
623        {
624            let mut state = state.borrow_mut();
625            let l = &mut state.layers;
626            if let Some(v) = mask.water {
627                l.water = v;
628            }
629            if let Some(v) = mask.blocks {
630                l.blocks = v;
631            }
632            if let Some(v) = mask.landmarks {
633                l.landmarks = v;
634            }
635            let t = &mut l.transports;
636            if let Some(v) = mask.taxi {
637                t.taxi = v;
638            }
639            if let Some(v) = mask.bus {
640                t.bus = v;
641            }
642            if let Some(v) = mask.metro {
643                t.metro = v;
644            }
645            if let Some(v) = mask.batobus {
646                t.batobus = v;
647            }
648            if let Some(v) = mask.underground {
649                t.underground = v;
650            }
651            if let Some(v) = mask.parallel_transports {
652                l.parallel_transports = v;
653            }
654            if let Some(lines) = &mask.hide_lines {
655                l.hidden_lines = lines.iter().map(|name| LineName::from(name.as_str())).collect();
656            }
657            if let Some(transports) = &mask.highlight {
658                l.highlight = transports
659                    .iter()
660                    .map(|name| {
661                        name.parse::<Transport>()
662                            .map_err(|_| McpError::InvalidParams(format!("unknown transport {name}")))
663                    })
664                    .collect::<Result<_, _>>()?;
665            }
666            if let Some(v) = mask.stations {
667                l.stations = v;
668            }
669            if let Some(v) = mask.labels {
670                l.labels = v;
671            }
672            if let Some(v) = mask.numbers {
673                l.numbers = v;
674            }
675            if let Some(v) = mask.scores {
676                l.scores = v;
677            }
678            if let Some(mode) = &mask.label_mode {
679                l.label_mode = mode.parse().map_err(|_| McpError::InvalidParams(format!("unknown label_mode {mode}")))?;
680            }
681            if let Some(mode) = &mask.landmark_mode {
682                l.landmark_mode = mode.parse().map_err(|_| McpError::InvalidParams(format!("unknown landmark_mode {mode}")))?;
683            }
684            if let Some(mode) = &mask.district_mode {
685                l.district_mode = mode.parse().map_err(|_| McpError::InvalidParams(format!("unknown district_mode {mode}")))?;
686            }
687            if let Some(mode) = &mask.park_mode {
688                l.park_mode = mode.parse().map_err(|_| McpError::InvalidParams(format!("unknown park_mode {mode}")))?;
689            }
690            if let Some(mode) = &mask.house_mode {
691                l.house_mode = mode.parse().map_err(|_| McpError::InvalidParams(format!("unknown house_mode {mode}")))?;
692            }
693            if let Some(v) = mask.compass {
694                l.compass = v;
695            }
696            if let Some(v) = mask.bridges {
697                l.bridges = v;
698            }
699        }
700        scene.update(Rc::new(state.borrow().clone()));
701        persist_view();
702        Ok(CommandResult::LayersApplied)
703    })
704    .unwrap_or(Err(McpError::StateUnavailable))
705}
