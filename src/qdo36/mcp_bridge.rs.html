1//! MCP bridge: publishes the app state snapshot the `qdo36-mcp`
2//! endpoint serves.
3//!
4//! The GTK side only *writes* a typed document into a `watch` channel;
5//! the MCP thread reads it independently - no GTK access from the
6//! server, ever (WP05 R5).
7
8use qdo36_mcp::AnchorStateSnapshot;
9use qdo36_mcp::MapInfo;
10use qdo36_mcp::StateSnapshot;
11use qdo36_mcp::spawn_http;
12use qdo36_tangibles::AnchorState;
13use qdo36_tangibles::SurfaceMm;
14use std::cell::RefCell;
15use std::net::SocketAddr;
16use tokio::sync::watch::Sender;
17use tokio::sync::watch::channel;
18use tracing::info;
19
20thread_local! {
21    /// Latest-state channel feeding the MCP `qdo36://state` resource
22    /// and its subscribers; installed by [`attach`].
23    static MCP_STATE_WATCH: RefCell<Option<Sender<StateSnapshot>>> = const { RefCell::new(None) };
24}
25
26/// Starts the MCP Streamable-HTTP endpoint on `addr` and installs the
27/// state channel. No-op friendly: the app is unchanged when
28/// `--mcp-listen` is not passed.
29pub fn attach(addr: SocketAddr) {
30    let (watch_tx, watch_rx) = channel(StateSnapshot::default());
31    MCP_STATE_WATCH.with(|slot| *slot.borrow_mut() = Some(watch_tx));
32    spawn_http(watch_rx, addr);
33    info!(%addr, "MCP endpoint attached");
34}
35
36/// Publishes the current anchor/surface snapshot for MCP consumers.
37/// Safe to call unconditionally - without [`attach`] the slot is
38/// `None` and this is a no-op.
39pub fn publish_state(anchor: &AnchorState, surface: Option<&SurfaceMm>) {
40    MCP_STATE_WATCH.with(|slot| {
41        let Some(tx) = slot.borrow().as_ref().cloned() else { return };
42        let snapshot = AnchorStateSnapshot::captured(anchor, surface);
43        tx.send_modify(|doc| doc.anchor = snapshot);
44    });
45}
46
47/// Publishes the loaded map's static facts (`map_info` tool).
48pub fn publish_map(info: MapInfo) {
49    MCP_STATE_WATCH.with(|slot| {
50        let Some(tx) = slot.borrow().as_ref().cloned() else { return };
51        tx.send_modify(|doc| doc.map = Some(info));
52    });
53}
