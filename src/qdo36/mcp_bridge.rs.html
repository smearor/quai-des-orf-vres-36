1//! MCP bridge: publishes the app state snapshot the `qdo36-mcp`
2//! endpoint serves, and drains the command channel into the
3//! app-owned [`GameHost`].
4//!
5//! The GTK side *writes* a typed document into a `watch` channel and
6//! *executes* `GameRequest`s on the main loop; the MCP thread reads
7//! the document and forwards commands - no GTK access from the
8//! server, ever (WP05 R5).

49thread_local! {
50    /// Latest-state channel feeding the MCP `qdo36://state` resource
51    /// and its subscribers; installed by [`attach`].
52    static MCP_STATE_WATCH: RefCell<Option<Sender<StateSnapshot>>> = const { RefCell::new(None) };
53    /// The app-owned game session answering MCP commands; installed
54    /// by [`attach`] (no host without `--mcp-listen`).
55    static GAME_HOST: RefCell<Option<GameHost>> = const { RefCell::new(None) };
56    /// The app window the `screenshot` command renders; installed by
57    /// [`register_window`].
58    static APP_WINDOW: RefCell<Option<adw::ApplicationWindow>> = const { RefCell::new(None) };
59    /// The live scene (state + widget + show-map callback) the view
60    /// commands mutate and `scene_layout` enumerates; installed by
61    /// [`register_scene`].
62    static SCENE: RefCell<Option<SceneHandle>> = const { RefCell::new(None) };
63}
64
65/// The registered scene: shared state, the scene widget and a
66/// callback lifting the map page over the setup page.
67type SceneHandle = (Rc<RefCell<SceneState>>, MapScene, Rc<dyn Fn()>);
68
69/// Runs `f` against the registered scene pair.
70fn with_scene<T>(f: impl FnOnce(&Rc<RefCell<SceneState>>, &MapScene) -> T) -> Option<T> {
71    SCENE.with(|slot| slot.borrow().as_ref().map(|(state, scene, _)| f(state, scene)))
72}
73
74/// Runs `f` against the game host; the host exists only while the MCP
75/// endpoint is attached.
76fn with_game_host<T>(f: impl FnOnce(&mut GameHost) -> T) -> Option<T> {
77    GAME_HOST.with(|slot| slot.borrow_mut().as_mut().map(f))
78}

80/// Starts the MCP Streamable-HTTP endpoint on `addr`, installs the
81/// state channel and drains `GameRequest`s on the GTK main loop.
82/// No-op friendly: the app is unchanged when `--mcp-listen` is not
83/// passed.
84pub fn attach(addr: SocketAddr) {
85    let (watch_tx, watch_rx) = channel(StateSnapshot::default());
86    let (command_tx, command_rx) = unbounded::<GameRequest>();
87    MCP_STATE_WATCH.with(|slot| *slot.borrow_mut() = Some(watch_tx));
88    GAME_HOST.with(|slot| *slot.borrow_mut() = Some(GameHost::with_rng(StdRng::from_rng(&mut rand::rng()))));
89    spawn_http(command_tx, watch_rx, addr);
90    info!(%addr, "MCP endpoint attached");
91    MainContext::default().spawn_local(async move {
92        while let Ok(request) = command_rx.recv().await {
93            // View/scene commands are app concerns, not session ones -
94            // the game host never sees them.
95            let reply = match &request.command {
96                CommandKind::Screenshot => mcp_screenshot(),
97                CommandKind::SceneLayout => scene_layout(),
98                CommandKind::SetView { zoom, rotate, pan, compass } => set_view(*zoom, *rotate, *pan, *compass),
99                CommandKind::SetLayers { mask } => set_layers(mask),
100                CommandKind::SetAnchor { center, rotation_rad } => set_anchor(*center, *rotation_rad),
101                CommandKind::ReloadMap => reload_map(),
102                CommandKind::SetParams {
103                    station_scale,
104                    landmark_scale,
105                    metro_scale,
106                    ring_width,
107                    label_scale,
108                } => set_params(*station_scale, *landmark_scale, *metro_scale, *ring_width, *label_scale),
109                _ => with_game_host(|host| host.handle(&request.command)).unwrap_or(Err(McpError::ChannelClosed)),
110            };
111            // Publish after every command - mutations become visible
112            // to move_log/game_config/game_state subscribers.
113            with_game_host(|host| publish_game(host.session_cloned()));
114            let _ = request.reply.send(reply);
115        }
116    });
117}

125/// Renders the registered window into a PNG texture; `None` when no
126/// window or surface is available.
127fn screenshot_texture() -> Option<gdk4::Texture> {
128    use gtk4::prelude::*;
129
130    let window = APP_WINDOW.with(|slot| slot.borrow().clone())?;
131    let surface = window.surface()?;
132    let snapshot = gtk4::Snapshot::new();
133    gtk4::WidgetPaintable::new(Some(&window)).snapshot(&snapshot, f64::from(window.width()), f64::from(window.height()));
134    let node = snapshot.to_node()?;
135    let renderer = gtk4::gsk::Renderer::for_surface(&surface)?;
136    let viewport = gtk4::graphene::Rect::new(0.0, 0.0, window.width() as f32, window.height() as f32);
137    let texture = renderer.render_texture(&node, Some(&viewport));
138    // A renderer may only be disposed while unrealized - dropping it
139    // while realized on the surface asserts in GSK.
140    renderer.unrealize();
141    Some(texture)
142}

144/// The `screenshot` command: PNG of the window, base64-encoded.
145fn mcp_screenshot() -> CommandReply {
146    use gdk4::prelude::TextureExt;
147
148    let Some(texture) = screenshot_texture() else {
149        return Err(McpError::StateUnavailable);
150    };
151    let bytes = texture.save_to_png_bytes();
152    Ok(CommandResult::Screenshot(ScreenshotReply {
153        png_base64: glib::base64_encode(bytes.as_ref()).to_string(),
154        width: Pixels::new(f64::from(texture.width())),
155        height: Pixels::new(f64::from(texture.height())),
156    }))
157}

159/// Registers the live scene for `scene_layout`/`set_view`/
160/// `set_layers`. Called once when the map scene is built.
161pub fn register_scene(state: &Rc<RefCell<SceneState>>, scene: &MapScene, show_map: Rc<dyn Fn()>) {
162    SCENE.with(|slot| *slot.borrow_mut() = Some((state.clone(), scene.clone(), show_map)));
163}

165/// The `set_anchor` command: virtual anchor pose on the surface +
166/// lift the map page over the setup page.
167fn set_anchor(center: Vec2<Millimeters>, rotation_rad: Option<RotationRad>) -> CommandReply {
168    use qdo36_tangibles::AnchorPose;
169    use qdo36_tangibles::AnchorState;
170
171    let Some((state, scene, show_map)) = SCENE.with(|slot| slot.borrow().clone()) else {
172        return Err(McpError::ChannelClosed);
173    };
174    {
175        let mut state = state.borrow_mut();
176        state.anchor = AnchorState::Virtual(AnchorPose {
177            center,
178            rotation_rad: rotation_rad.unwrap_or(RotationRad::new(0.0)),
179        });
180        scene.update(Rc::new(state.clone()));
181    }
182    show_map();
183    Ok(CommandResult::AnchorSet)
184}

186/// The `set_params` command: global render factors on the live scene.
187fn set_params(
188    station_scale: Option<ScaleFactor>,
189    landmark_scale: Option<ScaleFactor>,
190    metro_scale: Option<ScaleFactor>,
191    ring_width: Option<ScaleFactor>,
192    label_scale: Option<ScaleFactor>,
193) -> CommandReply {
194    with_scene(|state, scene| {
195        let mut state = state.borrow_mut();
196        let p = &mut state.params;
197        if let Some(v) = station_scale {
198            p.station_scale = v;
199        }
200        if let Some(v) = landmark_scale {
201            p.landmark_scale = v;
202        }
203        if let Some(v) = metro_scale {
204            p.metro_scale = v;
205        }
206        if let Some(v) = ring_width {
207            p.ring_width = v;
208        }
209        if let Some(v) = label_scale {
210            p.label_scale = v;
211        }
212        scene.update(Rc::new(state.clone()));
213        Ok(CommandResult::ParamsApplied)
214    })
215    .unwrap_or(Err(McpError::StateUnavailable))
216}

218/// The `reload_map` command: re-reads the city dataset from disk
219/// (dev checkout) or the bundled copy, swaps the scene map and
220/// hands it to the game host.
221fn reload_map() -> CommandReply {
222    use qdo36_map::City;
223
224    with_scene(|state, scene| {
225        let map = {
226            let state = state.borrow();
227            let city: City = state
228                .map
229                .meta()
230                .name
231                .as_str()
232                .parse()
233                .map_err(|e| McpError::InvalidParams(format!("unknown city: {e}")))?;
234            let text = city
235                .dataset_path()
236                .and_then(|path| std::fs::read_to_string(&path).ok())
237                .unwrap_or_else(|| city.dataset().to_string());
238            Rc::new(
239                text.parse::<GameMap>()
240                    .map_err(|e| McpError::InvalidParams(format!("dataset reload failed: {e}")))?,
241            )
242        };
243        set_game_map(map.clone());
244        publish_map(MapInfo::from_map(&map, Some(state.borrow().config.anchor_map_pos)));
245        state.borrow_mut().map = map;
246        scene.update(Rc::new(state.borrow().clone()));
247        Ok(CommandResult::MapReloaded)
248    })
249    .unwrap_or(Err(McpError::StateUnavailable))
250}

252/// The `scene_layout` command: every station and landmark with
253/// map-mm and widget-px positions under the current transform.
254fn scene_layout() -> CommandReply {
255    use gtk4::prelude::WidgetExt;
256    use qdo36_mcp::EntityKind;
257    use qdo36_mcp::SceneEntity;
258
259    let Some((entities,)) = with_scene(|state, scene| {
260        let state = state.borrow();
261        // The scene fills the window; fall back to its size when the
262        // widget reports no allocation yet.
263        let (width, height) = match (scene.width(), scene.height()) {
264            (w, h) if w > 0 && h > 0 => (w, h),
265            _ => APP_WINDOW
266                .with(|slot| slot.borrow().as_ref().map(|w| (w.width(), w.height())))
267                .unwrap_or((0, 0)),
268        };
269        let stage = RenderStage::new(&state, state.anchor.pose(), Pixels::new(width as f64), Pixels::new(height as f64));
270        let to_px = |pos| stage.to_px(pos);
271        let stations = state.map.stations().map(|s| SceneEntity {
272            kind: EntityKind::Station,
273            name: s.name.to_string(),
274            station: Some(s.id),
275            mm: s.pos,
276            px: to_px(s.pos),
277        });
278        let landmarks = state.map.backdrop().landmarks.iter().map(|l| SceneEntity {
279            kind: EntityKind::Landmark,
280            name: l.name.to_string(),
281            station: None,
282            mm: l.pos,
283            px: to_px(l.pos),
284        });
285        (stations.chain(landmarks).collect::<Vec<_>>(),)
286    }) else {
287        return Err(McpError::StateUnavailable);
288    };
289    Ok(CommandResult::Layout(qdo36_mcp::SceneLayout { entities }))
290}

292/// The `set_view` command: pan/rotate/zoom/compass on the live scene.
293fn set_view(zoom: Option<MapScale>, rotate: Option<DisplayRotation>, pan: Option<Vec2<Millimeters>>, compass: Option<Vec2<Millimeters>>) -> CommandReply {
294    with_scene(|state, scene| {
295        {
296            let mut state = state.borrow_mut();
297            if let Some(scale) = zoom {
298                state.scale = scale;
299            }
300            if let Some(rotation) = rotate {
301                state.display_rotation = rotation;
302            }
303            if let Some(offset) = pan {
304                state.pan = offset;
305            }
306            if let Some(pos) = compass {
307                state.compass_pos = pos;
308            }
309        }
310        scene.update(Rc::new(state.borrow().clone()));
311        Ok(CommandResult::ViewApplied)
312    })
313    .unwrap_or(Err(McpError::StateUnavailable))
314}

316/// On-screen control helpers - the same scene mutation path the
317/// MCP commands use, callable from GTK callbacks (main thread).
318/// Multiplies the view zoom by `factor` (zoom buttons: 1.25 / 0.8).
319pub(crate) fn zoom_by(factor: f64) {
320    with_scene(|state, scene| {
321        {
322            let mut state = state.borrow_mut();
323            let zoom = (state.scale.factor() * factor).clamp(MapScale::MIN.factor(), MapScale::MAX.factor());
324            if let Ok(scale) = MapScale::try_new(zoom) {
325                state.scale = scale;
326            }
327        }
328        scene.update(Rc::new(state.borrow().clone()));
329    });
330}

344/// Toggles the label orientation (radial <-> screen).
345pub(crate) fn toggle_label_mode() {
346    with_scene(|state, scene| {
347        {
348            let mut state = state.borrow_mut();
349            state.layers.label_mode = match state.layers.label_mode {
350                crate::ui::map_scene::LabelMode::Radial => crate::ui::map_scene::LabelMode::Screen,
351                crate::ui::map_scene::LabelMode::Screen => crate::ui::map_scene::LabelMode::Radial,
352            };
353        }
354        scene.update(Rc::new(state.borrow().clone()));
355    });
356}

374/// The `set_layers` command: layer-visibility toggles.
375fn set_layers(mask: &LayerMask) -> CommandReply {
376    with_scene(|state, scene| {
377        {
378            let mut state = state.borrow_mut();
379            let l = &mut state.layers;
380            if let Some(v) = mask.water {
381                l.water = v;
382            }
383            if let Some(v) = mask.landmarks {
384                l.landmarks = v;
385            }
386            let t = &mut l.transports;
387            if let Some(v) = mask.taxi {
388                t.taxi = v;
389            }
390            if let Some(v) = mask.bus {
391                t.bus = v;
392            }
393            if let Some(v) = mask.metro {
394                t.metro = v;
395            }
396            if let Some(v) = mask.batobus {
397                t.batobus = v;
398            }
399            if let Some(v) = mask.underground {
400                t.underground = v;
401            }
402            if let Some(lines) = &mask.hide_lines {
403                l.hidden_lines = lines.iter().map(|name| qdo36_map::LineName::from(name.as_str())).collect();
404            }
405            if let Some(transports) = &mask.highlight {
406                l.highlight = transports
407                    .iter()
408                    .map(|name| {
409                        name.parse::<qdo36_map::Transport>()
410                            .map_err(|_| McpError::InvalidParams(format!("unknown transport {name}")))
411                    })
412                    .collect::<Result<_, _>>()?;
413            }
414            if let Some(v) = mask.stations {
415                l.stations = v;
416            }
417            if let Some(v) = mask.labels {
418                l.labels = v;
419            }
420            if let Some(v) = mask.numbers {
421                l.numbers = v;
422            }
423            if let Some(v) = mask.scores {
424                l.scores = v;
425            }
426            if let Some(mode) = &mask.label_mode {
427                l.label_mode = match mode.as_str() {
428                    "radial" => crate::ui::map_scene::LabelMode::Radial,
429                    "screen" => crate::ui::map_scene::LabelMode::Screen,
430                    other => return Err(McpError::InvalidParams(format!("unknown label_mode {other}"))),
431                };
432            }
433            if let Some(v) = mask.compass {
434                l.compass = v;
435            }
436            if let Some(v) = mask.bridges {
437                l.bridges = v;
438            }
439        }
440        scene.update(Rc::new(state.borrow().clone()));
441        Ok(CommandResult::LayersApplied)
442    })
443    .unwrap_or(Err(McpError::StateUnavailable))
444}

446/// Hands the loaded map to the game host so `request_move` can
447/// validate against it. Called once at scene load; the `Rc` is
448/// cloned into an `Arc` so the host stays `Send`.
449pub fn set_game_map(map: Rc<GameMap>) {
450    with_game_host(|host| host.set_map(Arc::new(map.as_ref().clone())));

453/// Publishes the current session snapshot (`move_log`, `game_config`,
454/// resource subscribers). Safe to call unconditionally.
455pub fn publish_game(session: Option<GameSession>) {
456    MCP_STATE_WATCH.with(|slot| {
457        let Some(tx) = slot.borrow().as_ref().cloned() else { return };
458        tx.send_modify(|doc| doc.game = session);
459    });
460}

462/// Publishes the current anchor/surface snapshot for MCP consumers.
463/// Safe to call unconditionally - without [`attach`] the slot is
464/// `None` and this is a no-op.
465pub fn publish_state(anchor: &AnchorState, surface: Option<&SurfaceMm>) {
466    MCP_STATE_WATCH.with(|slot| {
467        let Some(tx) = slot.borrow().as_ref().cloned() else { return };
468        let snapshot = AnchorStateSnapshot::captured(anchor, surface);
469        tx.send_modify(|doc| doc.anchor = snapshot);
470    });
471}
