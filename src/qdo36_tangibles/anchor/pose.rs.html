9/// The Eiffel tower's measured pose on the surface.
10///
11/// `center` is the smoothed centroid in surface millimeters,
12/// `rotation_rad` is correspondence-anchored to the profile's pin
13/// frame (tangible-objects-rs pin-map fix), so it is stable and repeatable.
14#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
15pub struct AnchorPose {
16    /// Tangible centroid on the touch surface.
17    pub center: Vec2<Millimeters>,
18    /// Tangible orientation in radians (profile pin frame).
19    pub rotation_rad: RotationRad,
20}
21
22impl AnchorPose {
23    /// The neutral pose at the surface origin, facing north.
24    pub const ZERO: Self = Self {
25        center: Vec2::ZERO,
26        rotation_rad: RotationRad::ZERO,
27    };
28
29    /// Interpolates toward `to` by `factor` (`0.0` = `self`, `1.0` =
30    /// `to`): linear in position, shortest-arc in rotation.
31    pub fn lerp(self, to: AnchorPose, factor: f64) -> AnchorPose {
32        Self {
33            center: Vec2::new(self.center.x + factor * (to.center.x - self.center.x), self.center.y + factor * (to.center.y - self.center.y)),
34            rotation_rad: RotationRad::ema(self.rotation_rad, to.rotation_rad, SmoothingAlpha::new(factor)),
35        }
36    }
37}
38
39impl From<&TrackedTangible> for AnchorPose {
40    fn from(tangible: &TrackedTangible) -> Self {
41        Self {
42            center: tangible.center,
43            rotation_rad: tangible.rotation_rad,
44        }
45    }
