23/// Tangible source backed by the `tangible-objects` daemon.
24///
25/// Thin adapter over the upstream [`GraphQLClient`] (HTTP queries and
26/// mutations, `graphql-transport-ws` subscriptions): the typed stream
27/// of upstream `TangibleEvent`s is projected into our narrower
28/// [`TangibleEvent`] channel so the rest of the crate never sees
29/// non-tangible payload.
30///
31/// `connect` expects a Tokio runtime on the calling thread - the GTK
32/// app hosts a dedicated runtime thread for it.
33pub struct GraphQlSource {
34    /// The upstream typed client.
35    client: GraphQLClient,
36    /// Our registration owner for session-lifetime cleanup.
37    owner: Owner,
38}
39
40impl GraphQlSource {
41    /// Connects to the daemon at `endpoint` (e.g.
42    /// `http://127.0.0.1:8080` or the full `…/graphql` URL).
43    pub async fn connect(endpoint: &str) -> Result<Self, DaemonError> {
44        let url = if endpoint.ends_with("/graphql") {
45            endpoint.to_string()
46        } else {
47            format!("{}/graphql", endpoint.trim_end_matches('/'))
48        };
49        let client = GraphQLClient::new(Endpoint::new(url.clone())).map_err(|e| DaemonError::Connect { url, reason: e.to_string() })?;
50        Ok(Self {
51            client,
52            owner: Owner::new(REGISTRATION_OWNER),
53        })
54    }
55}

57impl TangibleSource for GraphQlSource {
58    async fn register_anchor_profile(&mut self, profile: &AnchorPattern) -> Result<PatternId, DaemonError> {
59        let result = self.client.register_pattern(profile.profile(), Some(&self.owner), Lifetime::Session).await?;
60        match result {
61            RegisterPatternResult::Registered(registered) => Ok(registered.pattern_id),
62            RegisterPatternResult::Rejected(rejected) => {
63                Err(DaemonError::Registration(format!("registerPattern rejected {}: {}", profile.profile().name, rejected.reason)))
64            }
65        }
66    }
67
68    async fn tangible_events(&self, pattern: PatternId) -> Receiver<TangibleEvent> {
69        let (tx, receiver) = async_channel::bounded(256);
70        match self.client.tangible_events(Some(pattern)).await {
71            Ok(stream) => {
72                let mut stream = Box::pin(stream);
73                tokio::spawn(async move {
74                    while let Some(item) = stream.next().await {
75                        match item {
76                            Ok(event) => {
77                                if let Some(projected) = TangibleEvent::project(&event)
78                                    && tx.try_send(projected).is_err()
79                                {
80                                    warn!("tangible event channel full/closed - dropping event");
81                                }
82                            }
83                            Err(err) => {
84                                warn!("tangible event stream error: {err} - asking caller to resync");
85                                let _ = tx.try_send(TangibleEvent::lagged(None));
86                            }
87                        }
88                    }
89                    debug!("tangible event subscription ended");
90                });
91            }
92            Err(err) => {
93                warn!("tangibleEvents subscription failed: {err} - channel carries Lagged only");
94                let _ = tx.try_send(TangibleEvent::lagged(None));
95            }
96        }
97        receiver
98    }

100    async fn tangibles(&self) -> Result<Vec<TrackedTangible>, DaemonError> {

108    async fn surface(&self) -> Result<SurfaceMm, DaemonError> {
109        let info = self
110            .client
111            .device_info()
112            .await?
113            .ok_or_else(|| DaemonError::Unavailable("daemon reports no device info".to_string()))?;
114        let mm = |res: Option<UnitsPerMm>, min: Option<RawAxisValue>, max: Option<RawAxisValue>| -> Option<Millimeters> {
115            let res = res?.as_f64();
116            let span = max?.as_f64() - min?.as_f64();
117            (res > 0.0).then(|| Millimeters::new(span / res))
118        };
119        let width = mm(info.x_res, info.x_min, info.x_max);
120        let height = mm(info.y_res, info.y_min, info.y_max);
121        width
122            .zip(height)
123            .map(SurfaceMm::from)
124            .ok_or_else(|| DaemonError::Unavailable("device info lacks resolution/extent".to_string()))
125    }
126
127    async fn unregister(&mut self) -> Result<(), DaemonError> {
128        self.client.unregister_owner(&self.owner).await?;
129        Ok(())
130    }
