3/// Errors talking to the `tangible-objects` daemon.
4#[derive(Debug, Error)]
5pub enum DaemonError {
6    /// The GraphQL endpoint is unreachable or refused the connection.
7    #[error("daemon unreachable at {url}: {reason}")]
8    Connect {
9        /// The configured daemon URL.
10        url: String,
11        /// The underlying error message.
12        reason: String,
13    },
14    /// A GraphQL query/subscription failed.
15    #[error("daemon query failed: {0}")]
16    Query(String),
17    /// The event subscription stream ended or errored.
18    #[error("subscription failed: {0}")]
19    Subscription(String),
20    /// A bundled figure profile is malformed.
21    #[error("invalid profile: {0}")]
22    Profile(String),
23    /// Pattern or field-set registration failed.
24    #[error("registration failed: {0}")]
25    Registration(String),
26    /// No answer within the expected time.
27    #[error("daemon timed out: {0}")]
28    Timeout(String),
29    /// The HTTP/WebSocket transport failed.
30    #[error("daemon transport failed: {0}")]
31    Transport(String),
32    /// The endpoint returned a GraphQL `errors` array.
33    #[error("GraphQL error: {0}")]
34    GraphQl(String),
35    /// A response could not be decoded into the expected type.
36    #[error("response decode failed: {0}")]
37    Decode(String),
38    /// The daemon cannot serve the request (missing device info,
39    /// pattern, ...).
40    #[error("daemon unavailable: {0}")]
41    Unavailable(String),
42}
43
44impl From<tangible_objects_client::ClientError> for DaemonError {
45    fn from(err: tangible_objects_client::ClientError) -> Self {
46        use tangible_objects_client::ClientError as C;
47        match err {
48            C::Http(_) | C::Ws(_) | C::NoData => Self::Transport(err.to_string()),
49            C::Subscription(_) | C::Protocol(_) | C::FieldNotFound(_) => Self::Subscription(err.to_string()),
50            C::Graphql(value) => Self::GraphQl(value.to_string()),
51            C::Decode(err) => Self::Decode(err.to_string()),
52        }
53    }
54}
