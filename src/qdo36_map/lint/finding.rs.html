1//! `Finding` - one map-quality violation (`GameMap::lint`).
2//!
3//! The dataset quality criteria (Q1-Q9, WP10): every finding kind
4//! maps to exactly one criterion.
5
6use crate::LineName;
7use crate::StationCount;
8use crate::StationDegree;
9use crate::StationId;
10use crate::Transport;
11use crate::lint::ComponentCount;
12use crate::lint::EdgePair;
13use crate::lint::LinearityCause;
14use crate::lint::MissingService;
15use serde::Deserialize;
16use serde::Serialize;
17use tangible_objects_model::Millimeters;
18use thiserror::Error;

20/// A dataset quality violation.
21#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, Error)]
22pub enum Finding {
23    /// Two edges of the same layer cross without a shared station
24    /// (Q1: each transport net is planar; metro lines may cross
25    /// each other, only same-line self-crossings report).
26    #[error("{transport} edges cross: {first:?} x {second:?}")]
27    Crossing {
28        /// The layer the crossing sits on.
29        transport: Transport,
30        /// First edge.
31        first: EdgePair,
32        /// Second edge.
33        second: EdgePair,
34    },
35    /// A transport's adjacency graph is split into islands (Q2).
36    #[error("{transport} net is disconnected ({components} components: {sizes:?} stations)")]
37    Disconnected {
38        /// The transport layer.
39        transport: Transport,
40        /// How many components the net has.
41        components: ComponentCount,
42        /// Stations per component, sorted descending.
43        sizes: Vec<StationCount>,
44    },
45    /// A metro line is not a simple path (Q3).
46    #[error("metro line {line} is not linear: {cause}")]
47    LineNotLinear {
48        /// The failing line.
49        line: LineName,
50        /// Why it fails.
51        cause: LinearityCause,
52    },
53    /// The batobus pier chain is not a simple path (Q7).
54    #[error("batobus route is not linear: {cause}")]
55    BatobusNotLinear {
56        /// Why it fails.
57        cause: LinearityCause,
58    },
59    /// A metro line shares no station with any other line - no
60    /// transfer possible (Q4).
61    #[error("metro line {0} shares no station with any other line")]
62    IsolatedLine(LineName),
63    /// A metro station lacks taxi and/or bus service (Q5).
64    #[error("metro station {station} lacks {missing:?} service")]
65    MetroStationUnserved {
66        /// The underserviced station.
67        station: StationId,
68        /// Which services are missing.
69        missing: MissingService,
70    },
71    /// Fewer than 90% of bus stations also have taxi service (Q6).
72    #[error("only {taxi_covered}/{bus_total} bus stations have taxi service (< 90%)")]
73    BusCoverageBelow {
74        /// Bus stations with taxi service.
75        taxi_covered: StationCount,
76        /// All bus stations.
77        bus_total: StationCount,
78    },
79    /// A station has no edges at all - unreachable from anywhere.
80    #[error("station {0} is orphaned (degree 0)")]
81    OrphanedStation(StationId),
82    /// A station has exactly one edge overall - a dead end that
83    /// traps whoever enters (Q9, informational).
84    #[error("station {0} is a leaf (degree 1)")]
85    LeafStation(StationId),
86    /// Two stations sit on the same position (Q9).
87    #[error("stations {0} and {1} share the same position")]
88    DuplicatePosition(StationId, StationId),
89    /// A station lies almost on an edge it does not belong to - the
90    /// route visually runs through the station without serving it,
91    /// so the direct edge is hard to tell from the detour via the
92    /// station (Q14, readability).
93    #[error("{transport} edge {edge:?} runs within {distance_mm:.1} of station {station}")]
94    StationOnEdge {
95        /// The transport layer the offending edge belongs to.
96        transport: Transport,
97        /// The edge passing the station.
98        edge: EdgePair,
99        /// The station sitting on the line.
100        station: StationId,
101        /// Perpendicular distance in map millimeters.
102        distance_mm: Millimeters,
103    },
104    /// A start-deal station has fewer than two edges - a dealt
105    /// detective would start trapped (Q10).
106    #[error("start station {station} is underconnected (degree {degree})")]
107    UnderconnectedStart {
108        /// The weak start station.
109        station: StationId,
110        /// Its total edge degree.
111        degree: StationDegree,
112    },
113    /// The density order is inverted: a sparser-class layer carries
114    /// more edges than a denser one, expected taxi >= bus >= metro
115    /// (Q11, soft).
116    #[error("density inversion: {sparser} has more edges than {denser}")]
117    DensityInversion {
118        /// The layer that should be sparser but is denser.
119        sparser: Transport,
120        /// The layer that should be denser.
121        denser: Transport,
122    },
123}
