1//! `EdgeSet` - a borrowed edge collection of one lint view.
2
3use crate::Edge;
4use crate::EdgeCount;
5use crate::StationCount;
6use crate::StationId;
7use crate::lint::Adjacency;
8use std::collections::HashSet;
9use std::ops::Deref;
10
11/// A set of edge references - per-transport, per-line or per
12/// finding group. `Deref` exposes it as a `&[&Edge]` slice for
13/// indexed iteration; the domain methods build graph views.
14#[derive(Default)]
15pub struct EdgeSet<'a> {
16    edges: Vec<&'a Edge>,
17}
18
19impl<'a> EdgeSet<'a> {
20    /// No edges in the set.
21    pub fn is_empty(&self) -> bool {
22        self.edges.is_empty()
23    }
24
25    /// Edges in the set.
26    pub fn len(&self) -> EdgeCount {
27        EdgeCount::of_len(self.edges.len())
28    }
29
30    /// Appends one edge reference (map-fold collection).
31    pub fn push(&mut self, edge: &'a Edge) {
32        self.edges.push(edge);
33    }
34
35    /// The undirected neighbor graph of the set.
36    pub fn adjacency(&self) -> Adjacency {
37        Adjacency::from_iter(self.edges.iter().copied())
38    }
39
40    /// The stations the set touches.
41    pub fn served(&self) -> HashSet<StationId> {
42        self.adjacency().served()
43    }
44
45    /// Stations any edge of the set touches, as a count.
46    pub fn served_count(&self) -> StationCount {
47        StationCount::of_len(self.served().len())
48    }
49}
50
51impl<'a> FromIterator<&'a Edge> for EdgeSet<'a> {
52    /// Collects borrowed edges into the set.
53    fn from_iter<T: IntoIterator<Item = &'a Edge>>(edges: T) -> Self {
54        Self {
55            edges: edges.into_iter().collect(),
56        }
57    }
58}
59
60impl<'a> Deref for EdgeSet<'a> {
61    type Target = [&'a Edge];
62
63    /// Borrowed edges as a slice (indexing + iteration).
64    fn deref(&self) -> &Self::Target {
65        &self.edges
66    }
67}
