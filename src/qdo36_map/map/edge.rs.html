8/// An undirected travel route between two stations.
9///
10/// Stored canonicalized with `from < to` - the game graph has no
11/// one-way routes.
12#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
13pub struct Edge {
14    /// Station at one end (the lower id after canonicalization).
15    pub from: StationId,
16    /// Station at the other end.
17    pub to: StationId,
18    /// Transport mode this route is used with.
19    pub transport: Transport,
20    /// Transit line serving this route (metro edges only - the line's
21    /// brand color styles the drawn route).
22    #[serde(default)]
23    pub line: Option<LineName>,
24    /// Intermediate stations the drawn route passes through without
25    /// stopping, in `from` -> `to` order. Empty routes along the
26    /// street mesh automatically (`GameMap::route_on_surface`); set `via` to
27    /// override a single edge's drawn path.
28    #[serde(default)]
29    pub via: Vec<StationId>,
30    /// Draw the straight chord and skip street-mesh routing (metro
31    /// edges only - relevant where `via` is empty and the mesh
32    /// detour reads worse than the direct line).
33    #[serde(default)]
34    pub direct: bool,
35}
36
37impl Edge {
38    /// Creates an edge, canonicalizing the endpoint order.
39    pub fn new(a: StationId, b: StationId, transport: Transport) -> Self {
40        let (from, to) = if a <= b { (a, b) } else { (b, a) };
41        Self {
42            from,
43            to,
44            transport,
45            line: None,
46            via: Vec::new(),
47            direct: false,
48        }
49    }
50
51    /// The neighbor entry as seen from `from` (points to `to`).
52    pub fn forward(&self) -> Neighbor {
53        Neighbor {
54            station: self.to,
55            transport: self.transport,
56            line: self.line.clone(),
57        }
58    }
59
60    /// The neighbor entry as seen from `to` (points back to `from`).
61    pub fn backward(&self) -> Neighbor {
62        Neighbor {
63            station: self.from,
64            transport: self.transport,
65            line: self.line.clone(),
66        }
67    }
68}

70#[cfg(test)]
71mod tests {
72    use super::*;
73
74    #[test]
75    fn canonicalizes_endpoint_order() {
76        let edge = Edge::new(StationId::new(9), StationId::new(3), Transport::Bus);
77        assert_eq!(edge.from, StationId::new(3));
78        assert_eq!(edge.to, StationId::new(9));
79    }
80
81    #[test]
82    fn neighbor_entries_point_both_ways() {
83        let edge = Edge::new(StationId::new(3), StationId::new(9), Transport::Bus);
84        assert_eq!(edge.forward().station, StationId::new(9));
85        assert_eq!(edge.backward().station, StationId::new(3));
86        assert_eq!(edge.forward().transport, Transport::Bus);
87    }
