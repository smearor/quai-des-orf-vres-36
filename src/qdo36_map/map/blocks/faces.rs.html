1//! `BlockFaces` - the set of faces in a block layout.
2
3use crate::Segment;
4use crate::map::DistrictName;
5use crate::map::Park;
6use crate::map::backdrop::Landmark;
7use crate::map::blocks::BlockFace;
8use smearor_commons_geometry::Millimeters;
9use smearor_commons_geometry::Vec2;
10use std::ops::Deref;
11use std::ops::DerefMut;
12
13/// The faces of a block layout as a set: tunnels attach, and
14/// district names pin and fill, on the whole collection.
15/// Derefs to the `[BlockFace]` slice.
16#[derive(Debug, Clone, Default, PartialEq)]
17pub struct BlockFaces(pub Vec<BlockFace>);
18
19impl BlockFaces {
20    /// Attaches every segment in `tunnels` to each face it
21    /// touches (see [`BlockFace::attach_tunnel`]).
22    pub fn attach_all(&mut self, tunnels: impl IntoIterator<Item = Segment<Millimeters>>) {
23        for segment in tunnels {
24            for face in self.0.iter_mut() {
25                face.attach_tunnel(segment);
26            }
27        }
28    }
29
30    /// Names the still-unnamed inner face containing `pos` after
31    /// `district` - the face pins the positioned quarter's name
32    /// (outer faces never carry one).
33    pub fn pin_district(&mut self, pos: &Vec2<Millimeters>, name: &DistrictName) {
34        if let Some(face) = self.0.iter_mut().find(|face| !face.is_outer() && face.district.is_none() && face.contains(pos)) {
35            face.district = Some(name.clone());
36        }
37    }
38
39    /// Names each still-unnamed inner face after the next name
40    /// in `names`, in order - the fallback for quarters without
41    /// a dataset position.
42    pub fn fill_districts(&mut self, names: impl IntoIterator<Item = DistrictName>) {
43        let mut names = names.into_iter();
44        for face in self.0.iter_mut().filter(|face| !face.is_outer() && face.district.is_none()) {
45            let Some(name) = names.next() else { break };
46            face.district = Some(name);
47        }
48    }
49
50    /// Parks and landmarks land in the inner face containing
51    /// them.
52    pub fn populate(&mut self, parks: &[Park], landmarks: &[Landmark]) {
53        for face in self.inner_mut() {
54            face.parks = parks.iter().filter(|park| face.contains(&park.pos)).cloned().collect();
55            face.landmarks = landmarks.iter().filter(|landmark| face.contains(&landmark.pos)).cloned().collect();
56        }
57    }
58
59    /// The inner (non-outer) faces.
60    pub fn inner(&self) -> impl Iterator<Item = &BlockFace> {
61        self.0.iter().filter(|face| !face.is_outer())
62    }
63
64    /// The inner (non-outer) faces, mutably.
65    pub fn inner_mut(&mut self) -> impl Iterator<Item = &mut BlockFace> {
66        self.0.iter_mut().filter(|face| !face.is_outer())
67    }
68}
69
70impl Deref for BlockFaces {
71    type Target = [BlockFace];
72
73    fn deref(&self) -> &Self::Target {
74        &self.0
75    }
76}
77
78impl DerefMut for BlockFaces {
79    fn deref_mut(&mut self) -> &mut Self::Target {
80        &mut self.0
81    }
82}
83
84impl<'a> IntoIterator for &'a BlockFaces {
85    type Item = &'a BlockFace;
86    type IntoIter = std::slice::Iter<'a, BlockFace>;
87
88    fn into_iter(self) -> Self::IntoIter {
89        self.0.iter()
90    }
