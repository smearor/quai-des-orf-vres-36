1//! `House` - one procedural building footprint inside a block.
2
3use crate::map::Polygon;
4
5use crate::map::blocks::Depth;
6use crate::map::blocks::Height;
7use crate::map::blocks::Units;
8use crate::map::blocks::Variant;
9
10/// A procedural building inside a block: its footprint polygon
11/// plus the frontage class and texture variant a style draws it
12/// with. Row houses are quads (front edge on the outline ring);
13/// corner houses wrap a sharp block vertex with a rounded facade
14/// and have more vertices.
15#[derive(Debug, Clone, PartialEq)]
16pub struct House {
17    /// Footprint polygon in map mm (≥3 vertices, closed implicitly).
18    pub footprint: Polygon,
19    /// Frontage width in units (1-4 for row houses, 1-2 per side
20    /// for corner houses) - drives the unit split lines and the
21    /// per-width texture variants.
22    pub units: Units,
23    /// Texture variant within the unit-width class (0..4) - styles
24    /// pick fills and hatches by `(units, variant)`.
25    pub variant: Variant,
26    /// How deep the house reaches into the block - rolled in
27    /// 0.5 mm steps per house.
28    pub depth: Depth,
29    /// Extrusion factor for the isometric modes (`STANDARD` = the
30    /// block's base height) - rolled in 0.25 steps, landmark
31    /// volumes use `LANDMARK`.
32    pub height: Height,
33}
