1//! Planar face extraction: the map's cut lines (streets, river,
2//! border) are split at every junction into a planar graph, then
3//! each directed half-edge's left face is traced by always taking
4//! the sharpest right turn at the next vertex.
5
6use crate::Segment;
7use crate::map::blocks::BlockCut;
8use crate::map::blocks::BlockFace;
9use crate::map::blocks::BlockFaces;

11use crate::map::blocks::cut_segment::CutSegment;
12use crate::map::blocks::splits::Splits;
13use smearor_commons_geometry::Millimeters;

15use std::collections::HashMap;
16use std::collections::HashSet;

18/// Snap radius for sealing T-junctions and floating-point noise, in
19/// mm: a segment endpoint within this distance of another segment's
20/// interior docks onto it (piers sit ~1.5 mm off the water course).
21const SNAP: Millimeters = Millimeters::new(2.5);
22
23/// The smallest face area worth keeping (mm²) - dead-end spurs and
24/// snapping slivers drop below it.
25const MIN_AREA: SquareMillimeters = SquareMillimeters::new(4.0);
26
27/// The union-find root of `index` (path halving).
28fn root(parent: &mut [usize], mut index: usize) -> usize {
29    while parent[index] != index {
30        parent[index] = parent[parent[index]];
31        index = parent[index];
32    }
33    index
34}
35
36/// Splits all segments at crossings and T-junctions, clusters the
37/// endpoints into nodes and traces every half-edge's left face.
38/// Returns the bounded faces (the unbounded outside and degenerate
39/// slivers are dropped).
40impl BlockFace {
41    /// The faces the `segments` cut set encloses.
42    pub fn extract(segments: &[CutSegment]) -> BlockFaces {
43        let straights: Vec<Segment<Millimeters>> = segments.iter().map(Segment::from).collect();
44        // Split parameters per segment: crossings split both sides,
45        // T-junctions only the segment being docked onto.
46        let mut splits = vec![Splits::default(); segments.len()];
47        for (i, a) in straights.iter().enumerate() {
48            for (j, b) in straights.iter().enumerate().skip(i + 1) {
49                if let Some(point) = a.intersection(b) {
50                    for (index, segment) in [(i, a), (j, b)] {
51                        let direction = segment.to - segment.from;
52                        let len_sq = direction.dot::<f64>(&direction).max(1e-18);
53                        splits[index].push((point - segment.from).dot::<f64>(&direction) / len_sq);
54                    }
55                }
56            }
57            for (j, b) in straights.iter().enumerate() {
58                if i == j {
59                    continue;
60                }
61                for endpoint in [b.from, b.to] {
62                    if let Some(projection) = a.interior_projection(&endpoint)
63                        && projection.point.distance(&endpoint) < SNAP
64                    {
65                        splits[i].push(projection.fraction);
66                    }
67                }
68            }
69        }
70        // Atomic fragments: each segment cut at its sorted split points.
71        let mut fragments: Vec<(Vec2<Millimeters>, Vec2<Millimeters>, BlockCut)> = Vec::new();
72        for (segment, ts) in segments.iter().zip(splits.iter_mut()) {
73            ts.finish();
74            let point = |t: f64| -> Vec2<Millimeters> {
75                Vec2::new(segment.from.x + (segment.to.x - segment.from.x) * t, segment.from.y + (segment.to.y - segment.from.y) * t)
76            };
77            for (t0, t1) in ts.cuts() {
78                let (from, to) = (point(t0), point(t1));
79                if from.distance(&to).as_f64() > 1e-9 {
80                    fragments.push((from, to, segment.cut));
81                }
82            }
83        }
84        // Cluster fragment endpoints into nodes (union-find over the
85        // snap radius).
86        let endpoints: Vec<Vec2<Millimeters>> = fragments.iter().flat_map(|(a, b, _)| [*a, *b]).collect();
87        let mut parent: Vec<usize> = (0..endpoints.len()).collect();
88        for i in 0..endpoints.len() {
89            for j in i + 1..endpoints.len() {
90                if endpoints[i].distance(&endpoints[j]) < SNAP {
91                    let (ri, rj) = (root(&mut parent, i), root(&mut parent, j));
92                    if ri != rj {
93                        parent[ri] = rj;
94                    }
95                }
96            }
97        }
98        // Margin preference when fragments coincide: `BlockCut`'s
99        // order - street > water > border.
100        let mut nodes: Vec<Vec2<Millimeters>> = Vec::new();
101        let mut node_ids: HashMap<usize, usize> = HashMap::new();
102        let mut adjacency: Vec<Vec<(usize, BlockCut, f64)>> = Vec::new();
103        let mut seen_edges: HashMap<(usize, usize), BlockCut> = HashMap::new();
104        for (index, (from, to, cut)) in fragments.iter().enumerate() {
105            let (u, v) = (
106                *node_ids.entry(root(&mut parent, index * 2)).or_insert_with(|| {
107                    nodes.push(*from);
108                    adjacency.push(Vec::new());
109                    nodes.len() - 1
110                }),
111                *node_ids.entry(root(&mut parent, index * 2 + 1)).or_insert_with(|| {
112                    nodes.push(*to);
113                    adjacency.push(Vec::new());
114                    nodes.len() - 1
115                }),
116            );
117            if u == v {
118                continue;
119            }
120            let key = if u < v { (u, v) } else { (v, u) };
121            match seen_edges.get(&key) {
122                Some(existing) if existing <= cut => continue,
123                _ => {
124                    seen_edges.insert(key, *cut);
125                }
126            }
127            adjacency[u].push((v, *cut, (to.y - from.y).atan2(to.x - from.x)));
128            adjacency[v].push((u, *cut, (from.y - to.y).atan2(from.x - to.x)));
129        }
130        for edges in adjacency.iter_mut() {
131            edges.sort_by(|a, b| a.2.total_cmp(&b.2));
132        }
133        // Trace left faces: at each vertex take the edge just before the
134        // twin in ascending-angle order (the sharpest right turn).
135        let mut faces = Vec::new();
136        let mut visited: HashSet<(usize, usize)> = HashSet::new();
137        let max_steps = adjacency.iter().map(Vec::len).sum::<usize>() + 4;
138        for u in 0..adjacency.len() {
139            for edge_index in 0..adjacency[u].len() {
140                let (mut from, mut to, mut cut) = {
141                    let entry = adjacency[u][edge_index];
142                    (u, entry.0, entry.1)
143                };
144                if visited.contains(&(from, to)) {
145                    continue;
146                }
147                let start = (from, to);
148                let mut vertices = vec![from];
149                let mut cuts = Vec::new();
150                loop {
151                    visited.insert((from, to));
152                    cuts.push(cut);
153                    vertices.push(to);
154                    let neighbors = &adjacency[to];
155                    let twin = neighbors.iter().position(|(node, _, _)| *node == from).unwrap_or(0);
156                    let (next, next_cut, _) = neighbors[(twin + neighbors.len() - 1) % neighbors.len()];
157                    (from, to, cut) = (to, next, next_cut);
158                    if (from, to) == start || cuts.len() > max_steps {
159                        break;
160                    }
161                }
162                vertices.pop();
163                let mut face = BlockFace {
164                    outline: vertices.iter().map(|node| nodes[*node]).collect(),
165                    cuts,
166                    tunnels: Vec::new(),
167                    district: None,
168                    parks: Vec::new(),
169                    landmarks: Vec::new(),
170                };
171                face.remove_spikes();
172                if face.outline.len() >= 3 && face.signed_area() > MIN_AREA {
173                    faces.push(face);
174                }
175            }
176        }
177        BlockFaces(faces)
178    }
179}

181#[cfg(test)]
182mod tests {
183    use super::*;
184    use crate::StationId;

187    fn street(x1: f64, y1: f64, x2: f64, y2: f64) -> CutSegment {
188        CutSegment {
189            from: Vec2::new(x1, y1),
190            to: Vec2::new(x2, y2),
191            cut: BlockCut::Street(StationPair::new(StationId::new(1), StationId::new(2))),
192        }
193    }
194
195    fn border(x_max: f64, y_max: f64) -> Vec<CutSegment> {
196        let corners = [Vec2::new(0.0, 0.0), Vec2::new(x_max, 0.0), Vec2::new(x_max, y_max), Vec2::new(0.0, y_max)];
197        (0..4)
198            .map(|i| CutSegment {
199                from: corners[i],
200                to: corners[(i + 1) % 4],
201                cut: BlockCut::Border,
202            })
203            .collect()
204    }
205
206    #[test]
207    fn a_cross_inside_the_border_gives_four_faces() {
208        let mut segments = border(100.0, 100.0);
209        segments.push(street(0.0, 50.0, 100.0, 50.0));
210        segments.push(street(50.0, 0.0, 50.0, 100.0));
211        let faces = BlockFace::extract(&segments);
212        assert_eq!(faces.len(), 4, "{:?}", faces.iter().map(|f| f.outline.len()).collect::<Vec<_>>());
213        assert!(faces.iter().all(|f| f.outline.len() == 4));
214    }
215
216    #[test]
217    fn a_river_plus_one_street_split_the_map() {
218        let mut segments = border(100.0, 100.0);
219        segments.push(CutSegment {
220            from: Vec2::new(-10.0, 50.0),
221            to: Vec2::new(110.0, 50.0),
222            cut: BlockCut::Water,
223        });
224        segments.push(street(50.0, 0.0, 50.0, 100.0));
225        let faces = BlockFace::extract(&segments);
226        assert_eq!(faces.len(), 4);
227        let water_edges: usize = faces.iter().map(|f| f.cuts.iter().filter(|c| **c == BlockCut::Water).count()).sum();
228        assert_eq!(water_edges, 4, "each of the two inner water fragments borders two faces");
229    }
230
231    #[test]
232    fn a_dead_end_street_leaves_one_face() {
233        let mut segments = border(100.0, 100.0);
234        segments.push(street(0.0, 50.0, 40.0, 50.0));
235        let faces = BlockFace::extract(&segments);
236        assert_eq!(faces.len(), 1);
237        // Border corners + the junction node where the spur docks.
238        assert_eq!(faces[0].outline.len(), 5);
239    }

241    #[test]
242    fn a_t_junction_seals_against_the_river() {
243        let mut segments = border(100.0, 100.0);
244        segments.push(CutSegment {
245            from: Vec2::new(-10.0, 50.0),
246            to: Vec2::new(110.0, 50.0),
247            cut: BlockCut::Water,
248        });
249        // Pier access street ending 1.5mm short of the waterline:
250        // it docks onto the river and splits the upper region.
251        segments.push(street(50.0, 0.0, 50.0, 48.5));
252        let faces = BlockFace::extract(&segments);
253        assert_eq!(faces.len(), 3, "{:?}", faces.iter().map(|f| f.signed_area()).collect::<Vec<_>>());
254    }
