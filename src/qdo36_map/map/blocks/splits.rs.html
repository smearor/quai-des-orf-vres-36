1//! `Splits` - a segment's cut parameters along 0..1.
2
3/// A segment's cut parameters along 0..1: where crossings and
4/// snapped endpoints split it into atomic fragments.
5#[derive(Clone, Default)]
6pub(super) struct Splits(pub Vec<f64>);
7
8impl Splits {
9    /// Adds a cut at parameter `t`.
10    pub fn push(&mut self, t: f64) {
11        self.0.push(t);
12    }
13
14    /// Closes the range: adds the 0/1 endpoints, sorts, dedups -
15    /// after `finish` the parameters tile the segment.
16    pub fn finish(&mut self) {
17        self.0.push(0.0);
18        self.0.push(1.0);
19        self.0.sort_by(f64::total_cmp);
20        self.0.dedup_by(|a, b| (*a - *b).abs() < 1e-9);
21    }
22
23    /// The consecutive cut windows (after [`Self::finish`]).
24    pub fn cuts(&self) -> impl Iterator<Item = (f64, f64)> + '_ {
25        self.0.windows(2).map(|w| (w[0], w[1]))
26    }
27}
