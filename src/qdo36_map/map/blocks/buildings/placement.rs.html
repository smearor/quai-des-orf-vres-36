1//! `Placement` - the state shared by the row and corner passes.
2
3use crate::map::Polygon;
4use crate::map::blocks::House;
5use crate::map::blocks::Obstacles;
6
7/// Shared placement state for one ring's passes: the block
8/// outline every footprint must stay inside, the obstacles
9/// carving the block (tunnels, decorative streets, parks), the
10/// houses placed so far and the determinism channels.
11pub(super) struct Placement<'a> {
12    /// Block outline all footprints must stay inside.
13    pub(super) outline: &'a Polygon,
14    /// Obstacle polygons inside the block - houses in their way
15    /// are skipped.
16    pub(super) obstacles: &'a Obstacles,
17    /// Houses placed so far (row and corner passes append).
18    pub(super) placed: Vec<House>,
19    /// Determinism seed for the noise channels.
20    pub(super) seed: u64,
21    /// Ring index mixed into the noise channels.
22    pub(super) ring_index: usize,
23}
24
25impl Placement<'_> {
26    /// Whether a candidate footprint may be placed: its interior
27    /// points (from `inside_from` on) stay inside the block
28    /// `outline` (facade points may ride on it), it overlaps no
29    /// obstacle (tunnels, decorative streets, parks) and no
30    /// already-placed house.
31    pub(super) fn fits(&self, footprint: &Polygon, inside_from: usize) -> bool {
32        footprint[inside_from..].iter().all(|p| self.outline.contains_or_touches(p))
33            && !self.obstacles.iter().any(|obstacle| footprint.overlaps(obstacle))
34            && !self.placed.iter().any(|other| footprint.overlaps(&other.footprint))
35    }
36}
