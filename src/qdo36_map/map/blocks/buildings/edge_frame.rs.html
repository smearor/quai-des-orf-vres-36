1//! `EdgeFrame` - one edge's direction data for house placement.
2
3use crate::map::Polygon;
4use smearor_commons_geometry::Millimeters;
5use smearor_commons_geometry::Vec2;
6
7use crate::map::blocks::Angle;
8
9/// One edge's direction data for house placement.
10pub(super) struct EdgeFrame {
11    /// Unit vector along the edge.
12    pub(super) direction: Vec2<Millimeters>,
13    /// Left unit normal, pointing into the block.
14    pub(super) inward: Vec2<Millimeters>,
15    /// Edge length.
16    pub(super) length: Millimeters,
17}
18
19impl EdgeFrame {
20    /// The frame of one edge `from -> to`: direction and inward
21    /// (left) unit normal plus the edge's length.
22    pub(super) fn new(from: Vec2<Millimeters>, to: Vec2<Millimeters>) -> Self {
23        let delta = to - from;
24        let length = delta.magnitude().max(Millimeters::new(1e-9));
25        let direction = delta.normalized();
26        Self {
27            direction,
28            inward: direction.perpendicular(),
29            length,
30        }
31    }
32
33    /// The edge frames of a ring polygon (one per edge).
34    pub(super) fn of_ring(ring: &Polygon) -> Vec<Self> {
35        ring.segments().map(|edge| Self::new(edge.from, edge.to)).collect()
36    }
37
38    /// The interior angle at vertex `i` of a positive-area
39    /// polygon's ring (sharp corners < `CORNER_MAX_ANGLE`).
40    pub(super) fn interior_angle(frames: &[EdgeFrame], i: usize) -> Angle {
41        let n = frames.len();
42        Angle::interior(frames[(i + n - 1) % n].direction, frames[i].direction)
43    }
44}
