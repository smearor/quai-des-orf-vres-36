1//! `CornerWedge` - the flatiron wedge at a ring vertex: the two
2//! edge frames plus the rolled dimensions.
3
4use crate::map::Polyline;
5use crate::map::blocks::Depth;
6use crate::map::blocks::buildings::CORNER_ARC_STEPS;
7use crate::map::blocks::buildings::arc_sweep;
8use crate::map::blocks::buildings::corner_nose::CornerNose;
9use crate::map::blocks::buildings::edge_frame::EdgeFrame;
10use crate::map::geometry::Line;
11use tangible_objects_model::Millimeters;
12use tangible_objects_model::Vec2;

14/// The flatiron wedge at a ring vertex: the corner position, the
15/// edge frames entering (`prev`) and leaving (`next`) it, and
16/// the rolled facade width and depth.
17pub(super) struct CornerWedge<'a> {
18    /// The vertex position in map mm.
19    pub corner: Vec2<Millimeters>,
20    /// The edge frame entering the vertex.
21    pub prev: &'a EdgeFrame,
22    /// The edge frame leaving the vertex.
23    pub next: &'a EdgeFrame,
24    /// The facade frontage (`units * house width`).
25    pub frontage: Millimeters,
26    /// The house depth.
27    pub depth: Depth,
28}
29
30impl CornerWedge<'_> {
31    /// The tangent nose arc bulging toward the vertex plus the
32    /// facade lengths consumed on both edges (the facade may not
33    /// run past half a neighbor edge).
34    pub fn nose(&self, interior: f64) -> CornerNose {
35        let depth = Millimeters::new(self.depth.as_f64());
36        let half = (interior / 2.0).max(0.05);
37        let radius = (depth * 0.9).min(self.frontage * 0.5).max(Millimeters::new(0.8));
38        let tangent_len = (radius / half.tan()).min(self.prev.length.min(self.next.length) * 0.5);
39        let center_dist = radius / half.sin();
40        let (bisector_x, bisector_y) = (self.next.direction.x - self.prev.direction.x, self.next.direction.y - self.prev.direction.y);
41        let bisector_len = bisector_x.hypot(bisector_y).max(1e-9);
42        let bisector = Vec2::<Millimeters>::new(bisector_x / bisector_len, bisector_y / bisector_len);
43        let center = self.corner + bisector * center_dist.as_f64();
44        let nose_prev = self.corner - self.prev.direction * tangent_len.as_f64();
45        let nose_next = self.corner + self.next.direction * tangent_len.as_f64();
46        let facade_prev = self.frontage.max(tangent_len).min(self.prev.length * 0.5);
47        let facade_next = self.frontage.max(tangent_len).min(self.next.length * 0.5);
48        let front_prev = self.corner - self.prev.direction * facade_prev.as_f64();
49        let front_next = self.corner + self.next.direction * facade_next.as_f64();
50        // The arc sweeps the circle part facing the vertex.
51        let start = (nose_prev.y - center.y).atan2(nose_prev.x - center.x);
52        let end = (nose_next.y - center.y).atan2(nose_next.x - center.x);
53        let vertex_angle = (self.corner.y - center.y).atan2(self.corner.x - center.x);
54        let sweep = arc_sweep(start, end, vertex_angle);
55        let mut points = vec![nose_prev];
56        for step in 1..CORNER_ARC_STEPS.as_usize() {
57            let angle = start + sweep * step as f64 / CORNER_ARC_STEPS.as_f64();
58            points.push(Vec2::new(center.x + angle.cos() * radius.as_f64(), center.y + angle.sin() * radius.as_f64()));
59        }
60        points.push(nose_next);
61        CornerNose {
62            front_prev,
63            front_next,
64            points: Polyline::new(points),
65            facade_prev,
66            facade_next,
67        }
68    }
69
70    /// The back wall behind `front_prev`/`front_next`: straight
71    /// sides at `depth` plus the offset inner corner where they
72    /// meet (miter-capped on very acute corners).
73    pub fn back_wall(&self, front_prev: Vec2<Millimeters>, front_next: Vec2<Millimeters>) -> [Vec2<Millimeters>; 3] {
74        let depth = self.depth.as_f64();
75        let back_next = front_next + self.next.inward * depth;
76        let back_prev = front_prev + self.prev.inward * depth;
77        let inner = Line {
78            origin: self.corner + self.prev.inward * depth,
79            direction: self.prev.direction,
80        }
81        .intersect(&Line {
82            origin: self.corner + self.next.inward * depth,
83            direction: self.next.direction,
84        })
85        .map(|point| {
86            // Miter cap for very acute corners.
87            let (miter_x, miter_y) = (point.x - self.corner.x, point.y - self.corner.y);
88            let miter_cap = depth * 4.0;
89            let miter_len = miter_x.hypot(miter_y);
90            if miter_len > miter_cap {
91                self.corner + Vec2::new(miter_x, miter_y) * (miter_cap / miter_len)
92            } else {
93                point
94            }
95        })
96        .unwrap_or_else(|| (back_prev + back_next) / 2.0);
97        [back_next, inner, back_prev]
98    }
99}
