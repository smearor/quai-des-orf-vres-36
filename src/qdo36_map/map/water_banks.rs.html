1//! `WaterBanks` - the two bank lines of a water body.
2
3use crate::map::Polyline;
4use tangible_objects_model::Millimeters;
5use tangible_objects_model::Vec2;
6
7/// Half the channel a water course claims on each side (mm) -
8/// the setback water `BlockCut`s keep from the course centerline.
9pub const WATER_HALF_WIDTH: Millimeters = Millimeters::new(3.5);
10
11/// A water body's two bank lines in map mm: `left`/`right` as
12/// seen along the course direction.
13pub struct WaterBanks {
14    /// The bank left of the course direction.
15    pub left: Polyline,
16    /// The bank right of the course direction.
17    pub right: Polyline,
18}
19
20impl WaterBanks {
21    /// Computes the banks `half_width` off the `points` centerline.
22    /// Both banks are empty when the centerline has fewer than two
23    /// points.
24    pub fn new(points: &[Vec2<Millimeters>], half_width: Millimeters) -> Self {
25        if points.len() < 2 {
26            return Self {
27                left: Polyline::default(),
28                right: Polyline::default(),
29            };
30        }
31        let course = Polyline::new(points.to_vec());
32        Self {
33            left: course.offset(half_width),
34            right: course.offset(-half_width),
35        }
36    }
37
38    /// The widest absolute parameter `t` where the line `center +
39    /// axis * t` meets either bank within `reach` - `Millimeters::ZERO`
40    /// when neither bank is reached.
41    pub fn crossing_span(&self, center: Vec2<Millimeters>, axis: Vec2<Millimeters>, reach: Millimeters) -> Millimeters {
42        [&self.left, &self.right]
43            .into_iter()
44            .filter_map(|bank| bank.axis_param(center, axis, reach))
45            .map(Millimeters::abs)
46            .fold(Millimeters::ZERO, Millimeters::max)
47    }
48}

50#[cfg(test)]
51mod tests {
52    use crate::map::Polyline;
53    use tangible_objects_model::Millimeters;
54    use tangible_objects_model::Vec2;
55
56    #[test]
57    fn offsets_straight_line_perpendicular() {
58        let line = vec![Vec2::<Millimeters>::new(0.0, 0.0), Vec2::new(10.0, 0.0)];
59        let bank = Polyline::new(line).offset(Millimeters::new(2.0));
60        assert_eq!(bank.len(), 2);
61        for (point, expected) in bank.iter().zip([(0.0, 2.0), (10.0, 2.0)]) {
62            assert!((point.x - expected.0).abs() < 1e-9);
63            assert!((point.y - expected.1).abs() < 1e-9);
64        }
65    }
66
67    #[test]
68    fn bend_vertex_averages_segment_normals() {
69        let line = vec![Vec2::<Millimeters>::new(0.0, 0.0), Vec2::new(10.0, 0.0), Vec2::new(10.0, 10.0)];
70        let corner = Polyline::new(line).offset(Millimeters::new(2.0))[1];
71        // Normals (0,1) + (-1,0) normalize to the (-1,1) diagonal.
72        assert!((corner.x - (10.0 - 2.0_f64.sqrt())).abs() < 1e-9);
73        assert!((corner.y - 2.0_f64.sqrt()).abs() < 1e-9);
74    }
