13/// An undirected travel route between two stations.
14///
15/// Stored canonicalized with `from < to` - the game graph has no
16/// one-way routes.
17#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
18pub struct Edge {
19    /// Station at one end (the lower id after canonicalization).
20    pub from: StationId,
21    /// Station at the other end.
22    pub to: StationId,
23    /// Transport mode this route is used with.
24    pub transport: Transport,
25    /// Transit line serving this route (metro edges only - the line's
26    /// brand color styles the drawn route).
27    #[serde(default)]
28    pub line: Option<LineName>,
29    /// Intermediate stations the drawn route passes through without
30    /// stopping, in `from` -> `to` order. Empty routes along the
31    /// street mesh automatically (`GameMap::route_on_surface`); set `via` to
32    /// override a single edge's drawn path.
33    #[serde(default)]
34    pub via: Vec<StationId>,
35    /// Draw the straight chord and skip street-mesh routing (metro
36    /// edges only - relevant where `via` is empty and the mesh
37    /// detour reads worse than the direct line).
38    #[serde(default)]
39    pub direct: bool,
40}
41
42impl Edge {
43    /// Creates an edge, canonicalizing the endpoint order.
44    pub fn new(a: StationId, b: StationId, transport: Transport) -> Self {
45        let (from, to) = if a <= b { (a, b) } else { (b, a) };
46        Self {
47            from,
48            to,
49            transport,
50            line: None,
51            via: Vec::new(),
52            direct: false,
53        }
54    }
55
56    /// Whether the edge's drawn path routes over the surface
57    /// street mesh: non-`direct` metro edges pass stations
58    /// without stopping and follow the streets above; tunnels,
59    /// `via` chains and direct chords do not.
60    /// Whether the edge runs `transport`.
61    pub fn is_transport(&self, transport: Transport) -> bool {
62        self.transport == transport
63    }
64
65    pub fn uses_street_mesh(&self) -> bool {
66        self.transport == Transport::Metro && self.via.is_empty() && !self.direct
67    }
68
69    /// The neighbor entry as seen from `from` (points to `to`).
70    pub fn forward(&self) -> Neighbor {
71        Neighbor {
72            station: self.to,
73            transport: self.transport,
74            line: self.line.clone(),
75        }
76    }
77
78    /// The neighbor entry as seen from `to` (points back to `from`).
79    pub fn backward(&self) -> Neighbor {
80        Neighbor {
81            station: self.from,
82            transport: self.transport,
83            line: self.line.clone(),
84        }
85    }
86
87    /// The edge's station chain `from` -> `via` -> `to` - the drawn
88    /// route's stop order (`via` stations are passed through
89    /// without stopping).
90    pub fn station_chain(&self) -> impl Iterator<Item = StationId> + '_ {
91        [self.from].into_iter().chain(self.via.iter().copied()).chain([self.to])
92    }
93
94    /// The full station chain `from -> via -> to` as an
95    /// `EdgePath`.
96    pub fn chain(&self) -> EdgePath {
97        EdgePath(self.station_chain().collect())
98    }
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104
105    #[test]
106    fn canonicalizes_endpoint_order() {
107        let edge = Edge::new(StationId::new(9), StationId::new(3), Transport::Bus);
108        assert_eq!(edge.from, StationId::new(3));
109        assert_eq!(edge.to, StationId::new(9));
110    }
111
112    #[test]
113    fn neighbor_entries_point_both_ways() {
114        let edge = Edge::new(StationId::new(3), StationId::new(9), Transport::Bus);
115        assert_eq!(edge.forward().station, StationId::new(9));
116        assert_eq!(edge.backward().station, StationId::new(3));
117        assert_eq!(edge.forward().transport, Transport::Bus);
118    }
