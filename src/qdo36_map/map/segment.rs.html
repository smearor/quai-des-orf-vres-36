1//! `Segment` - a straight segment between two positions.
2
3use tangible_objects_model::Vec2;
4
5/// A straight segment in unit `U` (`Vec2<Millimeters>` on the map,
6/// `Vec2<Pixels>` on screen) - the geometry an edge draws between
7/// its two endpoints.
8#[derive(Debug, Clone, Copy, PartialEq)]
9pub struct Segment<U> {
10    /// Segment start position.
11    pub from: Vec2<U>,
12    /// Segment end position.
13    pub to: Vec2<U>,
14}
15
16impl<U> Segment<U> {
17    /// A segment from `from` to `to`.
18    pub const fn new(from: Vec2<U>, to: Vec2<U>) -> Self {
19        Self { from, to }
20    }
21}
22
23impl<U: Copy> Segment<U> {
24    /// The segment's center point.
25    pub fn midpoint(&self) -> Vec2<U> {
26        Vec2::new((self.from.x + self.to.x) / 2.0, (self.from.y + self.to.y) / 2.0)
27    }
28
29    /// Strict crossing - `true` when the interiors of both segments
30    /// intersect (shared endpoints are junctions, not crossings).
31    pub fn crosses(&self, other: &Self) -> bool {
32        let orient = |p: Vec2<U>, q: Vec2<U>, r: Vec2<U>| (q.x - p.x) * (r.y - p.y) - (q.y - p.y) * (r.x - p.x);
33        orient(self.from, self.to, other.from) * orient(self.from, self.to, other.to) < 0.0
34            && orient(other.from, other.to, self.from) * orient(other.from, other.to, self.to) < 0.0
35    }
36
37    /// Perpendicular distance of `point` to the segment when the
38    /// closest point lies between the endpoints - `None` when the
39    /// projection falls beyond either end or the segment is
40    /// degenerate.
41    pub fn interior_distance(&self, point: &Vec2<U>) -> Option<f64> {
42        let (vx, vy) = (self.to.x - self.from.x, self.to.y - self.from.y);
43        let len_sq = vx * vx + vy * vy;
44        if len_sq == 0.0 {
45            return None;
46        }
47        let t = ((point.x - self.from.x) * vx + (point.y - self.from.y) * vy) / len_sq;
48        if !(0.0..=1.0).contains(&t) {
49            return None;
50        }
51        Some((point.x - (self.from.x + t * vx)).hypot(point.y - (self.from.y + t * vy)))
52    }
53
54    /// The point where the two segments meet - `None` when they are
55    /// parallel, degenerate or the meeting lies outside either
56    /// segment.
57    pub fn intersection(&self, other: &Self) -> Option<Vec2<U>> {
58        let (vx, vy) = (self.to.x - self.from.x, self.to.y - self.from.y);
59        let (wx, wy) = (other.to.x - other.from.x, other.to.y - other.from.y);
60        let denom = vx * wy - vy * wx;
61        if denom == 0.0 {
62            return None;
63        }
64        let (ox, oy) = (other.from.x - self.from.x, other.from.y - self.from.y);
65        let t = (ox * wy - oy * wx) / denom;
66        let u = (ox * vy - oy * vx) / denom;
67        if !(0.0..=1.0).contains(&t) || !(0.0..=1.0).contains(&u) {
68            return None;
69        }
70        Some(Vec2::new(self.from.x + t * vx, self.from.y + t * vy))
71    }
72}
