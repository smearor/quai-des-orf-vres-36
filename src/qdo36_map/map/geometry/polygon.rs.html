10/// The miter cap for [`Polygon::offset`] corners - acute corners
11/// may extend at most this many edge distances.
12const MITER_CAP: f64 = 4.0;
13
14/// A closed ring of points: footprints, block outlines, park
15/// blobs - the last point connects back to the first. Derefs to
16/// the `[Vec2]` slice so every `&[Vec2]` geometry helper keeps
17/// working.
18#[derive(Debug, Clone, PartialEq)]
19pub struct Polygon(pub Vec<Vec2<Millimeters>>);
20
21impl Polygon {
22    /// A polygon from its vertex ring.
23    pub fn new(points: Vec<Vec2<Millimeters>>) -> Self {
24        Self(points)
25    }
26
27    /// Removes and returns the vertex at `index`.
28    pub fn remove(&mut self, index: usize) -> Vec2<Millimeters> {
29        self.0.remove(index)
30    }
31
32    /// The ring's edge segments - the last vertex connects back
33    /// to the first.
34    pub fn segments(&self) -> impl Iterator<Item = Segment<Millimeters>> + '_ {
35        let points = &self.0;
36        (0..points.len()).map(|i| Segment::new(points[i], points[(i + 1) % points.len()]))
37    }
38
39    /// The shoelace signed area in mm² (positive =
40    /// counter-clockwise in the coordinate frame).
41    pub fn signed_area(&self) -> SquareMillimeters {
42        self.segments()
43            .map(|edge| edge.from.cross::<SquareMillimeters>(&edge.to))
44            .sum::<SquareMillimeters>()
45            * 0.5
46    }
47
48    /// The ring's vertex centroid (uniform vertex average).
49    pub fn centroid(&self) -> Vec2<Millimeters> {
50        let count = self.0.len().max(1) as f64;
51        Vec2::new(self.0.iter().map(|p| p.x).sum::<f64>() / count, self.0.iter().map(|p| p.y).sum::<f64>() / count)
52    }
53
54    /// Whether `point` lies strictly inside the ring (even-odd
55    /// ray cast).
56    pub fn contains(&self, point: &Vec2<Millimeters>) -> bool {
57        let mut inside = false;
58        for edge in self.segments() {
59            let (a, b) = (edge.from, edge.to);
60            let crosses_ray = (a.y > point.y) != (b.y > point.y);
61            if crosses_ray && point.x < a.x + (b.x - a.x) * (point.y - a.y) / (b.y - a.y) {
62                inside = !inside;
63            }
64        }
65        inside
66    }
67
68    /// Whether `point` lies inside the ring or on its boundary -
69    /// [`Self::contains`] plus a tolerance for facade points
70    /// riding on an outline edge.
71    pub fn contains_or_touches(&self, point: &Vec2<Millimeters>) -> bool {
72        const EDGE_EPS: Millimeters = Millimeters::new(0.06);
73        if self.contains(point) {
74            return true;
75        }
76        self.segments().any(|edge| {
77            edge.interior_distance(point).is_some_and(|d| d < EDGE_EPS) || point.distance(&edge.from) < EDGE_EPS || point.distance(&edge.to) < EDGE_EPS
78        })
79    }
80
81    /// Whether the ring is simple - no two non-adjacent edges
82    /// cross.
83    pub fn is_simple(&self) -> bool {
84        let edges: Vec<Segment<Millimeters>> = self.segments().collect();
85        let n = edges.len();
86        for i in 0..n {
87            for j in (i + 2)..n {
88                if i == 0 && j == n - 1 {
89                    continue;
90                }
91                if edges[i].crosses(&edges[j]) {
92                    return false;
93                }
94            }
95        }
96        true
97    }
98
99    /// Whether `segment` enters the ring: a strict edge crossing
100    /// or an endpoint inside. Tangent grazes do not count.
101    pub fn crossed(&self, segment: &Segment<Millimeters>) -> bool {
102        self.segments().any(|edge| segment.crosses(&edge)) || self.contains(&segment.from) || self.contains(&segment.to)
103    }
104
105    /// Whether `other` overlaps the ring: an edge crossing or a
106    /// vertex of either inside the other.
107    pub fn overlaps(&self, other: &[Vec2<Millimeters>]) -> bool {
108        let other = Polygon::new(other.to_vec());
109        self.segments().any(|edge| other.segments().any(|other_edge| edge.crosses(&other_edge)))
110            || self.iter().any(|p| other.contains(p))
111            || other.iter().any(|p| self.contains(p))
112    }
113
114    /// Pulls each edge of the ring (positive signed area,
115    /// interior left of each edge) `distances[i]` mm inward.
116    /// Corner vertices are the intersections of the two offset
117    /// edge lines, miter-capped. Not validated - the caller
118    /// checks the result (area, containment).
119    pub fn offset(&self, distances: &[f64]) -> Polygon {
120        let n = self.0.len();
121        let mut lines = Vec::with_capacity(n);
122        for (i, edge) in self.segments().enumerate() {
123            let direction = edge.to - edge.from;
124            let unit = direction.normalized();
125            // Left normal points into the polygon (positive area).
126            lines.push(Line {
127                origin: edge.from + unit.perpendicular() * distances[i],
128                direction: unit,
129            });
130        }
131        Polygon::new(
132            (0..n)
133                .map(|i| {
134                    let previous = (i + n - 1) % n;
135                    let (l1, l2) = (lines[previous], lines[i]);
136                    let corner = self.0[i];
137                    let distance = distances[previous].max(distances[i]);
138                    // Near-parallel fallback: average the two
139                    // normals - the bounded bisector inset.
140                    let bisector = || {
141                        let normal = l1.direction.perpendicular() * distances[previous] + l2.direction.perpendicular() * distances[i];
142                        corner + normal * (distance / normal.magnitude().as_f64().max(1e-9))
143                    };
144                    let Some(inset) = l1.intersect(&l2) else {
145                        return bisector();
146                    };
147                    let miter = inset - corner;
148                    if miter.magnitude().as_f64() <= MITER_CAP * distance {
149                        return inset;
150                    }
151                    // Overshot miter: acute corners legitimately run
152                    // long in the bisector direction - cap them.
153                    // Degenerate corners (near-180° vertices, e.g.
154                    // a `via` station on a straight street) fly
155                    // off sideways - there the bounded bisector
156                    // inset is right.
157                    let bis = bisector();
158                    let miter_bisector = bis - corner;
159                    let aligned = miter.dot::<f64>(&miter_bisector) / (miter.magnitude().as_f64() * miter_bisector.magnitude().as_f64()).max(1e-9);
160                    if aligned > 0.5 {
161                        corner + miter * (MITER_CAP * distance / miter.magnitude().as_f64())
162                    } else {
163                        bis
164                    }
165                })
166                .collect(),
167        )
168    }
169
170    /// The chord of the ring through `point` along `axis`: the
171    /// segment between the nearest outline crossings bracketing
172    /// `point` on the line `point + axis * t` - `None` when
173    /// `point` sits outside.
174    pub fn chord(&self, point: &Vec2<Millimeters>, axis: &Vec2<Millimeters>) -> Option<Segment<Millimeters>> {
175        let epsilon = Millimeters::new(0.01);
176        let (t_lo, t_hi) =
177            self.segments()
178                .filter_map(|edge| edge.param_on_line(point, axis))
179                .fold((Millimeters::new(f64::MIN), Millimeters::INFINITY), |(lo, hi), t| {
180                    (
181                        if t < -epsilon { Millimeters::max(lo, t) } else { lo },
182                        if t > epsilon { Millimeters::min(hi, t) } else { hi },
183                    )
184                });
185        if !t_lo.as_f64().is_finite() || !t_hi.as_f64().is_finite() {
186            return None;
187        }
188        Some(Segment::new(*point + *axis * t_lo.as_f64(), *point + *axis * t_hi.as_f64()))
189    }
190
191    /// The distance from `point` to the nearest ring edge (mm) -
192    /// how far inside the polygon it sits; `deepest_point`
193    /// maximizes it.
194    pub fn clearance(&self, point: &Vec2<Millimeters>) -> Millimeters {
195        self.segments().map(|edge| edge.distance(point)).fold(Millimeters::INFINITY, Millimeters::min)
196    }
197
198    /// The interior point with the largest clearance to the
199    /// ring's edges (sampled on a 7x7 grid) - the spot a label
200    /// sits best.
201    pub fn deepest_point(&self) -> Option<Vec2<Millimeters>> {
202        self.deepest(7, &[])
203    }
204
205    /// Like [`Self::deepest_point`], but skips candidates inside
206    /// `avoid` polygons or within `AVOID_GAP` mm of their edges -
207    /// so a district caption never lands on top of a park name.
208    pub fn deepest_point_avoiding(&self, avoid: &[Polygon]) -> Option<Vec2<Millimeters>> {
209        self.deepest(9, avoid)
210    }
211
212    /// The grid-sampled deepest interior point, optionally
213    /// blocking zones the caption must dodge.
214    fn deepest(&self, steps: usize, avoid: &[Polygon]) -> Option<Vec2<Millimeters>> {
215        const AVOID_GAP: Millimeters = Millimeters::new(3.0);
216        let blocked = |point: &Vec2<Millimeters>| {
217            avoid
218                .iter()
219                .any(|polygon| polygon.contains(point) || polygon.segments().any(|edge| edge.interior_distance(point).is_some_and(|d| d < AVOID_GAP)))
220        };
221        let (min_x, max_x) = self.0.iter().fold((f64::MAX, f64::MIN), |(lo, hi), p| (lo.min(p.x), hi.max(p.x)));
222        let (min_y, max_y) = self.0.iter().fold((f64::MAX, f64::MIN), |(lo, hi), p| (lo.min(p.y), hi.max(p.y)));
223
224        let mut best: Option<(Millimeters, Vec2<Millimeters>)> = None;
225        for row in 1..steps {
226            for col in 1..steps {
227                let point = Vec2::new(min_x + (max_x - min_x) * row as f64 / steps as f64, min_y + (max_y - min_y) * col as f64 / steps as f64);
228                if !self.contains(&point) || blocked(&point) {
229                    continue;
230                }
231                let score = self.clearance(&point);
232                if best.is_none_or(|(best_score, _)| score > best_score) {
233                    best = Some((score, point));
234                }
235            }
236        }
237        best.map(|(_, point)| point)
238    }
239}

241impl<'a> IntoIterator for &'a Polygon {
242    type Item = &'a Vec2<Millimeters>;
243    type IntoIter = std::slice::Iter<'a, Vec2<Millimeters>>;
244
245    fn into_iter(self) -> Self::IntoIter {
246        self.0.iter()
247    }
248}
249
250impl Deref for Polygon {
251    type Target = [Vec2<Millimeters>];
252
253    fn deref(&self) -> &Self::Target {
254        &self.0
255    }
256}
257
258impl FromIterator<Vec2<Millimeters>> for Polygon {
259    fn from_iter<I: IntoIterator<Item = Vec2<Millimeters>>>(iter: I) -> Self {
260        Self::new(iter.into_iter().collect())
261    }
262}
263
264impl From<Vec<Vec2<Millimeters>>> for Polygon {
265    fn from(points: Vec<Vec2<Millimeters>>) -> Self {
266        Self::new(points)
267    }
268}

270impl From<Polygon> for Vec<Vec2<Millimeters>> {
271    fn from(polygon: Polygon) -> Self {
272        polygon.0
273    }
