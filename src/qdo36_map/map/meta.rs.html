19/// Map-wide metadata of the dataset.
20///
21/// Carries the city's identity and rule defaults: Paris ships the
22/// Scotland Yard classics (24 rounds, reveals at 3/8/13/18/24); other
23/// cities may tune both - different street geometry justifies
24/// different pacing.
25#[derive(Debug, Clone, PartialEq, Deserialize, Serialize)]
26pub struct MapMeta {
27    /// The city this dataset describes ("paris", "berlin").
28    pub name: CityName,
29    /// One-line city flavor for the splash/setup screens.
30    #[serde(default)]
31    pub slogan: MapSlogan,
32    /// Short noir framing text (2-3 sentences) for the setup dialog.
33    #[serde(default)]
34    pub story: MapStory,
35    /// The city's fugitive: alias + home station (`[map.fugitive]`).
36    ///
37    /// `None` for generated maps without a designated fugitive.
38    #[serde(default)]
39    pub fugitive: Option<FugitiveSpec>,
40    /// The city's own transport names
41    /// (`[map.transport_names]` - batobus becomes "Spreefähre" in
42    /// Berlin, "Batobus" in Paris).
43    #[serde(default)]
44    pub transport_names: BTreeMap<Transport, TransportName>,
45    /// Transit lines with brand colors
46    /// (`[map.lines]` - "U1" = "#7DAD4C" in Berlin, "M1" = "#FFCD00" in
47    /// Paris); referenced by metro edges.
48    #[serde(default)]
49    pub lines: Palette<LineName, u8>,
50    /// Map coordinate bounds, TOML shape `bounds_mm = [x_min, y_min, x_max, y_max]`.
51    #[serde(rename = "bounds_mm", with = "serde_quad")]
52    pub bounds: Region,
53    /// Maximum number of rounds; the fugitive wins when reached.
54    ///
55    /// Default: the Scotland Yard classic, 24.
56    #[serde(default = "default_max_rounds")]
57    pub max_rounds: RoundNumber,
58    /// The city's detective starting stock (`[map]
59    /// detective_stock`); defaults to the classic 10 taxi / 8 bus /
60    /// 4 metro.
61    #[serde(default = "TicketCounts::detective_start")]
62    pub detective_stock: TicketCounts,
63    /// The city's fugitive starting pool (`[map] fugitive_stock`);
64    /// defaults to unlimited travel + 5 black + 2 double.
65    #[serde(default = "TicketCounts::fugitive_start")]
66    pub fugitive_stock: TicketCounts,
67    /// Rounds at which the fugitive's position is revealed.
68    ///
69    /// Default: the classic schedule `[3, 8, 13, 18, 24]`.
70    #[serde(default = "default_reveal_rounds")]
71    pub reveal_rounds: BTreeSet<RoundNumber>,
72    /// The city's anchor landmark and the station it pins down.
73    ///
74    /// `None` for generated maps without a designated anchor
75    /// (`qdo36-mapgen` output until curated).
76    #[serde(default)]
77    pub anchor: Option<AnchorSpec>,
78}
79

81    RoundNumber::DEFAULT_MAX
