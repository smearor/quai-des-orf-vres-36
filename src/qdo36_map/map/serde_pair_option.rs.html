1//! `serde_pair_option` - `[x, y]` pair serde for `Option<Vec2>`.
2//!
3//! Same flat pair shape as `smearor_commons_geometry::serde_pair`,
4//! but tolerant of an absent position (`pos_mm` missing or `null`).
5
6use serde::Deserialize;
7use serde::Deserializer;
8use serde::Serialize;
9use serde::Serializer;
10use smearor_commons_geometry::Vec2;
11
12/// Serializes `Some(p)` as `[p.x, p.y]`, `None` as nothing.
13pub fn serialize<U, S: Serializer>(pos: &Option<Vec2<U>>, serializer: S) -> Result<S::Ok, S::Error> {
14    match pos {
15        Some(point) => [point.x, point.y].serialize(serializer),
16        None => serializer.serialize_none(),
17    }
18}
19
20/// Deserializes `[x, y]` into `Some(Vec2)`, absent into `None`.
21pub fn deserialize<'de, U, D: Deserializer<'de>>(deserializer: D) -> Result<Option<Vec2<U>>, D::Error> {
22    <Option<[f64; 2]>>::deserialize(deserializer).map(|pair| pair.map(Vec2::from))
23}
