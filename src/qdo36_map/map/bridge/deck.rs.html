18/// How far a deck reaches across its water course, in river
19/// half-widths: stations further out count as off-water, and a
20/// deck axis looks no farther than this for its banks.
21const REACH_HALF_WIDTHS: f64 = 4.0;
22
23/// Extra span beyond the measured bank-to-bank distance - the
24/// apron lands the deck ends on the banks (mm).
25const SPAN_APRON: Millimeters = Millimeters::new(1.2);
26
27/// The drawn bridge of a `bridge`-kind station: a deck
28/// perpendicular to the river, spanning bank to bank. Two cases:
29/// `on_water` stations sit on the water - the deck centers on the
30/// station marker and above-ground (taxi/bus) edges dock at its
31/// ends. Bridgehead stations stand on a bank - the deck centers on
32/// the water below them and their edges stay ordinary roads.
33#[derive(Debug, Clone, Copy, PartialEq)]
34pub struct BridgeDeck {
35    /// Deck center (map mm): the station position when `on_water`,
36    /// its projection on the river course for bridgeheads.
37    pub center: Vec2<Millimeters>,
38    /// Unit deck axis: perpendicular to the river course.
39    pub axis: Vec2<Millimeters>,
40    /// Half the deck length.
41    pub half_span: Millimeters,
42    /// The station itself is the bridge (marker mid-river) - only
43    /// those decks take edge attachments.
44    pub on_water: bool,
45}
46
47impl BridgeDeck {
48    /// The deck at `near` on `course`: perpendicular to the bank
49    /// segment, centered on `station` when it sits on the water,
50    /// spanning bank to bank plus the apron. `None` on a
51    /// degenerate course segment.
52    pub(super) fn at(station: &Station, near: &NearPoint, course: &Polyline, river_half_width: Millimeters) -> Option<Self> {
53        let course_segment = Segment::new(course[near.segment], course[near.segment + 1]);
54        let direction = course_segment.to - course_segment.from;
55        if direction.magnitude() <= Millimeters::new(1e-9) {
56            return None;
57        }
58        let axis = direction.normalized().perpendicular();
59        let on_water = near.distance <= river_half_width;
60        let center = if on_water {
61            station.pos
62        } else {
63            course_segment.from + direction * near.fraction
64        };
65        // The span: where the axis through the center meets each
66        // bank; a short apron lands the ends on the banks.
67        let banks = WaterBanks::new(course, river_half_width);
68        let half_span = banks.crossing_span(center, axis, river_half_width * REACH_HALF_WIDTHS).max(river_half_width) + SPAN_APRON;
69        Some(Self {
70            center,
71            axis,
72            half_span,
73            on_water,
74        })
75    }
76
77    /// Which side of the deck `pos` lies on, measured along
78    /// `axis`.
79    pub fn side_of(&self, pos: Vec2<Millimeters>) -> DeckSide {
80        let along = (pos - self.center).dot::<f64>(&self.axis);
81        if along > 0.0 {
82            DeckSide::Along
83        } else if along < 0.0 {
84            DeckSide::Against
85        } else {
86            DeckSide::Center
87        }
88    }
89
90    /// The deck end on `pos`'s side of the river - where an
91    /// above-ground edge attaches. A centered position lands on
92    /// the `along` end.
93    pub fn end_on_side(&self, pos: Vec2<Millimeters>) -> Vec2<Millimeters> {
94        let side = if self.side_of(pos).factor() < 0.0 { -1.0 } else { 1.0 };
95        self.center + self.axis * (self.half_span.as_f64() * side)
96    }
97
98    /// Both deck ends.
99    pub fn ends(&self) -> (Vec2<Millimeters>, Vec2<Millimeters>) {
100        (self.center - self.axis * self.half_span.as_f64(), self.center + self.axis * self.half_span.as_f64())
101    }
102
103    /// Docks a batobus route's first point onto this deck: drops
104    /// the course points inside the bend zone (the raw route would
105    /// otherwise overshoot to the dock's projection and double
106    /// back - a hairpin), then splices two waypoints upstream of
107    /// the deck center along the river tangent, part way toward
108    /// the bank-side dock end.
109    pub fn dock_start(&self, route: &mut Polyline) {
110        self.dock(route, true);
111    }
112
113    /// Docks a batobus route's last point onto this deck - the
114    /// end counterpart of [`Self::dock_start`].
115    pub fn dock_end(&self, route: &mut Polyline) {
116        self.dock(route, false);
117    }
118
119    /// The docking bend: `at_start` docks the route's first
120    /// point, `false` its last.
121    fn dock(&self, route: &mut Polyline, at_start: bool) {
122        const FAR: Millimeters = Millimeters::new(5.5);
123        let deck = self;
124        let tangent: Vec2<Millimeters> = deck.axis.perpendicular();
125        let t_of = |p: Vec2<Millimeters>| (p - deck.center).dot::<Millimeters>(&tangent);
126        let Some(dock) = (if at_start { route.first() } else { route.last() }).copied() else {
127            return;
128        };
129        let bend = |upstream: Millimeters, bank_frac: f64, side: f64| -> Vec2<Millimeters> {
130            deck.center + tangent * (upstream.as_f64() * side) + (dock - deck.center) * bank_frac
131        };
132        let side_of =
133            |route: &[Vec2<Millimeters>], index: usize| route.get(index).map(|w| if t_of(*w) >= Millimeters::ZERO { 1.0 } else { -1.0 }).unwrap_or(1.0);
134        if at_start {
135            while route.len() > 2 && t_of(route[1]).abs() < FAR {
136                route.0.remove(1);
137            }
138            let side = side_of(route, 1);
139            route.0.splice(1..1, [bend(Millimeters::new(2.0), 0.7, side), bend(FAR, 0.3, side)]);
140        } else {
141            while route.len() > 2 && t_of(route[route.len() - 2]).abs() < FAR {
142                route.0.remove(route.len() - 2);
143            }
144            let side = side_of(route, route.len() - 2);
145            route
146                .0
147                .splice(route.len() - 1..route.len() - 1, [bend(FAR, 0.3, side), bend(Millimeters::new(2.0), 0.7, side)]);
148        }
149    }
150}

152impl GameMap {
153    /// One `BridgeDeck` per `bridge`-kind station near water,
154    /// computed against the smoothed courses the bands draw from.
155    /// A station inside the band is `on_water` (it is the bridge);
156    /// one within `4 * river_half_width` is a bridgehead (deck on
157    /// the water at its projection); anything further is land - its
158    /// `bridge` kind serves the dataset crossing rules only and
159    /// gets no deck.
160    pub fn bridge_decks(&self, river_half_width: Millimeters) -> HashMap<StationId, BridgeDeck> {
161        let water = self.smoothed_water();
162        let mut decks = HashMap::new();
163        for station in self.stations_of_kind(StationKind::Bridge) {
164            // The closest point on any course.
165            let mut best: Option<NearPoint> = None;
166            for (course_i, course) in water.iter().enumerate() {
167                let Some((seg_i, nearest)) = course.nearest_segment(&station.pos) else {
168                    continue;
169                };
170                let distance = station.pos.distance(&nearest.point);
171                if best.as_ref().is_none_or(|best| distance < best.distance) {
172                    best = Some(NearPoint {
173                        distance,
174                        course: course_i,
175                        segment: seg_i,
176                        fraction: nearest.fraction,
177                    });
178                }
179            }
180            let Some(near) = best else { continue };
181            if near.distance > river_half_width * REACH_HALF_WIDTHS {
182                continue;
183            }
184            if let Some(deck) = BridgeDeck::at(station, &near, &water[near.course], river_half_width) {
185                decks.insert(station.id, deck);
186            }
187        }
188        decks
189    }
190
191    /// The drawn path of an above-ground edge (map mm): on-water
192    /// bridge-station endpoints snap to their deck end on the
193    /// neighbor's side - roads meet the bridge instead of drawing
194    /// onto the water. An interior on-water bridge station (`via`)
195    /// enters at the near deck end, passes the station mid-deck and
196    /// leaves at the far end. A hop that crosses water next to a
197    /// bridgehead station bends onto that station's deck, so roads
198    /// always cross on a bridge - never beside it.
199    /// Returns the route points plus, per point, the index of the
200    /// `ids` hop its outgoing segment belongs to - the parallel-
201    /// transports overlay shifts each segment by its hop's offset.
202    pub fn road_route(&self, ids: &[StationId], decks: &HashMap<StationId, BridgeDeck>, water_segments: &[Segment<Millimeters>]) -> RoadRoute {
203        let station_pos = |index: usize| ids.get(index).and_then(|id| self.station(*id)).map(|station| station.pos);
204        let mut route = Vec::with_capacity(ids.len() + 2);
205        let mut hops = Vec::with_capacity(ids.len() + 2);
206        let hop_pair = |hop: usize| StationPair::new(ids[hop], ids[hop + 1]);
207        for (i, id) in ids.iter().enumerate() {
208            let Some(station) = self.station(*id) else { continue };
209            match decks.get(id) {
210                Some(deck) if deck.on_water && (i == 0 || i + 1 == ids.len()) => {
211                    let neighbor = station_pos(if i == 0 { 1 } else { i - 1 }).unwrap_or(station.pos);
212                    route.push(deck.end_on_side(neighbor));
213                    hops.push(hop_pair(i.saturating_sub(1)));
214                }
215                Some(deck) if deck.on_water => {
216                    route.push(deck.end_on_side(station_pos(i - 1).unwrap_or(station.pos)));
217                    route.push(station.pos);
218                    route.push(deck.end_on_side(station_pos(i + 1).unwrap_or(station.pos)));
219                    hops.extend([hop_pair(i - 1), hop_pair(i), hop_pair(i)]);
220                }
221                _ => {
222                    route.push(station.pos);
223                    hops.push(hop_pair(i.min(ids.len().saturating_sub(2))));
224                }
225            }
226            // Bend the next hop onto a bridge: when the drawn segment
227            // crosses water and a hop endpoint owns a bridgehead deck,
228            // the road enters the near deck end and leaves the far one.
229            let (Some(&from), Some(to_raw)) = (route.last(), station_pos(i + 1)) else {
230                continue;
231            };
232            // An on-water endpoint snaps to its deck end - the drawn
233            // hop ends on the bank and crosses nothing.
234            let to = match ids.get(i + 1).and_then(|next| decks.get(next)).filter(|deck| deck.on_water) {
235                Some(deck) => deck.end_on_side(from),
236                None => to_raw,
237            };
238            let road = Segment::new(from, to);
239            let Some(crossing) = water_segments.iter().find_map(|water| road.crossing_point(water)) else {
240                continue;
241            };
242            let Some(deck) = [decks.get(id), ids.get(i + 1).and_then(|next| decks.get(next))]
243                .into_iter()
244                .flatten()
245                .filter(|deck| !deck.on_water)
246                .min_by(|a, b| {
247                    let da = (a.center.x - crossing.x).hypot(a.center.y - crossing.y);
248                    let db = (b.center.x - crossing.x).hypot(b.center.y - crossing.y);
249                    da.partial_cmp(&db).unwrap_or(std::cmp::Ordering::Equal)
250                })
251            else {
252                continue;
253            };
254            route.push(deck.end_on_side(from));
255            route.push(deck.end_on_side(to));
256            hops.extend([hop_pair(i), hop_pair(i)]);
257        }
258        RoadRoute {
259            polyline: Polyline::new(route),
260            hops,
261        }
262    }

265#[cfg(test)]
266mod tests {

269    fn deck() -> BridgeDeck {
270        // River flows +x; deck crosses it vertically.
271        BridgeDeck {
272            center: Vec2::new(50.0, 100.0),
273            axis: Vec2::new(0.0, 1.0),
274            half_span: Millimeters::new(6.0),
275            on_water: true,
276        }
277    }
278
279    #[test]
280    fn ends_are_symmetric_around_the_station() {
281        let (a, b) = deck().ends();
282        assert!((a.x - 50.0).abs() < 1e-9 && (a.y - 94.0).abs() < 1e-9);
283        assert!((b.x - 50.0).abs() < 1e-9 && (b.y - 106.0).abs() < 1e-9);
284    }
285
286    #[test]
287    fn end_on_side_picks_the_neighbors_bank_side() {
288        let deck = deck();
289        let north = deck.end_on_side(Vec2::new(50.0, 80.0));
290        let south = deck.end_on_side(Vec2::new(50.0, 120.0));
291        assert!((north.y - 94.0).abs() < 1e-9);
292        assert!((south.y - 106.0).abs() < 1e-9);
293    }
