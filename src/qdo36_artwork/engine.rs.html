1//! String-level substitution engine for parametrized master SVGs.
2//!
3//! The masters mark their parametrized points with stable element ids
4//! (for example `<stop id="tint-mid" ...>`). Substitution is purely
5//! textual: the tag carrying the id is located, then the attribute value
6//! or the text content inside the tag is replaced.
7
8use crate::error::ArtworkError;
9
10/// Reports whether `position` lies inside an `<!-- ... -->` XML comment.
11pub(crate) fn inside_comment(svg: &str, position: usize) -> bool {
12    match svg[..position].rfind("<!--") {
13        Some(comment_start) => match svg[..position].rfind("-->") {
14            Some(comment_end) => comment_start > comment_end,
15            None => true,
16        },
17        None => false,
18    }
19}

21/// Locates the `<...>` tag carrying `id="..."` and returns its byte range.
22///
23/// Iterates over all occurrences of the id marker and accepts the first one
24/// that sits inside a real element tag — occurrences inside XML comments
25/// (such as the parametrization documentation in the master header) and
26/// inside text content are skipped.
27fn element_tag_range(svg: &str, master: &str, id: &str) -> Result<(usize, usize), ArtworkError> {
28    let missing = || ArtworkError::MissingElement {
29        master: master.to_owned(),
30        id: id.to_owned(),
31    };
32    let id_marker = format!("id=\"{id}\"");
33    let mut search_from = 0;
34    while let Some(relative) = svg[search_from..].find(&id_marker) {
35        let id_position = search_from + relative;
36        search_from = id_position + id_marker.len();
37        if inside_comment(svg, id_position) {
38            continue;
39        }
40        let Some(tag_start) = svg[..id_position].rfind('<') else {
41            continue;
42        };
43        let is_element_tag = svg[tag_start + 1..].starts_with(|c: char| c.is_ascii_alphabetic())
44            && !svg[tag_start..id_position].contains('>')
45            && !inside_comment(svg, tag_start);
46        if !is_element_tag {
47            continue;
48        }
49        let Some(relative_end) = svg[id_position..].find('>') else {
50            continue;
51        };
52        return Ok((tag_start, id_position + relative_end));
53    }
54    Err(missing())
55}

57/// Replaces the value of `attribute` on the element carrying `id`.
58pub(crate) fn set_attribute(svg: &str, master: &str, id: &str, attribute: &str, value: &str) -> Result<String, ArtworkError> {
59    let (tag_start, tag_end) = element_tag_range(svg, master, id)?;
60    let missing = || ArtworkError::MissingAttribute {
61        master: master.to_owned(),
62        id: id.to_owned(),
63        attribute: attribute.to_owned(),
64    };
65    let attribute_marker = format!("{attribute}=\"");
66    let attribute_position = svg[tag_start..tag_end].find(&attribute_marker).ok_or_else(missing)?;
67    let value_start = tag_start + attribute_position + attribute_marker.len();
68    let value_end = svg[value_start..tag_end].find('"').map(|relative| value_start + relative).ok_or_else(missing)?;
69    Ok(format!("{}{value}{}", &svg[..value_start], &svg[value_end..]))
70}

72/// Extracts the element name from a tag slice (for example `g` for `<g ...>`).
73fn element_name(svg: &str, tag_start: usize) -> Option<&str> {
74    let name_start = tag_start + 1;
75    let name_end = svg[name_start..]
76        .find(|c: char| !(c.is_ascii_alphanumeric() || c == '-' || c == '_' || c == ':'))
77        .map(|relative| name_start + relative)?;
78    Some(&svg[name_start..name_end])
79}

81/// Finds the next `<name` opening tag at or after `cursor`, requiring a
82/// non-identifier character after the name (so `<g` does not match
83/// `<gradient`). Returns the tag start and the position after its `>`,
84/// plus whether the tag is self-closing.
85pub(crate) fn find_open_tag(svg: &str, name: &str, cursor: usize) -> Option<(usize, usize, bool)> {
86    let open_marker = format!("<{name}");
87    let mut search_from = cursor;
88    while let Some(relative) = svg[search_from..].find(&open_marker) {
89        let open_start = search_from + relative;
90        let after_name = open_start + open_marker.len();
91        let boundary = svg[after_name..].chars().next();
92        search_from = after_name;
93        if matches!(boundary, Some(c) if c.is_ascii_alphanumeric() || c == '-' || c == '_' || c == ':') {
94            continue;
95        }
96        let tag_end = svg[after_name..].find('>').map(|relative| after_name + relative)?;
97        let self_closing = svg[..tag_end].trim_end().ends_with('/');
98        return Some((open_start, tag_end + 1, self_closing));
99    }
100    None
101}

103/// Replaces the inner markup of the element carrying `id` with `markup`.
104///
105/// Handles arbitrarily nested content including same-name child elements
106/// and self-closing elements. A self-closing target element is rewritten
107/// into an open/close pair.
108pub(crate) fn set_inner(svg: &str, master: &str, id: &str, markup: &str) -> Result<String, ArtworkError> {
109    let missing = || ArtworkError::MissingElement {
110        master: master.to_owned(),
111        id: id.to_owned(),
112    };
113    let (tag_start, tag_end) = element_tag_range(svg, master, id)?;
114    let name = element_name(svg, tag_start).ok_or_else(missing)?;
115    if svg[..tag_end].trim_end().ends_with('/') {
116        let open_tag = svg[tag_start..tag_end].trim_end().trim_end_matches('/').trim_end();
117        return Ok(format!("{}{open_tag}>{markup}</{name}>{}", &svg[..tag_start], &svg[tag_end + 1..]));
118    }
119    let close_marker = format!("</{name}>");
120    let mut depth = 1usize;
121    let mut cursor = tag_end + 1;
122    let content_end = loop {
123        let next_close = svg[cursor..].find(&close_marker).map(|relative| cursor + relative);
124        let Some(close) = next_close else {
125            return Err(missing());
126        };
127        match find_open_tag(svg, name, cursor) {
128            Some((open_start, after_tag, self_closing)) if open_start < close => {
129                if !self_closing {
130                    depth += 1;
131                }
132                cursor = after_tag;
133            }
134            _ => {
135                depth -= 1;
136                if depth == 0 {
137                    break close;
138                }
139                cursor = close + close_marker.len();
140            }
141        }
142    };
143    Ok(format!("{}{markup}{}", &svg[..tag_end + 1], &svg[content_end..]))
144}

146/// Replaces the character data of the `<text>` element carrying `id`.
147pub(crate) fn set_text(svg: &str, master: &str, id: &str, content: &str) -> Result<String, ArtworkError> {
148    let (tag_start, tag_end) = element_tag_range(svg, master, id)?;
149    let missing = || ArtworkError::MissingElement {
150        master: master.to_owned(),
151        id: id.to_owned(),
152    };
153    if !svg[tag_start..tag_end].starts_with("<text") {
154        return Err(missing());
155    }
156    let content_start = tag_end + 1;
157    let content_end = svg[content_start..]
158        .find("</text>")
159        .map(|relative| content_start + relative)
160        .ok_or_else(missing)?;
161    Ok(format!("{}{content}{}", &svg[..content_start], &svg[content_end..]))
162}

164#[cfg(test)]
165mod tests {
166    use super::*;
167
168    const MASTER: &str = r##"<svg><stop id="tint-mid" offset="70%" stop-color="#2563EB" /><text id="rank-letter">I</text></svg>"##;
169
170    #[test]
171    fn set_attribute_replaces_value() {
172        let result = set_attribute(MASTER, "master.svg", "tint-mid", "stop-color", "#DC2626").unwrap_or_default();
173        assert_eq!(
174            result,
175            r##"<svg><stop id="tint-mid" offset="70%" stop-color="#DC2626" /><text id="rank-letter">I</text></svg>"##
176        );
177    }
178
179    #[test]
180    fn set_attribute_fails_on_missing_element() {
181        let result = set_attribute(MASTER, "master.svg", "tint-unknown", "stop-color", "#DC2626");
182        assert!(matches!(result, Err(ArtworkError::MissingElement { .. })));
183    }
184
185    #[test]
186    fn set_attribute_fails_on_missing_attribute() {
187        let result = set_attribute(MASTER, "master.svg", "tint-mid", "fill", "#DC2626");
188        assert!(matches!(result, Err(ArtworkError::MissingAttribute { .. })));
189    }
190
191    #[test]
192    fn set_text_replaces_content() {
193        let result = set_text(MASTER, "master.svg", "rank-letter", "II").unwrap_or_default();
194        assert_eq!(
195            result,
196            r##"<svg><stop id="tint-mid" offset="70%" stop-color="#2563EB" /><text id="rank-letter">II</text></svg>"##
197        );

200    #[test]

206    #[test]
207    fn set_inner_replaces_content() {
208        let master = r##"<svg><g id="glyph"><circle r="1" /></g></svg>"##;
209        let result = set_inner(master, "master.svg", "glyph", "<rect width=\"2\" height=\"2\" />").unwrap_or_default();
210        assert_eq!(result, r##"<svg><g id="glyph"><rect width="2" height="2" /></g></svg>"##);
211    }
212
213    #[test]
214    fn set_inner_handles_nested_same_name_elements() {
215        let master = r##"<svg><g id="glyph"><g><g /></g><circle r="1" /></g><circle r="2" /></svg>"##;
216        let result = set_inner(master, "master.svg", "glyph", "<path d=\"M0,0\" />").unwrap_or_default();
217        assert_eq!(result, r##"<svg><g id="glyph"><path d="M0,0" /></g><circle r="2" /></svg>"##);
218    }

222        let master = r##"<svg><g id="glyph" transform="translate(0,1)" /></svg>"##;
223        let result = set_inner(master, "master.svg", "glyph", "<circle r=\"1\" />").unwrap_or_default();
224        assert_eq!(result, r##"<svg><g id="glyph" transform="translate(0,1)"><circle r="1" /></g></svg>"##);
225    }

227    #[test]
228    fn set_inner_does_not_match_name_prefixes() {

231        assert_eq!(result, r##"<svg><g id="glyph">X</g><circle r="2" /></svg>"##);
232    }

234    #[test]

238    }

240    #[test]
241    fn substitutions_skip_id_occurrences_inside_comments() {
242        let master = r##"<!-- <text id="rank-letter"> doc --><svg><text id="rank-letter" fill="#fef0cd">I</text></svg>"##;
243        let result = set_text(master, "master.svg", "rank-letter", "II").unwrap_or_default();
244        assert_eq!(result, r##"<!-- <text id="rank-letter"> doc --><svg><text id="rank-letter" fill="#fef0cd">II</text></svg>"##);
245        let result = set_attribute(master, "master.svg", "rank-letter", "fill", "#dfbe6f").unwrap_or_default();
246        assert_eq!(result, r##"<!-- <text id="rank-letter"> doc --><svg><text id="rank-letter" fill="#dfbe6f">I</text></svg>"##);
247    }
