1//! Artwork variant generator for Quai des Orfèvres 36.
2//!
3//! Reads parametrized master SVGs and a `variants.toml` table, substitutes
4//! the parametrized element attributes, inner markup and text contents per
5//! variant, and writes one output SVG per variant. Intended for use both as
6//! the `qdo36-artwork` binary and as a build dependency of other workspace
7//! crates (for example inside `crates/app/build.rs`).

31/// Generates all configured variants and returns the written file paths.
32///
33/// `config_path` points at a `variants.toml` table. Master SVG references
34/// inside the table are resolved relative to the directory containing the
35/// configuration file, so each `masters/{city}/{style}/` directory forms a
36/// self-contained style pack of masters plus its own `variants.toml`.
37/// `out_dir` is the directory the generated variants are written to
38/// (created if absent).
39///
40/// # Errors
41///
42/// Returns [`ArtworkError`] when the configuration cannot be read or parsed,
43/// when a master file is missing, when a parametrized element id or
44/// attribute is absent from a master, or when an output file cannot be
45/// written.
46pub fn generate(config_path: &Path, out_dir: &Path) -> Result<Vec<PathBuf>, ArtworkError> {
47    let config_source = fs::read_to_string(config_path).map_err(|source| ArtworkError::Io {
48        path: config_path.to_owned(),
49        source,
50    })?;
51    let spec: Spec = toml::from_str(&config_source)?;
52    let city = spec.city.as_deref().unwrap_or("paris");
53    let style = spec.style.as_deref().unwrap_or("belle-epoque");
54    let masters_dir = config_path.parent().unwrap_or(Path::new("."));
55    #[cfg(feature = "outline")]
56    let fonts = if spec.outline.unwrap_or(true) {
57        let fonts_dir = spec
58            .fonts_dir
59            .as_deref()
60            .map(|dir| masters_dir.join(dir))
61            .ok_or(ArtworkError::MissingFontsDir)?;
62        Some(fonts::FontRegistry::load(&fonts_dir))
63    } else {
64        None
65    };
66    fs::create_dir_all(out_dir).map_err(|source| ArtworkError::Io {
67        path: out_dir.to_owned(),
68        source,
69    })?;
70    let mut written = Vec::new();
71    for asset in &spec.assets {
72        let master_path = masters_dir.join(&asset.master);
73        let master_source = fs::read_to_string(&master_path).map_err(|source| ArtworkError::Io {
74            path: master_path.clone(),
75            source,
76        })?;
77        for variant in &asset.variants {
78            let mut svg = master_source.clone();
79            for substitution in &variant.substitutions {
80                let value = substitution.resolve()?;
81                svg = engine::set_attribute(&svg, &asset.master, &substitution.id, &substitution.attribute, &value)?;
82            }
83            for inner_fill in &variant.fills {
84                svg = engine::set_inner(&svg, &asset.master, &inner_fill.id, &inner_fill.markup)?;
85            }
86            for text_fill in &variant.texts {
87                svg = engine::set_text(&svg, &asset.master, &text_fill.id, &text_fill.content)?;
88            }
89            #[cfg(feature = "outline")]
90            if let Some(fonts) = &fonts {
91                svg = outline::outline_texts(&svg, fonts)?;
92            }
93            let file_name = asset.output.replace("{name}", &variant.name).replace("{city}", city).replace("{style}", style);
94            let out_path = out_dir.join(&file_name);
95            if let Some(parent) = out_path.parent() {
96                fs::create_dir_all(parent).map_err(|source| ArtworkError::Io {
97                    path: parent.to_owned(),
98                    source,
99                })?;
100            }
101            fs::write(&out_path, &svg).map_err(|source| ArtworkError::Io {
102                path: out_path.clone(),
103                source,
104            })?;
105            written.push(out_path);
106        }
107    }
108    #[cfg(feature = "outline")]
109    if let Some(fonts) = &fonts {
110        for static_asset in &spec.statics {
111            let file_name = static_asset.path.replace("{city}", city).replace("{style}", style);
112            let path = out_dir.join(&file_name);
113            let source = fs::read_to_string(&path).map_err(|source| ArtworkError::Io { path: path.clone(), source })?;
114            let outlined = outline::outline_texts(&source, fonts)?;
115            fs::write(&path, &outlined).map_err(|source| ArtworkError::Io { path: path.clone(), source })?;
116        }
117    }
118    Ok(written)
119}
