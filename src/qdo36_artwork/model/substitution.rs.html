1//! Attribute substitution declaration.
2
3use serde::Deserialize;
4use smearor_commons_color::Color;
5
6/// Default blend factor toward white for luminance-based lightening.
7const DEFAULT_LIGHTEN_FACTOR: f64 = 0.45;
8
9/// One attribute substitution on a parametrized element.
10#[derive(Debug, Deserialize)]
11#[serde(deny_unknown_fields)]
12pub struct Substitution {
13    /// Id of the parametrized element in the master.
14    pub id: String,
15    /// Attribute whose value is replaced (for example `stop-color`).
16    pub attribute: String,
17    /// `#rrggbb` colour value written to the attribute.
18    pub value: Color<u8>,
19    /// Optional luminance threshold; colours darker than this are lightened
20    /// by `lighten_factor` instead of being written verbatim.
21    #[serde(default)]
22    pub lighten_if_below: Option<f64>,
23    /// Blend factor toward white applied by `lighten_if_below` (0.0..=1.0).
24    #[serde(default = "default_lighten_factor")]
25    pub lighten_factor: f64,
26}
27
28/// Serde default for [`Substitution::lighten_factor`].
29fn default_lighten_factor() -> f64 {
30    DEFAULT_LIGHTEN_FACTOR
31}
32
33impl Substitution {
34    /// Resolves the effective attribute colour, applying the optional
35    /// luminance-based lightening for dark colours.
36    pub fn resolve(&self) -> Color<u8> {
37        match self.lighten_if_below {
38            Some(threshold) if self.value.luminance_601().as_f64() * 255.0 < threshold => self.value.lighten(self.lighten_factor),
39            _ => self.value,
40        }
41    }
42}
