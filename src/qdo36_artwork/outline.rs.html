1//! Text-to-path outlining for generated SVGs.
2//!
3//! Every `<text>` element in the produced variant is shaped with the
4//! bundled OFL fonts and replaced by an equivalent `<g><path .../></g>`
5//! group, so the shipped artwork carries no font dependency at all.
6//!
7//! Limitations (documented contract): `<tspan>`, `dy`/`textLength`,
8//! multi-line text and `x`/`y` coordinate lists are not supported — the
9//! masters use single-anchor single-line labels only. Elements containing
10//! child markup or occurring inside XML comments are left untouched.
11

19/// Style attributes carried over from the `<text>` element to the emitted
20/// outline group; font/layout attributes are consumed by the pass.
21const STYLE_ATTRIBUTES: &[&str] = &[
22    "id",
23    "fill",
24    "fill-opacity",
25    "stroke",
26    "stroke-width",
27    "stroke-linecap",
28    "stroke-linejoin",
29    "stroke-dasharray",
30    "paint-order",
31    "opacity",
32    "filter",
33    "transform",
34];

36/// Reads `name="..."` out of an element tag slice.
37fn attribute<'a>(tag: &'a str, name: &str) -> Option<&'a str> {
38    let marker = format!("{name}=\"");
39    let start = tag.find(&marker)? + marker.len();
40    let end = tag[start..].find('"')? + start;
41    Some(&tag[start..end])
42}

44/// Decodes the handful of XML entities used in text content.
45fn decode_entities(text: &str) -> String {
46    text.replace("&lt;", "<")
47        .replace("&gt;", ">")
48        .replace("&amp;", "&")
49        .replace("&quot;", "\"")
50        .replace("&#39;", "'")
51}

53/// Parses an SVG length list and returns its first numeric component.
54fn first_length(tag: &str, name: &str, default: f64) -> f64 {
55    attribute(tag, name)
56        .and_then(|value| value.split([',', ' ']).find(|part| !part.is_empty()).and_then(|part| part.parse().ok()))
57        .unwrap_or(default)
58}

60/// Accumulates font-unit path commands as absolute user-space `d` data.
61struct PathBuilder {
62    d: String,
63    /// User-space x of the glyph origin (anchor + pen + x-offset).
64    origin_x: f64,
65    /// User-space y of the glyph baseline (baseline - y-offset).
66    baseline_y: f64,
67    /// Font-units to user-units scale (`font-size / units-per-em`).
68    scale: f64,
69}
70
71impl PathBuilder {
72    fn x(&self, x: f32) -> f64 {
73        self.origin_x + f64::from(x) * self.scale
74    }
75
76    fn y(&self, y: f32) -> f64 {
77        self.baseline_y - f64::from(y) * self.scale
78    }
79}
80
81impl OutlineBuilder for PathBuilder {
82    fn move_to(&mut self, x: f32, y: f32) {
83        self.d.push_str(&format!("M{:.2},{:.2} ", self.x(x), self.y(y)));
84    }
85
86    fn line_to(&mut self, x: f32, y: f32) {
87        self.d.push_str(&format!("L{:.2},{:.2} ", self.x(x), self.y(y)));
88    }
89
90    fn quad_to(&mut self, x1: f32, y1: f32, x: f32, y: f32) {
91        self.d
92            .push_str(&format!("Q{:.2},{:.2} {:.2},{:.2} ", self.x(x1), self.y(y1), self.x(x), self.y(y)));
93    }
94
95    fn curve_to(&mut self, x1: f32, y1: f32, x2: f32, y2: f32, x: f32, y: f32) {
96        self.d.push_str(&format!(
97            "C{:.2},{:.2} {:.2},{:.2} {:.2},{:.2} ",
98            self.x(x1),
99            self.y(y1),
100            self.x(x2),
101            self.y(y2),
102            self.x(x),
103            self.y(y)
104        ));
105    }
106
107    fn close(&mut self) {
108        self.d.push('Z');
109    }
110}

112/// Replaces every `<text>` element in `svg` with outlined glyph paths.
113///
114/// `fonts` resolves `font-family`/`font-weight` pairs against the bundled
115/// font directory. Anchor handling (`text-anchor="start|middle|end"`) and
116/// `letter-spacing` are honoured; style attributes are carried onto the
117/// emitted `<g>` so strokes, halos and filters keep working.
118pub(crate) fn outline_texts(svg: &str, fonts: &FontRegistry) -> Result<String, ArtworkError> {
119    let mut out = String::with_capacity(svg.len() + svg.len() / 2);
120    let mut cursor = 0usize;
121    while let Some((open_start, tag_end, self_closing)) = find_open_tag(svg, "text", cursor) {
122        if inside_comment(svg, open_start) || self_closing {
123            out.push_str(&svg[cursor..tag_end]);
124            cursor = tag_end;
125            continue;
126        }
127        let tag = &svg[open_start..tag_end];
128        let Some(content_end) = svg[tag_end..].find("</text>").map(|relative| tag_end + relative) else {
129            out.push_str(&svg[cursor..tag_end]);
130            cursor = tag_end;
131            continue;
132        };
133        let element_end = content_end + "</text>".len();
134        out.push_str(&svg[cursor..open_start]);
135        let content = &svg[tag_end..content_end];
136        if content.contains('<') {
137            out.push_str(&svg[open_start..element_end]);
138        } else {
139            out.push_str(&outline_text_element(tag, &decode_entities(content.trim()), fonts)?);
140        }
141        cursor = element_end;
142    }
143    out.push_str(&svg[cursor..]);
144    Ok(out)
145}

147/// Outlines a single `<text>` element into a `<g><path/></g>` group.
148fn outline_text_element(tag: &str, content: &str, fonts: &FontRegistry) -> Result<String, ArtworkError> {
149    let family = attribute(tag, "font-family").unwrap_or("sans-serif");
150    let weight = match attribute(tag, "font-weight").unwrap_or("normal") {
151        "bold" | "bolder" => 700u16,
152        other => other.parse::<u16>().unwrap_or(400),
153    };
154    let font_size = first_length(tag, "font-size", 16.0);
155    let x = first_length(tag, "x", 0.0);
156    let y = first_length(tag, "y", 0.0);
157    let letter_spacing = first_length(tag, "letter-spacing", 0.0);
158    let anchor = attribute(tag, "text-anchor").unwrap_or("start");
159    let face_id = fonts.face_id(family, weight)?;
160    let paths = fonts.with_face_data(face_id, |data, index| {
161        let Some(shaper) = rustybuzz::Face::from_slice(data, index) else {
162            return String::new();
163        };
164        let Ok(outliner) = rustybuzz::ttf_parser::Face::parse(data, index) else {
165            return String::new();
166        };
167        let scale = font_size / f64::from(shaper.units_per_em());
168        let mut buffer = rustybuzz::UnicodeBuffer::new();
169        buffer.push_str(content);
170        buffer.set_direction(rustybuzz::Direction::LeftToRight);
171        let shaped = rustybuzz::shape(&shaper, &[], buffer);
172        let positions = shaped.glyph_positions();
173        let count = positions.len();
174        let spacing_total = letter_spacing * (count.saturating_sub(1)) as f64;
175        let advance_total: f64 = positions.iter().map(|position| position.x_advance as f64 * scale).sum::<f64>() + spacing_total;
176        let mut origin_x = x;
177        if anchor == "middle" {
178            origin_x -= advance_total / 2.0;
179        } else if anchor == "end" {
180            origin_x -= advance_total;
181        }
182        let mut pen_x = 0.0f64;
183        let mut paths = String::new();
184        for (index, (info, position)) in shaped.glyph_infos().iter().zip(positions.iter()).enumerate() {
185            let glyph_id = GlyphId(info.glyph_id as u16);
186            let mut builder = PathBuilder {
187                d: String::new(),
188                origin_x: origin_x + pen_x + f64::from(position.x_offset) * scale,
189                baseline_y: y - f64::from(position.y_offset) * scale,
190                scale,
191            };
192            if outliner.outline_glyph(glyph_id, &mut builder).is_some() {
193                paths.push_str(&builder.d);
194            }
195            pen_x += position.x_advance as f64 * scale;
196            if index + 1 < count {
197                pen_x += letter_spacing;
198            }
199        }
200        paths
201    })?;
202    let mut attributes = String::new();
203    for name in STYLE_ATTRIBUTES {
204        if let Some(value) = attribute(tag, name) {
205            attributes.push_str(&format!(" {name}=\"{value}\""));
206        }
207    }
208    Ok(format!("<g{attributes}><path d=\"{}\"/></g>", paths.trim_end()))
209}

211#[cfg(test)]
212mod tests {
213    use super::*;
214    use std::path::PathBuf;
215
216    fn fonts() -> FontRegistry {
217        let dir = PathBuf::from(concat!(env!("CARGO_MANIFEST_DIR"), "/../app/resources/fonts"));
218        FontRegistry::load(&dir)
219    }
220
221    #[test]
222    fn outlines_text_element() {
223        let svg = r##"<svg><text x="10" y="20" font-family="'Patrick Hand'" font-size="24" fill="#000">M1</text></svg>"##;
224        let result = outline_texts(svg, &fonts()).unwrap_or_default();
225        assert!(!result.contains("<text"));
226        assert!(result.contains("<path d=\"M"));
227        assert!(result.contains("fill=\"#000\""));
228    }
229
230    #[test]
231    fn honours_middle_anchor() {
232        let svg = r##"<svg><text x="100" y="20" font-family="'Orbitron'" font-size="20" font-weight="bold" text-anchor="middle">M</text></svg>"##;
233        let result = outline_texts(svg, &fonts()).unwrap_or_default();
234        assert!(result.contains("<path d=\"M"));
235        assert!(!result.contains("<text"));
236    }
237
238    #[test]
239    fn keeps_halo_stroke_on_group() {
240        let svg = r##"<svg><text x="0" y="9" font-family="'Comic Neue'" font-size="26" fill="#111" stroke="#eee" stroke-width="3" paint-order="stroke">M1</text></svg>"##;
241        let result = outline_texts(svg, &fonts()).unwrap_or_default();
242        assert!(result.contains("stroke=\"#eee\""));
243        assert!(result.contains("paint-order=\"stroke\""));
244    }
245
246    #[test]
247    fn leaves_tspan_content_untouched() {
248        let svg = r##"<svg><text x="0" y="0"><tspan>A</tspan></text></svg>"##;
249        let result = outline_texts(svg, &fonts()).unwrap_or_default();
250        assert!(result.contains("<tspan"));
251    }
252
253    #[test]
254    fn is_idempotent_on_outlined_input() {
255        let fonts = fonts();
256        let svg = r##"<svg><text x="0" y="0" font-family="'Comic Neue'">A</text></svg>"##;
257        let once = outline_texts(svg, &fonts).unwrap_or_default();
258        let twice = outline_texts(&once, &fonts).unwrap_or_default();
259        assert_eq!(once, twice);
260    }
