1//! Bundled font registry used by the text-outlining pass.
2//!
3//! Loads every TTF found below the shared `resources/fonts` directory into
4//! a [`fontdb::Database`] and resolves CSS-style `font-family` stacks plus
5//! `font-weight` values to concrete faces for shaping.
6
7use crate::error::ArtworkError;
8use std::path::Path;
9
10/// Registry of bundled OpenType faces resolved by family name and weight.
11pub(crate) struct FontRegistry {
12    db: fontdb::Database,
13}
14
15impl FontRegistry {
16    /// Loads every font file found below `fonts_dir` (a missing directory
17    /// yields an empty registry, which later fails per family).
18    pub(crate) fn load(fonts_dir: &Path) -> Self {
19        let mut db = fontdb::Database::new();
20        db.load_fonts_dir(fonts_dir);
21        Self { db }
22    }
23
24    /// Resolves the first available family in `family_list` at `weight` to
25    /// a face identifier. Generic families (`sans-serif`, `serif`,
26    /// `monospace`) are honoured as last-resort fallbacks.
27    pub(crate) fn face_id(&self, family_list: &str, weight: u16) -> Result<fontdb::ID, ArtworkError> {
28        let family_list = family_list.replace("&#39;", "'").replace("&quot;", "\"");
29        for family_name in family_list.split(',') {
30            let name = family_name.trim().trim_matches(|c| c == '\'' || c == '"');
31            if name.is_empty() {
32                continue;
33            }
34            let family = match name {
35                "sans-serif" => fontdb::Family::SansSerif,
36                "serif" => fontdb::Family::Serif,
37                "monospace" => fontdb::Family::Monospace,
38                "cursive" => fontdb::Family::Cursive,
39                other => fontdb::Family::Name(other),
40            };
41            let query = fontdb::Query {
42                families: &[family],
43                weight: fontdb::Weight(weight),
44                ..fontdb::Query::default()
45            };
46            if let Some(id) = self.db.query(&query) {
47                return Ok(id);
48            }
49        }
50        Err(ArtworkError::MissingFontFamily { family: family_list })
51    }
52
53    /// Runs `f` with the raw face data of `id`; the data only lives for the
54    /// callback's duration, so every font operation (shaping, glyph
55    /// extraction) must happen inside `f`.
56    pub(crate) fn with_face_data<R>(&self, id: fontdb::ID, f: impl FnOnce(&[u8], u32) -> R) -> Result<R, ArtworkError> {
57        self.db
58            .with_face_data(id, f)
59            .ok_or_else(|| ArtworkError::MissingFontFamily { family: format!("id {id}") })
60    }
61}
